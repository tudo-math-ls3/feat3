91 template<typename DataType_, typename IndexType_, typename Space_>
92 void assemble(LAFEM::UnitFilter<DataType_, IndexType_>& filter, const Space_& space) const
93 {
94 // create a sparse vector factory of correct size
95 LAFEM::SparseVectorFactory<DataType_, IndexType_> svf(space.get_num_dofs());
96
97 // add existing filter entries to factory
98 if(!filter.empty())
99 {
100 XASSERT(filter.size() == svf.size());
101 if(!filter.hollow())
102 svf.add(filter.get_filter_vector());
103 }
104
105 // assemble filter vector
106 Intern::UnitAsmWrapper<shape_dim>::assemble(svf, space, _cells);
107
108 // assign filter vector
109 filter.get_filter_vector() = svf.make_sv();
110 }

346 template<typename DT_, typename IT_, int block_size_, typename Space_>
347 static void assemble(LAFEM::SparseVectorFactory<DT_, IT_, block_size_>& svf, const Space_& space, const std::set<Index>& cells)
348 {
349 // create a dof-assignment object
350 typename Space_::template DofAssignment<shape_dim_>::Type dof_assign(space);
351
352 Tiny::Vector<DT_, block_size_> null_val;
353 null_val.format();
354
355 // loop over all target indices
356 std::set<Index>::const_iterator it(cells.begin());
357 std::set<Index>::const_iterator jt(cells.end());
358 for(; it != jt; ++it)
359 {
360 dof_assign.prepare(*it);
361 const int num_assign(dof_assign.get_num_assigned_dofs());
362 for(int j(0); j < num_assign; ++j)
363 {
364 svf.add(IT_(dof_assign.get_index(j)), null_val);
365 }
366 dof_assign.finish();
367 }
368 }

371 template<typename DT_, typename IT_, typename Space_, typename Function_>
372 static void assemble(
373 LAFEM::SparseVectorFactory<DT_, IT_, 1>& svf,
374 const Space_& space,
375 const std::set<Index>& cells,
376 const Function_& function)
377 {
378 // create a node-functional object
379 typedef typename Space_::template NodeFunctional<shape_dim_, DT_>::Type NodeFunc;
380
381 // check for empty node functional set
382 static constexpr Index max_dofs = NodeFunc::max_assigned_dofs;
383 if(max_dofs <= 0)
384 return;
385
386 // create node functional
387 NodeFunc node_func(space);
388
389 // create a dof-assignment object
390 typename Space_::template DofAssignment<shape_dim_, DT_>::Type dof_assign(space);
391
392 // create node data; avoid zero-length vectors
393 Tiny::Vector<DT_, max_dofs+1> node_data;
394
395 // loop over all target indices
396 std::set<Index>::const_iterator it(cells.begin());
397 std::set<Index>::const_iterator jt(cells.end());
398 for(; it != jt; ++it)
399 {
400 node_func.prepare(*it);
401 node_func(node_data, function);
402 node_func.finish();
403
404 dof_assign.prepare(*it);
405
406 const int num_assign(dof_assign.get_num_assigned_dofs());
407 for(int j(0); j < num_assign; ++j)
408 {
409 svf.add(dof_assign.get_index(j), node_data[j]);
410 }
411 dof_assign.finish();
412 }
