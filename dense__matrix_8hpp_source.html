254 void read_from(FileMode mode, const String& filename)
255 {
256 std::ios_base::openmode bin = std::ifstream::in | std::ifstream::binary;
257 if(mode == FileMode::fm_mtx)
258 bin = std::ifstream::in;
259 std::ifstream file(filename.c_str(), bin);

272 void read_from(FileMode mode, std::istream& file)
273 {
274 this->clear();
275
276 switch(mode)
277 {
278 case FileMode::fm_mtx:
279 {
280 Index trows, tcols;
281 String line;
282 std::getline(file, line); // !!? Test on overflow error... could be an enormous matrix... !??
283 //for now, just array real general (aka dense) matrices
284 const bool array_format((line.find("%%MatrixMarket matrix array real general") != String::npos) ? true : false);
285 if (array_format == false)
286 {
287 XABORTM("Input-file is not a compatible array real mtx-file");
288 }
289
290 while(!file.eof())
291 {
292 std::getline(file,line);
293 if (file.eof())
294 XABORTM("Input-file is empty");
295
296 String::size_type begin(line.find_first_not_of(" "));
297 if (line.at(begin) != '%')
298 break;
299 }
300 //Read in number of num_rows and num_cols
301 {
302 String::size_type begin(line.find_first_not_of(" "));
303 line.erase(0, begin);
304 String::size_type end(line.find_first_of(" "));
305 String srow(line, 0, end);
306 trows = Index(atol(srow.c_str()));
307 line.erase(0, end);
308
309 begin = line.find_first_not_of(" ");
310 line.erase(0, begin);
311 end = line.find_first_of(" ");
312 String scol(line, 0, end);
313 tcols = Index(atol(scol.c_str()));
314 line.erase(0, end);
315 }
316
317 DenseMatrix<DT_, IT_> tmp(Index(trows), tcols);
318 Index i(0);
319 Memory::TypedView<DT_> tmp_view(tmp.elements_view_w());
320
321 //Read in value of lines:
322 while(!file.eof())
323 {
324 std::getline(file, line);
325 if(file.eof())
326 break;
327
328 String::size_type begin = line.find_first_not_of(" ");
329 line.erase(0, begin);
330 String::size_type end = line.find_first_of(" ");
331 String sval(line, 0, end);
332 DT_ tval((DT_)atof(sval.c_str()));
333
334 tmp_view[i] = tval;
335
336 ++i;
337 }
338 XASSERTM(i == trows * tcols, "Dense MTX file did not contain enough entries!");
339
340 tmp_view.release();
341 this->move(std::move(tmp));
342
343 }
344 break;
345
346 case FileMode::fm_dm:
347 [[fallthrough]];
348
349 case FileMode::fm_binary:
350 this->template _deserialize<double, std::uint64_t>(FileMode::fm_dm, file);
351 break;
352
353 default:
354 XABORTM("Filemode not supported!");
355 }
356 }

364 void write_out(FileMode mode, const String& filename) const
365 {
366 std::ios_base::openmode bin = std::ofstream::out | std::ofstream::binary;
367 if(mode == FileMode::fm_mtx)
368 bin = std::ofstream::out;
369 std::ofstream file;
370 char* buff = nullptr;
371 if(mode == FileMode::fm_mtx)
372 {
373 buff = new char[LAFEM::file_out_stream_buffer_size];
374 file.rdbuf()->pubsetbuf(buff, LAFEM::file_out_stream_buffer_size);
375 }
376 file.open(filename.c_str(), bin);
377 if(! file.is_open())
378 XABORTM("Unable to open Matrix file " + filename);
379 write_out(mode, file);
380 file.close();
381 delete[] buff;
382 }

390 void write_out(FileMode mode, std::ostream& file) const
391 {
392 switch(mode)
393 {
394 case FileMode::fm_mtx:
395 {
396 file << "%%MatrixMarket matrix array real general\n";
397 file << this->num_rows() << " " << this->num_cols() << " " << this->num_nzes() << "\n";
398 file << std::scientific << std::setprecision(Type::Traits<DT_>::format_precision);
399
400 Memory::TypedView<DT_> elem_view(this->elements_view_r());
401 Index tsize = this->num_rows() * this->num_cols();
402 for (Index i(0); i < tsize; ++i)
403 {
404 file << elem_view(i) << "\n";
405 }
406 break;
407 }
408
409 case FileMode::fm_dm:
410 [[fallthrough]];
411
412 case FileMode::fm_binary:
413 this->template _serialize<double, std::uint64_t>(FileMode::fm_dm, file);
414 break;
415
416 default:
417 XABORTM("Filemode not supported!");
418 }

878 TimeStamp ts_stop;
879
