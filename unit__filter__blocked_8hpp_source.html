389 template<int block_width_>
390 void filter_mat(SparseMatrixBCSR<DT_, IT_, block_size_, block_width_> & matrix) const
391 {
392 if(_sv.hollow())
393 return;
394
395 XASSERTM(_sv.size() == matrix.num_rows(), "Matrix size does not match!");
396
397 Arch::UnitFilterBlockBCSR::template exec<DT_, IT_, block_size_, block_width_>(matrix.val_arbiter(), matrix.row_ptr_arbiter(),
398 matrix.col_idx_arbiter(), _sv.elements_arbiter(), _sv.indices_arbiter(), _sv.num_nzes(), true, _ignore_nans);
399 }

437 template<int block_width_>
438 void filter_weak_matrix_rows(SparseMatrixBCSR<DT_, IT_, block_size_, block_width_> & matrix_a,
439 const SparseMatrixBCSR<DT_, IT_, block_size_, block_width_>& matrix_m) const
440 {
441 if(_sv.hollow())
442 return;
443
444 XASSERTM(matrix_a.val_arbiter() != matrix_m.val_arbiter(), "Matrices are not allowed to hold the same data");
445
446 XASSERTM(_sv.size() == matrix_a.num_rows(), "Matrix size does not match!");
447 XASSERTM(_sv.size() == matrix_m.num_rows(), "Matrix size does not match!");
448 XASSERTM(matrix_a.num_cols() == matrix_m.num_cols(), "matrix A and M must share their layout");
449 XASSERTM(matrix_a.num_nzes() == matrix_m.num_nzes(), "matrix A and M must share their layout");
450
451 Arch::UnitFilterBlockWeakBCSR::template exec<DT_, IT_, block_size_, block_width_>(
452 matrix_a.val_arbiter(), matrix_m.val_arbiter(), matrix_a.row_ptr_arbiter(),
453 _sv.elements_arbiter(), _sv.indices_arbiter(), _sv.num_nzes(), _ignore_nans);
454 }
