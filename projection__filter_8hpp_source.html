28 template<typename DataType_>
29 struct ClosestPointHelpers
30 {
31 template<int dim>
32 using Vec = Tiny::Vector<DataType_, dim>;
33
34 template<int dim_>
35 static Vec<dim_> on_segment(const Vec<dim_>& a, const Vec<dim_>& b, const Vec<dim_>& p)
36 {
37 const Vec<dim_> ab = b - a;
38 const Vec<dim_> ap = p - a;
39
40 const DataType_ alpha = Math::clamp(Tiny::dot(ab, ap) / Tiny::dot(ab, ab), 0.0, 1.0);
41 return a + alpha * ab;
42 }
43
44 static Vec<3> on_triangle(const Vec<3>& a, const Vec<3>& b, const Vec<3>& c, const Vec<3>& p)
45 {
46 // https://stackoverflow.com/a/74395029
47 const Vec<3> ab = b - a;
48 const Vec<3> ac = c - a;
49 const Vec<3> ap = p - a;
50
51 const DataType_ d1 = dot(ab, ap);
52 const DataType_ d2 = dot(ac, ap);
53 if (d1 <= 0.0 && d2 <= 0.0)
54 {
55 return a; //#1
56 }
57
58 const Vec<3> bp = p - b;
59 const DataType_ d3 = dot(ab, bp);
60 const DataType_ d4 = dot(ac, bp);
61 if (d3 >= 0.0 && d4 <= d3)
62 {
63 return b; //#2
64 }
65
66 const Vec<3> cp = p - c;
67 const DataType_ d5 = dot(ab, cp);
68 const DataType_ d6 = dot(ac, cp);
69 if (d6 >= 0.0 && d5 <= d6)
70 {
71 return c; //#3
72 }
73
74 const DataType_ vc = (d1 * d4) - (d3 * d2);
75 if (vc <= 0.0 && d1 >= 0.0 && d3 <= 0.0)
76 {
77 const DataType_ v = d1 / (d1 - d3);
78 return a + v * ab; //#4
79 }
80
81 const DataType_ vb = (d5 * d2) - (d1 * d6);
82 if (vb <= 0.0 && d2 >= 0.0 && d6 <= 0.0)
83 {
84 const DataType_ v = d2 / (d2 - d6);
85 return a + v * ac; //#5
86 }
87
88 const DataType_ va = (d3 * d6) - (d5 * d4);
89 if (va <= 0.0 && (d4 - d3) >= 0.0 && (d5 - d6) >= 0.0)
90 {
91 const DataType_ v = (d4 - d3) / ((d4 - d3) + (d5 - d6));
92 return b + v * (c - b); //#6
93 }
94
95 const DataType_ denom = 1.0 / (va + vb + vc);
96 const DataType_ v = vb * denom;
97 const DataType_ w = vc * denom;
98 return a + v * ab + w * ac; //#0
99 }
100 };

180 {

185 static VertexType on_facet(const MeshType& mesh, Index facet, const VertexType& p)
186 {
187 const auto& vertices = mesh.get_vertex_set();
188 const auto& indices = mesh.template get_index_set<2, 0>();
189 const VertexType a = vertices[indices(facet, 0)];
190 const VertexType b = vertices[indices(facet, 1)];
191 const VertexType c = vertices[indices(facet, 2)];
192 const VertexType d = vertices[indices(facet, 3)];
193
194 auto p1 = ClosestPointHelpers<typename MeshType::CoordType>::on_triangle(a, b, c, p);
195 auto p2 = ClosestPointHelpers<typename MeshType::CoordType>::on_triangle(c, b, d, p);
196
197 if((p1 - p).norm_euclid_sqr() < (p2 - p).norm_euclid_sqr())
198 {
199 return p1;
200 }
201 else
202 {
203 return p2;
204 }
205 }

209 template<typename MeshType_, int dim_ = MeshType_::world_dim - 1>
210 typename MeshType_::VertexType project_onto_mesh_part(
211 const MeshType_& mesh,
212 const Geometry::MeshPart<MeshType_>& mesh_part,
213 const typename MeshType_::VertexType& p)
214 {
215 using DataType = typename MeshType_::CoordType;
216 using VertexType = typename MeshType_::VertexType;
217
218 using FacetShape = typename Shape::FaceTraits<typename MeshType_::ShapeType, dim_>::ShapeType;
219
220 const auto& vertices = mesh.get_vertex_set();
221
222 VertexType closest_point = vertices[0];
223 if constexpr (dim_ > 0)
224 {
225 closest_point = project_onto_mesh_part<MeshType_, dim_ - 1>(mesh, mesh_part, p);
226 }
227
228 const auto& target_set = mesh_part.template get_target_set<dim_>();
229 DataType distance_to_closest = (p - closest_point).norm_euclid_sqr();
230 for(Index i(0); i < mesh_part.get_num_entities(dim_); i++)
231 {
232 VertexType candidate = ClosestPoint<FacetShape, MeshType_>::on_facet(mesh, target_set[i], p);
233 DataType distance = (p - candidate).norm_euclid_sqr();
234 if(distance < distance_to_closest)
235 {
236 closest_point = candidate;
237 distance_to_closest = distance;
238 }
239 }
240
241 return closest_point;
242 }

424 private:
425 void _filter(VectorType& vector) const
426 {
427 Memory::TypedView<ValueType> view = vector.elements_view_rw();
428 for(auto& [dof, position, type] : _entries)
429 {
430 ValueType velocity = view[dof];
431
432 DataType speed = velocity.norm_euclid();
433
434 DataType scale = 0.01 / speed;
435
436 // Add velocity to location
437 ValueType displaced = position + scale * velocity;
438
439 // Project onto surface
440 ValueType projected;
441 switch(type)
442 {
443 case EntryType::Boundary:
444 projected = Intern::project_onto_mesh_part(*_mesh, _neumann_zero_boundary, displaced);
445 break;
446 case EntryType::Feature:
447 projected = Intern::project_onto_mesh_part(*_mesh, _feature_network, displaced);
448 break;
449 case EntryType::Fixed:
450 projected = position;
451 break;
452 }
453
454 // Determine new velocity
455 view[dof] = (projected - position) * (1.0 / scale);
456 }
457 }
