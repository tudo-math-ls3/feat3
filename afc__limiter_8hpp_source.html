272 void stiff_mat_to_m()
273 {
274 XASSERTM(_init_symbolic_called, "Need to call init_symbolic() first");
275 XASSERTM(_matrix_stiff_set, "stiffnes matrix not set");
276 auto mat_l_plus = _matrix_stiff.local().clone(LAFEM::CloneMode::Weak);
277 mat_l_plus.format();
278 auto mat_stiff_t = _matrix_stiff.local().transpose();
279
280 const Memory::TypedView<IndexType> row_ptr_view = _matrix_mass.local().row_ptr_view_r();
281 const Memory::TypedView<IndexType> col_idx_view = _matrix_mass.local().col_idx_view_r();
282 const Memory::TypedView<DataType> val_stiff = _matrix_stiff.local().val_view_r();
283 const Memory::TypedView<DataType> val_stiff_t = mat_stiff_t.val_view_r();
284 Memory::TypedView<DataType> val_plus = mat_l_plus.val_view_rw();
285 for(IndexType row = 0; row < _num_rows; ++row)
286 {
287 const IndexType start = row_ptr_view[row];
288 const IndexType end = row_ptr_view[row + 1];
289
290 IndexType diag = end;
291 DataType lump = DataType(0);
292
293 for(IndexType k = start; k < end; ++k)
294 {
295 if(col_idx_view[k] == row)
296 diag = k;
297 else
298 {
299 auto val = Math::max(DataType(0), Math::max(val_stiff[k], val_stiff_t[k]));
300 val_plus[k] += val;
301 lump += val;
302 }
303 }
304 val_plus[diag] += -lump;
305 }
306 // 0 to 1 to 0???!!!
307 _matrix_stiff.local().axpy(mat_l_plus, -1);
308 }

386 void init_symbolic()
387 {
388 XASSERTM(_matrix_mass_set && _matrix_conv_set, "Massmatrix and convection matrix must be set!");
389 watch_init.start();
390
391 // symbolic init, structure of matrices must not be changend after this
392
393 // matrix structures for internal matrices
394 _matrix_hlp = _matrix_mass.clone(LAFEM::CloneMode::Layout);
395 _matrix_right = _matrix_mass.clone(LAFEM::CloneMode::Layout);
396 _matrix_art_diff = _matrix_mass.clone(LAFEM::CloneMode::Layout);
397
398 // only need matrix_left for implicit time stepping
399 if(_theta > 0)
400 _matrix_left = _matrix_mass.clone(LAFEM::CloneMode::Layout);
401
402 // need those all the time to iterate over matrixes
403 _num_rows = _matrix_mass.local().num_rows();
404
405 // we need one to clone our help vectors from
406 _vec_mass_lump = _matrix_mass.create_vector_l();
407 //now we also need way more help vectors, 5 for explicit MCL, 8 for FCT and 7 for implicit MCL
408 for(int i(0); i < 8; ++i) // everyone need at least 4 vecs
409 _vec_hlp.push_back(_vec_mass_lump.clone(LAFEM::CloneMode::Layout));
410 // also initialize vec_bc and set it to zero in case its not set by user
411 _vec_bc = _vec_mass_lump.clone(LAFEM::CloneMode::Layout);
412 _vec_bc.format();
413
414 // the stuff for synchronising
415 _row_gate = _matrix_mass.get_row_gate();
416 _col_gate = _matrix_mass.get_col_gate();
417
418 _synch_matrix_ticket = std::make_shared<Global::SynchMatrix<LocalMatrixType, typename Intern::MatrixTypeHelper<MatrixType_>::RowMirror>>(
419 *_row_gate->_comm, _row_gate->_ranks, _row_gate->_mirrors, _col_gate->_mirrors);
420
421 // matrix for 1 to 0 conversion of matrices
422 _matrix_freq = _matrix_mass.local().clone(LAFEM::CloneMode::Layout);
423 _matrix_freq.format(DataType(1.0));
424
425 // sync it to know which matrix entry lives on how many patches
426 _synch_matrix_ticket->init(_matrix_freq);
427 _synch_matrix_ticket->exec(_matrix_freq);
428
429 // solver for implicit time stepping
430 if(_theta > 0)
431 {
432 _matrix_schwarz = _matrix_left.convert_to_1();
433
434 _setup_lin_solver();
435
436 _solver->set_plot_mode(Solver::PlotMode::none);
437 _solver->set_tol_abs(1e-6);
438 _solver->init_symbolic();
439 }
440
441 watch_init.stop();
442
443 _init_symbolic_called = true;
444 }

517 void init_time_stepping(DataType stepsize)
518 {
519 XASSERTM(_init_stepsize_called, "Need to call init_stepsize() first");
520 XASSERTM(stepsize > 0, "stepsize has to be greater than zero! If using implicit euler as time stepping scheme (theta = 1), you need to specify a stepsize yourself!");
521
522 watch_init.start();
523
524 _stepsize = stepsize;
525
526 // assemble matrix_right fpr calculation of right hand side for low order solution
527 // we start with R = K - D + L + N
528 _matrix_right.local().copy(_matrix_conv.local());
529 _matrix_right.local().axpy(_matrix_art_diff.local(), DataType(-1.0));
530 if(_matrix_stiff_set)
531 {
532 // if(_bc_config == 2)
533 // {
534 // _filter.local().filter_mat(_matrix_stiff.local());
535
536 // auto& vec_dof_fltr = _filter.local().get_filter_vector();
537 // auto* idx_fltr = vec_dof_fltr.indices();
538
539 // DataType* val_stiff = _matrix_stiff.local().val();
540 // for(Index i = 0; i < vec_dof_fltr.used_elements(); ++i)
541 // {
542 // const IndexType row = idx_fltr[i];
543 // for(IndexType j(_row_ptr[row]); j < _row_ptr[row + 1]; ++j)
544 // val_stiff[j] = DataType(0);
545 // }
546 // }
547
548 _matrix_right.local().axpy(_matrix_stiff.local());
549 }
550 if(_matrix_react_set)
551 {
552 // _filter.local().filter_mat(_matrix_react.local());
553 _matrix_right.local().axpy(_matrix_react.local());
554 }
555
556 // then we scale according to our time scheme and limiter
557 DataType scale_right = (_theta - DataType(1.0)) * (_limiter == 1 ? _stepsize : 1);
558 if(_theta > 0)
559 _matrix_left.local().scale(_matrix_right.local(), _stepsize * _theta);
560 _matrix_right.local().scale(_matrix_right.local(), scale_right);
561
562 if(_limiter == 1)
563 {
564 // we need to add the lmped mass matrix onto the other matrices,
565 // therefore we need the vector to be type 0
566 auto vec_mass_lump_type_0 = _matrix_mass.lump_rows(false);
567 const Memory::TypedView<DataType> lump_vec_view(vec_mass_lump_type_0.local().elements_view_r());
568 const Memory::TypedView<IndexType> row_ptr_view(_matrix_right.local().row_ptr_view_r());
569 const Memory::TypedView<IndexType> col_idx_view(_matrix_right.local().col_idx_view_r());
570 Memory::TypedView<DataType> val_right = _matrix_right.local().val_view_rw();
571 Memory::TypedView<DataType> val_left = _matrix_left.local().val_view_rw();
572 for(IndexType i = 0; i < _num_rows; ++i)
573 {
574 for(IndexType j = row_ptr_view[i]; j < row_ptr_view[i+1]; ++j)
575 {
576 if(col_idx_view[j] == i)
577 {
578 val_right[j] += lump_vec_view(i);
579 if(_theta > 0)
580 val_left[j] += lump_vec_view(i);
581 }
582 }
583 }
584 }
585
586 // _filter.local().filter_mat(_matrix_right.local());
587
588 if(_theta > 0)
589 {
590 _matrix_schwarz = _matrix_left.convert_to_1();
591 _solver->init_numeric();
592 }
593 else
594 {
595 // if theta = 0, we solve with the lumped mass matrix (which is diagonal)
596 _vec_mass_lump_inverse = _vec_mass_lump.clone(LAFEM::CloneMode::Layout);
597 _vec_mass_lump_inverse.component_invert(_vec_mass_lump);
598 }
599
600 _init_time_stepping_called = true;
601 watch_init.stop();
602 }

726 void limit_fct(VectorType& vec_sol, const VectorType& vec_rhs)
727 {
728 if(_bc_config == 2)
729 _filter.filter_sol(vec_sol);
730 // solve for low order solution
731 VectorType& vec_rhs_low = _vec_hlp.at(2);
732 // if(_bc_config == 2)
733 // _filter.filter_rhs(vec_rhs_low);
734 _matrix_right.apply(vec_rhs_low, vec_sol);
735 if(_bc_config == 1)
736 vec_rhs_low.axpy(_vec_bc, _stepsize);
737 vec_rhs_low.axpy(vec_rhs, _stepsize);
738
739 VectorType& vec_sol_low = _vec_hlp.at(3);
740 vec_sol_low.format();
741 if(_theta == 0)
742 vec_sol_low.component_product(vec_rhs_low, _vec_mass_lump_inverse);
743 else
744 {
745 if(_bc_config == 2)
746 {
747 _filter.filter_rhs(vec_rhs_low);
748 _filter.filter_sol(vec_sol_low);
749 }
750 Solver::Status solving = Solver::solve(*_solver, vec_sol_low, vec_rhs_low, _matrix_left, _filter);
751
752 if (solving != Solver::Status::success)
753 {
754 std::cerr << std::endl
755 << "ERROR: solving was not successful" << std::endl;
756 Runtime::abort();
757 }
758 }
759
760 // local bounds for each node
761 VectorType& vec_min = _vec_hlp.at(4);
762 VectorType& vec_max = _vec_hlp.at(5);
763 _calc_local_bounds_fct(vec_min.local(), vec_max.local(), vec_sol_low.local());
764 // which need a special synchronisation, where we don't add the values up, but look for the min/max of the bordering patches
765 _synch_vector_max(vec_min.local(), *_row_gate->_comm, _row_gate->_ranks, _col_gate->_mirrors);
766 _synch_vector_max(vec_max.local(), *_row_gate->_comm, _row_gate->_ranks, _col_gate->_mirrors);
767
768 //for the sum of the negative/positive fluxes that go into each node
769 VectorType& vec_neg_flux = _vec_hlp.at(6);
770 VectorType& vec_pos_flux = _vec_hlp.at(7);
771 // calculates above mentioned variables, matrix_hlp has the unlimited fluxes
772 _calc_antidiffusive_flux_fct(_matrix_hlp.local(), vec_neg_flux.local(), vec_pos_flux.local(), vec_sol.local(), vec_sol_low.local());
773 // the sum of the fluxes has to be synchronised
774 vec_neg_flux.sync_0();
775 vec_pos_flux.sync_0();
776
777 // limits the antidiffusive fluxes and return vector, that has to be added to low order solution
778 VectorType& vec_lim_flux = _vec_hlp.at(2);
779 _calc_lim_flux_fct(vec_lim_flux.local(), vec_min.local(), vec_max.local(), vec_neg_flux.local(), vec_pos_flux.local(), _matrix_hlp.local(), _vec_mass_lump.local());
780 // the limited flux has to be synced as well
781 vec_lim_flux.sync_0();
782
783 //update solution with limited flux
784 vec_sol.copy(vec_lim_flux);
785 vec_sol.axpy(vec_sol_low);
786 }

793 void limit_mcl_explicit(VectorType& vec_sol, const VectorType& vec_rhs)
794 {
795 VectorType& vec_lim_flux = _vec_hlp.at(4);
796 _calc_lim_flux_mcl(vec_lim_flux, vec_sol);
797
798 VectorType& vec_hlp = _vec_hlp.at(1);
799 // add fluxes to rhs of ode form
800 vec_hlp.copy(_vec_hlp.at(0));
801 vec_hlp.axpy(vec_lim_flux);// in vec_hlp[0] ist rhs_low = R*u gespeichert
802 // also add rhs
803 vec_hlp.axpy(vec_rhs);
804 // solve ode by multipling with inverse lumped mass matrix
805 vec_hlp.component_product(vec_hlp, _vec_mass_lump_inverse);
806 // scaling with time step
807 vec_hlp.scale(vec_hlp, _stepsize);
808 // and finally add to solution of the old time step (i.e. the explicit euler step)
809 vec_sol.axpy(vec_hlp);
810 }

951 void _calc_antidiffusive_flux_fct(LocalMatrixType& matrix_flux, LocalVectorType& vec_neg_flux,
952 LocalVectorType& vec_pos_flux, LocalVectorType& vec_sol_corr, LocalVectorType& vec_sol_low)
953 {
954 // calc derivatve
955 LocalVectorType& vec_deriv = _vec_hlp.at(0).local();
956 vec_deriv.copy(vec_sol_low);
957 vec_deriv.axpy(vec_sol_corr, -1.0);
958 vec_deriv.scale(vec_deriv, 1.0/_stepsize);
959
960 matrix_flux.format();
961 vec_neg_flux.format();
962 vec_pos_flux.format();
963
964
965 const Memory::TypedView<IndexType> row_ptr_view = _matrix_mass.local().row_ptr_view_r();
966 const Memory::TypedView<IndexType> col_idx_view = _matrix_mass.local().col_idx_view_r();
967 const Memory::TypedView<DataType> val_mass = _matrix_mass.local().val_view_r();
968 const Memory::TypedView<DataType> val_art_diff = _matrix_art_diff.local().val_view_r();
969 const Memory::TypedView<DataType> val_deriv = vec_deriv.elements_view_r();
970 const Memory::TypedView<DataType> val_sol_corr = vec_sol_corr.elements_view_r();
971 const Memory::TypedView<DataType> val_sol_low = vec_sol_low.elements_view_r();
972 Memory::TypedView<DataType> val_neg_flux = vec_neg_flux.elements_view_rw();
973 Memory::TypedView<DataType> val_pos_flux = vec_pos_flux.elements_view_rw();
974 Memory::TypedView<DataType> val_flux = matrix_flux.val_view_r();
975
976 for(IndexType i = 0; i < _num_rows; ++i)
977 {
978 const IndexType start = row_ptr_view[i];
979 const IndexType end = row_ptr_view[i+1];
980
981 for(IndexType m = start; m < end; ++m)
982 {
983 const IndexType j = col_idx_view[m];
984
985 // flux is calculated as in Christophs matlab code
986 const DataType flux = _stepsize * val_mass[m] * (val_deriv(j) - val_deriv(i))
987 - _stepsize * _theta * val_art_diff[m] * (val_sol_low(i) - val_sol_low(j))
988 - _stepsize * (1.0 - _theta) * val_art_diff[m] * (val_sol_corr(i) - val_sol_corr(j));
989
990 // prelimiting step
991 if(flux * (val_sol_low(i) - val_sol_low(j)) > 0)
992 val_flux[m] = 0;
993 else
994 val_flux[m] = flux;
995
996 // sum of negative (sum_i!=j max(0, -f_ij)) and positive (sum_i!=j max(0, f_ij)) flux into one node
997 if(j != i)
998 {
999 val_neg_flux[i] += Math::max(0.0, -val_flux[m]);
1000 val_pos_flux[i] += Math::max(0.0, val_flux[m]);
1001 }
1002 }
1003 }

1006 // returns the limited fluxes of the fct-scheme, that have to be added to the low order solution
1007 void _calc_lim_flux_fct(LocalVectorType& vec_lim_flux, LocalVectorType& vec_min, LocalVectorType& vec_max,
1008 LocalVectorType& vec_neg_flux, LocalVectorType& vec_pos_flux, LocalMatrixType& matrix_flux, LocalVectorType& vec_mass_lump)
1009 {
1010 LocalVectorType& vec_bnd_neg = _vec_hlp.at(0).local();
1011 vec_bnd_neg.format();
1012 LocalVectorType& vec_bnd_pos = _vec_hlp.at(1).local();
1013 vec_bnd_pos.format();
1014
1015 const Memory::TypedView<DataType> val_mass_lump = vec_mass_lump.elements_view_r();
1016 const Memory::TypedView<DataType> val_min = vec_min.elements_view_r();
1017 const Memory::TypedView<DataType> val_max = vec_max.elements_view_r();
1018 const Memory::TypedView<DataType> val_pos_flux = vec_pos_flux.elements_view_r();
1019 const Memory::TypedView<DataType> val_neg_flux = vec_neg_flux.elements_view_r();
1020
1021 Memory::TypedView<DataType> vbn = vec_bnd_neg.elements_view_rw();
1022 Memory::TypedView<DataType> vbp = vec_bnd_pos.elements_view_rw();
1023
1024 for(IndexType row = 0; row < _num_rows; ++row)
1025 {
1026 // distance from low order solution to local min/max, looks like this because of definition of local min/max
1027 DataType Q_p = val_mass_lump(row) * val_max(row);
1028 DataType Q_m = val_mass_lump(row) * val_min(row);
1029
1030 // bounds for correction factors
1031 vbp[row] = Math::min(1.0, Q_p/val_pos_flux(row));
1032 vbn[row] = Math::min(1.0, Q_m/val_neg_flux(row));
1033 }
1034
1035 if(_bc_config == 2)
1036 {
1037 // look for every boundary node and set the pos and neg bound to 1
1038 // (because then alpha_ij is also one)
1039 auto& vec_dof_fltr = _filter.local().get_filter_vector();
1040 const Memory::TypedView<IndexType> idx_fltr = vec_dof_fltr.indices_view_r();
1041 for(Index row = 0; row < vec_dof_fltr.num_nzes(); ++row)
1042 vbn[idx_fltr[row]] = DataType(1);
1043 }
1044
1045 // calc correction factor and multiply together with unlim fluxes
1046 const Memory::TypedView<DataType> val_flux = matrix_flux.val_view_r();
1047
1048 vec_lim_flux.format();
1049 Memory::TypedView<DataType> val_lim_flux = vec_lim_flux.elements_view_rw();
1050
1051 const Memory::TypedView<IndexType> row_ptr_view = _matrix_mass.local().row_ptr_view_r();
1052 const Memory::TypedView<IndexType> col_idx_view = _matrix_mass.local().col_idx_view_r();
1053 for(IndexType i = 0; i < _num_rows; ++i)
1054 {
1055 const IndexType start = row_ptr_view[i];
1056 const IndexType end = row_ptr_view[i+1];
1057
1058 // add up to get flux into one node
1059 DataType sum = 0;
1060 for(IndexType m = start; m < end; ++m)
1061 {
1062 const IndexType j = col_idx_view[m];
1063 if(i == j) continue;
1064
1065 if(val_flux[m] > 0)
1066 sum += Math::min(vbp(i), vbn(j)) * val_flux[m];
1067 else if(val_flux[m] < 0)
1068 sum += Math::min(vbn(i), vbp(j)) * val_flux[m];
1069 }
1070 val_lim_flux[i] = -sum/val_mass_lump(i);
1071 }
1072 }
