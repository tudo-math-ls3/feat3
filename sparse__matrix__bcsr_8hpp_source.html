134 template<bool transposed_, typename VecR_, typename VecX_>
135 void _apply(VecR_& r, const VecX_& x) const
136 {
137 if constexpr(transposed_)
138 {
139 XASSERTM(r.size_raw() == this->num_cols_raw(), "Vector size of r does not match!");
140 XASSERTM(x.size_raw() == this->num_rows_raw(), "Vector size of x does not match!");
141 }
142 else
143 {
144 XASSERTM(r.size_raw() == this->num_rows_raw(), "Vector size of r does not match!");
145 XASSERTM(x.size_raw() == this->num_cols_raw(), "Vector size of x does not match!");
146 }
147
148 if (this->num_nzes() == Index(0))
149 {
150 r.format();
151 return;
152 }
153
154 XASSERTM(r.elements_arbiter() != x.elements_arbiter(), "Vector x and r must not share the same memory!");
155
156 TimeStamp ts_start;
157 FEAT_KERNEL_MARKER_START("BCSR-apply");
158
159 Arch::MatVecMultBCSRDense::template exec<DT_, IT_, block_height_, block_width_, transposed_>(
160 r.elements_arbiter(),DT_(1), x.elements_arbiter(), Memory::Arbiter(),
161 this->val_arbiter(), this->col_idx_arbiter(), this->row_ptr_arbiter(),
162 this->num_rows(), this->num_cols(), this->num_nzes());
163
164 FEAT_KERNEL_MARKER_STOP("BCSR-apply");
165 TimeStamp ts_stop;
166
167 Statistics::add_flops(this->num_nzes_raw() * 2);
168 Statistics::add_time_reduction(ts_stop.elapsed(ts_start));
169 }

171 template<bool transposed_, typename VecR_, typename VecX_, typename VecY_>
172 void _apply(VecR_& r, const VecX_& x, const VecY_& y, const DT_ alpha) const
173 {
174 if constexpr(transposed_)
175 {
176 XASSERTM(r.size_raw() == this->num_cols_raw(), "Vector size of r does not match!");
177 XASSERTM(x.size_raw() == this->num_rows_raw(), "Vector size of x does not match!");
178 XASSERTM(y.size_raw() == this->num_cols_raw(), "Vector size of y does not match!");
179 }
180 else
181 {
182 XASSERTM(r.size_raw() == this->num_rows_raw(), "Vector size of r does not match!");
183 XASSERTM(x.size_raw() == this->num_cols_raw(), "Vector size of x does not match!");
184 XASSERTM(y.size_raw() == this->num_rows_raw(), "Vector size of y does not match!");
185 }
186
187 // we check alpha strictly for equality to 0, because testing if |alpha| < eps may
188 // lead to false triggers if the matrix/vector contents are also < eps
189 if((this->num_nzes() == Index(0)) || (alpha == DT_(0)))
190 {
191 r.copy(y);
192 return;
193 }
194
195 XASSERTM(r.elements_arbiter() != x.elements_arbiter(), "Vector x and r must not share the same memory!");
196
197 TimeStamp ts_start;
198 FEAT_KERNEL_MARKER_START("BCSR-apply");
199
200 Arch::MatVecMultBCSRDense::template exec<DT_, IT_, block_height_, block_width_, transposed_>(
201 r.elements_arbiter(),alpha, x.elements_arbiter(), y.elements_arbiter(),
202 this->val_arbiter(), this->col_idx_arbiter(), this->row_ptr_arbiter(),
203 this->num_rows(), this->num_cols(), this->num_nzes());
204
205 FEAT_KERNEL_MARKER_STOP("BCSR-apply");
206 TimeStamp ts_stop;
207
208 Statistics::add_flops(this->num_nzes_raw() * 2);
209 Statistics::add_time_reduction(ts_stop.elapsed(ts_start));
210 }

459 template <typename DT2_, typename IT2_>
460 void convert(const SparseMatrixCSR<DT2_, IT2_>& other)
461 {
462 this->clear();
463 // setup internal sizes
464 Index n_rows = Index(other.num_rows()) / Index(block_height);
465 Index n_cols = Index(other.num_cols()) / Index(block_width);
466
467 // temprorary arrays for row and column ptr
468 std::vector<IndexType> tmp_row_ptr(n_rows+1u);
469 std::vector<IndexType> tmp_col_ptr;
470
471 // ptrs of csr matrix
472 const Memory::TypedView<IT2_> crs_row = other.row_ptr_view_r();
473 const Memory::TypedView<IT2_> crs_col = other.col_idx_view_r();
474 const Memory::TypedView<DT2_> crs_val = other.val_view_r();
475
476 // reserve to sensical size, we will have at most used_element blocks
477 tmp_col_ptr.reserve(other.num_nzes());
478
479 // for each block row, gather which blocks should be actually allocated
480 std::set<IndexType> tmp_uni_col;
481
482 tmp_row_ptr[0] = IndexType(0);
483 for(Index b_row = 0; b_row < n_rows; ++b_row)
484 {
485 for(Index l_row = b_row*Index(block_height); l_row < (b_row+1)*Index(block_height); ++l_row)
486 {
487 for(IT2_ idx = crs_row[l_row]; idx < crs_row[l_row+1]; ++idx)
488 {
489 tmp_uni_col.insert(IndexType(crs_col[idx])/IndexType(block_width));
490 }
491 }
492 tmp_row_ptr[b_row+1] = IndexType(tmp_uni_col.size()) + tmp_row_ptr[b_row];
493 std::for_each(tmp_uni_col.begin(), tmp_uni_col.end(), [&](const auto& ele){tmp_col_ptr.push_back(ele);});
494 // std::printf("Brow %i: ", int(b_row));
495 // std::for_each(tmp_uni_col.begin(), tmp_uni_col.end(), [&](const auto& ele){std::cout << ele << ", ";});
496 // std::cout << "\n";
497 tmp_uni_col.clear();
498 }
499
500 Index elements_size = tmp_col_ptr.size();
501 // std::printf("Elements size %i\n", int(elements_size));
502 // std::printf("Tmp col ptr:\n");
503 // std::for_each(tmp_col_ptr.begin(), tmp_col_ptr.end(), [&](const auto& ele){std::cout << ele << ", ";});
504 // std::cout << "\n";
505
506 // allocate actual matrix data
507 this->_scalar_index.push_back(n_rows);
508 this->_scalar_index.push_back(n_cols);
509 this->_scalar_index.push_back(elements_size);
510
511 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * (_rows() + 1)));
512 this->_indices_size.push_back(_rows() + 1);
513
514 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * _nzes()));
515 this->_indices_size.push_back(_nzes());
516
517 this->_elements.push_back(Memory::Arbiter(sizeof(DT_) * num_nzes_raw()));
518 this->_elements_size.push_back(num_nzes_raw());
519
520 // format data
521 this->format();
522
523 Memory::TypedView<IT_> row_ptr(this->_indices.at(0).view(Memory::Location::main, Memory::Access::write));
524 Memory::TypedView<IT_> col_idx(this->_indices.at(1).view(Memory::Location::main, Memory::Access::write));
525 Memory::TypedView<DT_> val(this->_elements.at(0).view(Memory::Location::main, Memory::Access::write));
526
527 // now, fill up array data
528 std::copy(tmp_row_ptr.begin(), tmp_row_ptr.end(), row_ptr.get_w());
529 std::copy(tmp_col_ptr.begin(), tmp_col_ptr.end(), col_idx.get_w());
530 // std::printf("Row ptr:\n");
531 // std::for_each(row_ptr, row_ptr+n_rows+1, [&](const auto& ele){std::cout << ele << ", ";});
532 // std::cout << "\n";
533 // std::printf("Col ptr:\n");
534 // std::for_each(col, col+elements_size, [&](const auto& ele){std::cout << ele << ", ";});
535 // std::cout << "\n";
536
537 // clear temprorary arrays
538 tmp_row_ptr.clear();
539 tmp_col_ptr.clear();
540
541 // run through crs data and simply write to the correct location, this can be done
542 // in parallel
543 FEAT_PRAGMA_OMP(parallel for)
544 for(Index b_row = Index(0); b_row < n_rows; ++b_row)
545 {
546 for(Index l_row = Index(0); l_row < Index(block_height); ++l_row)
547 {
548 IndexType b_idx = row_ptr[b_row];
549 for(IT2_ idx = crs_row[b_row*Index(block_height) + l_row]; idx < crs_row[b_row*Index(block_height) + l_row + Index(1)]; ++idx)
550 {
551 //advance block row index until we find our current row
552 while(col_idx[b_idx] != IndexType(crs_col[idx]/block_width))
553 {
554 ++b_idx;
555 }
556 ASSERTM(col_idx[b_idx] == IndexType(crs_col[idx]/block_width), "Blocked Column Array does not fit csr columns");
557 // write data
558 val[b_idx*block_height*block_width + block_width*l_row + (crs_col[idx]%block_width)] = DT_(crs_val[idx]);
559 }
560 }
561 }
562 //done
563 }

1217 bool same_layout(const SparseMatrixBCSR& x) const
1218 {
1219 if (this->num_rows() != x.num_rows())
1220 return false;
1221 if (this->num_cols() != x.num_cols())
1222 return false;
1223 if (this->num_nzes() != x.num_nzes())
1224 return false;
1225
1226 if(this->num_nzes() == Index(0))
1227 return true;
1228
1229 // check if the arbiters for row_ptr and col_idx are the same
1230 if((this->row_ptr_arbiter() == x.row_ptr_arbiter()) && (this->col_idx_arbiter() == x.col_idx_arbiter()))
1231 return true;
1232
1233 const Memory::TypedView<IT_> row_ptr_a = this->row_ptr_view_r();
1234 const Memory::TypedView<IT_> col_idx_a = this->col_idx_view_r();
1235 const Memory::TypedView<IT_> row_ptr_b = x.row_ptr_view_r();
1236 const Memory::TypedView<IT_> col_idx_b = x.col_idx_view_r();
1237
1238 const Index n = this->num_rows();
1239 for (Index i(0) ; i <= n; ++i)
1240 {
1241 if (row_ptr_a[i] != row_ptr_b[i])
1242 {
1243 return false;
1244 }
1245 }
1246
1247 const Index nze = this->num_nzes();
1248 for (Index i(0) ; i < nze ; ++i)
1249 {
1250 if (col_idx_a[i] != col_idx_b[i])
1251 {
1252 return false;
1253 }
1254 }
1255
1256 return true;

1278 void transpose(const SparseMatrixBCSR<DT_, IT_, block_width_, block_height_> & x)
1279 {
1280 using XType = SparseMatrixBCSR<DT_, IT_, block_width_, block_height_>;
1281 if (x.num_nzes() == 0)
1282 {
1283 this->move(SparseMatrixBCSR(x.num_rows(), x.num_cols()));
1284 return;
1285 }
1286
1287 const Index txrows(x.num_rows());
1288 const Index txcolumns(x.num_cols());
1289 const Index txnonzeros(x.num_nzes());
1290
1291 const Memory::TypedView<IT_> ptxcol_idx(x.col_idx_view_r());
1292 const Memory::TypedView<IT_> ptxrow_ptr(x.row_ptr_view_r());
1293 const Memory::TypedView<typename XType::ValueType> ptxval(x.val_view_r());
1294
1295 DenseVector<IT_, IT_> tcol_idx(txnonzeros);
1296 DenseVector<DT_, IT_> tval(txnonzeros * block_height_ * block_width_);
1297 DenseVector<IT_, IT_> trow_ptr(txcolumns + 1, IT_(0));
1298
1299 Memory::TypedView<IT_> ptrow_ptr(trow_ptr.elements_view_rw());
1300 Memory::TypedView<IT_> ptcol_idx(tcol_idx.elements_view_w());
1301 Memory::TypedView<ValueType> ptval(std::forward<Memory::View&&>(tval.elements_view_w()));
1302
1303 for (Index i(0); i < txnonzeros; ++i)
1304 {
1305 ++ptrow_ptr[ptxcol_idx[i] + 1];
1306 }
1307
1308 for (Index i(1); i < txcolumns - 1; ++i)
1309 {
1310 ptrow_ptr[i + 1] += ptrow_ptr[i];
1311 }
1312
1313 for (Index i(0); i < txrows; ++i)
1314 {
1315 for (IT_ k(ptxrow_ptr[i]); k < ptxrow_ptr[i+1]; ++k)
1316 {
1317 const IT_ l(ptxcol_idx[k]);
1318 const IT_ j(ptrow_ptr[l]);
1319 ptval[j].set_transpose(ptxval[k]);
1320 ptcol_idx[j] = IT_(i);
1321 ++ptrow_ptr[l];
1322 }
1323 }
1324
1325 for (Index i(txcolumns); i > 0; --i)
1326 {
1327 ptrow_ptr[i] = ptrow_ptr[i - 1];
1328 }
1329 ptrow_ptr[0] = 0;
1330
1331 this->move(SparseMatrixBCSR<DT_, IT_, block_height_, block_width_> (txcolumns, txrows, trow_ptr, tcol_idx, tval));
1332 }

1718 void add_double_mat_product(
1719 const LAFEM::SparseMatrixBCSR<DT_, IT_, block_height_, block_width_>& d,
1720 const LAFEM::SparseMatrixBCSR<DT_, IT_, block_height_, block_width_>& a,
1721 const LAFEM::SparseMatrixBCSR<DT_, IT_, block_height_, block_width_>& b,
1722 const DT_ alpha = DT_(1),
1723 const bool allow_incomplete = false)
1724 {
1725 // validate matrix dimensions
1726 XASSERT(block_height_ == block_width_);
1727 XASSERT(this->num_rows() == d.num_rows());
1728 XASSERT(d.num_cols() == a.num_rows());
1729 XASSERT(a.num_cols() == b.num_rows());
1730 XASSERT(b.num_cols() == this->num_cols());
1731
1732 // fetch matrix arrays:
1733 Memory::TypedView<ValueType> data_x = this->val_view_rw();
1734 const Memory::TypedView<ValueType> data_d = d.val_view_r();
1735 const Memory::TypedView<ValueType> data_a = a.val_view_r();
1736 const Memory::TypedView<ValueType> data_b = b.val_view_r();
1737 const Memory::TypedView<IT_> row_ptr_x = this->row_ptr_view_r();
1738 const Memory::TypedView<IT_> col_idx_x = this->col_idx_view_r();
1739 const Memory::TypedView<IT_> row_ptr_d = d.row_ptr_view_r();
1740 const Memory::TypedView<IT_> col_idx_d = d.col_idx_view_r();
1741 const Memory::TypedView<IT_> row_ptr_a = a.row_ptr_view_r();
1742 const Memory::TypedView<IT_> col_idx_a = a.col_idx_view_r();
1743 const Memory::TypedView<IT_> row_ptr_b = b.row_ptr_view_r();
1744 const Memory::TypedView<IT_> col_idx_b = b.col_idx_view_r();
1745
1746 // loop over all rows of D and X, resp.
1747 for(IT_ i(0); i < IT_(this->num_rows()); ++i)
1748 {
1749 // loop over all non-zeros D_ik in row i of D
1750 for(IT_ ik(row_ptr_d[i]); ik < row_ptr_d[i+1]; ++ik)
1751 {
1752 // get column index k
1753 const IT_ k = col_idx_d[ik];
1754
1755 // loop over all non-zeros A_kl in row k of A
1756 for(IT_ kl(row_ptr_a[k]); kl < row_ptr_a[k+1]; ++kl)
1757 {
1758 // get column index l
1759 const IT_ l = col_idx_a[kl];
1760

2009 Index new_start(0);

2011 for (Index row(0) ; row < this->num_rows() ; ++row)
2012 {
2013 Index row_size(row_ptr[perm_pos[row] + 1] - row_ptr[perm_pos[row]]);

2016 for (Index i(new_start), j(row_ptr[perm_pos[row]]) ; i < new_start + row_size ; ++i, ++j)
2017 {
2018 temp_col_idx[i] = col_idx[j];
2019 temp_val[i] = val[j];
2020 }
2021
2022 new_start += row_size;
2023 temp_row_ptr[row+1] = (IT_)new_start;
2024 }
2025

2027 Adjacency::Permutation perm_col_inv = perm_col.inverse();
2028 perm_pos = perm_col_inv.get_perm_pos();

2033 for (Index i(0) ; i < num_nzes() ; ++i)
2034 {
2035 col_idx[i] = (IT_)perm_pos[temp_col_idx[i]];
2036 }
2037
2038 delete[] temp_row_ptr;
2039 delete[] temp_col_idx;
2040 delete[] temp_val;

2043 IT_ swap_key;
2044 ValueType swap_val;
2045 for (Index row(0) ; row < num_rows() ; ++row)
2046 {
2047 Index offset(row_ptr[row]);
2048 Index row_size(row_ptr[row+1] - row_ptr[row]);
2049 for (Index i(1), j ; i < row_size ; ++i)

2052 swap_val = val[i + offset];

2056 col_idx[j + offset] = col_idx[j - 1 + offset];

2059 }
