186 class NvtxRangeStack
187 {
188 private:
190 std::vector<std::uint64_t> _ids;
191
192 public:
193 NvtxRangeStack() = default;
194
196 NvtxRangeStack(const NvtxRangeStack&) = delete;
197 NvtxRangeStack& operator=(const NvtxRangeStack&) = delete;
198
200 ~NvtxRangeStack()
201 {
202 while(!_ids.empty())
203 pop();
204 }
205
207 void push([[maybe_unused]] const char* name)
208 {
209#ifdef FEAT_HAVE_CUDA
210 _ids.push_back(cuda_start_range(name));
211#endif
212 }
213
215 void pop()
216 {
217#ifdef FEAT_HAVE_CUDA
218 if(!_ids.empty())
219 {
220 cuda_stop_range(_ids.back());
221 _ids.pop_back();
222 }
223#endif
224 }
225
227 std::size_t size() const
228 {
229 return _ids.size();
230 }
231 };
232
234 inline NvtxRangeStack& nvtx_thread_stack()
235 {
236 thread_local NvtxRangeStack stack;
237 return stack;
238 }

251 class ScopedNvtxRange
252 {
253 private:
255 NvtxRangeStack& _stack;
256
257 public:
259 explicit ScopedNvtxRange(const char* name) :
260 _stack(nvtx_thread_stack())
261 {
262 _stack.push(name);
263 }
264
266 ScopedNvtxRange(NvtxRangeStack& stack, const char* name) :
267 _stack(stack)
268 {
269 _stack.push(name);
270 }
271
273 ScopedNvtxRange(const ScopedNvtxRange&) = delete;
274 ScopedNvtxRange& operator=(const ScopedNvtxRange&) = delete;
275
277 ~ScopedNvtxRange()
278 {
279 _stack.pop();
280 }
281 };
