297 template<bool gradient_, typename PointRange_, typename VectorType_, typename Space_>
298 static std::vector<typename Intern::DiscreteEvaluatorTypeHelper<PointRange_, VectorType_, Space_, gradient_>::EvalType>
299 _remote_eval(const Dist::Comm& comm, const PointRange_& points, const VectorType_& vector, const Space_& space)
300 {
301 // We need to know the types of the values produced by evaluating the finite-element function
302 // and the appropriate *DiscreteEvalData wrapper. The DiscreteEvaluatorTypeHelper
303 // provides the correct types based on the VectorType_ and gradient_ params.
304 using Helper = Intern::DiscreteEvaluatorTypeHelper<PointRange_, VectorType_, Space_, gradient_>;
305
306 using DataType = typename Helper::DataType;
307 using BoundingBoxType = typename Helper::BoundingBoxType;
308 using PointType = typename Helper::PointType;
309 using EvalType = typename Helper::EvalType;
310 using EvalValueType = typename Helper::EvalValueType;
311
312 const auto total_ranks = std::size_t(comm.size());
313 const auto rank = std::size_t(comm.rank());
314 const auto num_points = std::size_t(std::distance(points.begin(), points.end()));
315
316 // Exchange bounding boxes with other ranks
317 std::vector<BoundingBoxType> bbs(static_cast<std::size_t>(total_ranks));
318 bbs.at(rank) = get_local_bb<BoundingBoxType>(space.get_mesh());
319 comm.allgather(bbs.data(), 1, bbs.data(), 1);
320
321 // Build structure for exchanging points
322 std::vector<int> ranks;
323 ranks.reserve(num_points);
324 std::vector<Index> indices;
325 ranks.reserve(num_points);
326
327 Index point_idx(0);
328 for(auto it = points.begin(); it != points.end(); ++it)
329 {
330 const PointType& p = *it;
331 for(std::size_t i(0); i < total_ranks; i++)
332 {
333 const BoundingBoxType& bb = bbs[i];
334
335 if(is_in_bb(bb, p))
336 {
337 ranks.push_back(int(i));
338 indices.push_back(point_idx);
339 }
340 }
341 point_idx++;
342 }
343
344 // Cache inverse mapping
345 // The inverse mapping creates a list of bounding boxes for each cell.
346 // This way we create this list only once per rank, rather than once per point
347 Trafo::InverseMapping<typename Space_::TrafoType, DataType> inv_mapping(space.get_trafo());
348
349 // Make remote calls. Each ranks returns a vector of values for each point, in case a rank manages to unmap the point on multiple cells
350 ScalarRemoteLambda<EvalValueType(const PointType&)> rl(comm);
351 std::vector<EvalValueType> evals = rl.call(
352 ranks,
353 indices,
354 points,
355 [&vector, &space, &inv_mapping](const PointType& p)
356 {
357 auto unmapping = inv_mapping.unmap_point(p);
358 if constexpr(gradient_)
359 {
360 return Assembly::DiscreteEvaluator::eval_fe_gradient(unmapping, vector, space).mean_value();
361 }
362 else
363 {
364 return Assembly::DiscreteEvaluator::eval_fe_function(unmapping, vector, space).mean_value();
365 }
366 });
367
368 // Assemble final results from individual calls
369 std::vector<EvalType> result(std::size_t(std::distance(points.begin(), points.end())));
370
371 for(std::size_t i(0); i < evals.size(); i++)
372 {
373 // To which point did this remote lambda call belong?
374 const Index idx = indices[i];
375
376 // Add evaluation result to total results for that point
377 result[idx].values.push_back(evals[i]);
378 }
379
380 return result;
381 }

384 template<typename BoundingBoxType_, typename PointType_>
385 static bool is_in_bb(const BoundingBoxType_& bb, const PointType_& p)
386 {
387 using DataType = typename BoundingBoxType_::ValueType;
388 constexpr int dim = BoundingBoxType_::m;
389
390 constexpr DataType bb_eps(1E-2);
391 for(int i(0); i < dim; i++)
392 {
393 if(p[i] > (bb[1][i] + bb_eps) || p[i] < (bb[0][i] - bb_eps))
394 {
395 return false;
396 }
397 }
398 return true;
399 }

402 template<typename BoundingBoxType_, typename MeshType_>
403 static BoundingBoxType_ get_local_bb(const MeshType_& mesh)
404 {
405 const auto& vertex_set = mesh.get_vertex_set();
406
407 if(vertex_set.get_num_vertices() == 0)
408 {
409 return BoundingBoxType_{0};
410 }
411
412 // Determine local bounding box
413 BoundingBoxType_ local_bb;
414
415 local_bb[0] = vertex_set[0];
416 local_bb[1] = vertex_set[0];
417 for(Index vtx_idx(0); vtx_idx < vertex_set.get_num_vertices(); vtx_idx++)
418 {
419 const auto& v = vertex_set[vtx_idx];
420
421 for(int i(0); i < MeshType_::world_dim; i++)
422 {
423 Math::mini(local_bb[0][i], v[i]);
424 Math::maxi(local_bb[1][i], v[i]);
425 }
426 }
427
428 return local_bb;
429 }
