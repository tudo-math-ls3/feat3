105 void _alloc(Index rows, Index cols, Index nz_rows, Index nzes)
106 {
107 XASSERT(this->_scalar_index.empty());
108 XASSERT(this->_indices.empty());
109 XASSERT(this->_elements.empty());
110
111 this->_scalar_index.push_back(rows);
112 this->_scalar_index.push_back(cols);
113 this->_scalar_index.push_back(nz_rows);
114 this->_scalar_index.push_back(nzes);

116 if(rows <= Index(0))
117 return;
118
119 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * (nz_rows + 1)));
120 this->_indices_size.push_back(nz_rows + 1);
121
122 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * nz_rows));
123 this->_indices_size.push_back(nz_rows);
124
125 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * nzes));
126 this->_indices_size.push_back(nzes);
127
128 this->_elements.push_back(Memory::Arbiter(sizeof(DT_) * nzes));
129 this->_elements_size.push_back(nzes);
130 }

288 explicit SparseMatrixCSCR(const SparseMatrixCSR<DT_, IT_> & csr, const VectorMirror<DT_, IT_> & non_zero_rows)
289 {
290 XASSERT(non_zero_rows.num_indices() > 0);
291
292 const Memory::TypedView<IT_> row_ptr_in = csr.row_ptr_view_r();
293 const Memory::TypedView<DT_> val_in = csr.val_view_r();
294 const Memory::TypedView<IT_> col_idx_in = csr.col_idx_view_r();
295 const Memory::TypedView<IT_> mir_idx = non_zero_rows.indices_view_r();
296
297 // count number of non-zeros
298 Index nz_rows = non_zero_rows.num_indices();
299 Index nnzes(0);
300 for (Index i(0) ; i < nz_rows ; ++i)
301 nnzes += row_ptr_in[mir_idx[i]+1] - row_ptr_in[mir_idx[i]];
302
303 this->_alloc(csr.num_rows(), csr.num_cols(), nz_rows, nnzes);
304
305 Memory::TypedView<DT_> val = this->val_view_w();
306 Memory::TypedView<IT_> row_ptr = this->row_ptr_view_w();
307 Memory::TypedView<IT_> col_idx = this->col_idx_view_w();
308 Memory::TypedView<IT_> row_idx = this->row_idx_view_w();
309
310 row_ptr[0] = IT_(0);
311 for(Index i(0); i < nz_rows; ++i)
312 {
313 IT_ row = mir_idx[i];
314 IT_ j = row_ptr[i];
315 row_idx[i] = row;
316 for(IT_ k(row_ptr_in[row]); k < row_ptr_in[row+1]; ++k, ++j)
317 {
318 val[j] = val_in[k];
319 col_idx[j] = col_idx_in[k];
320 }
321 row_ptr[i+1] = j;
322 }
323
324 XASSERT(row_ptr[nz_rows] == nnzes);
325 }

334 explicit SparseMatrixCSCR(const Adjacency::Graph & graph)
335 {
336 // get number of rows, columns and indices
337 Index num_rows = graph.get_num_nodes_domain();
338 Index num_cols = graph.get_num_nodes_image();
339 Index num_nzes = graph.get_num_indices();
340
341 if (num_nzes <= Index(0))
342 {
343 this->_alloc(num_rows, num_cols, Index(0), Index(0));
344 return;
345 }
346
347 // get graph arrays
348 const Index* dom_ptr = graph.get_domain_ptr();
349 const Index* img_idx = graph.get_image_idx();
350
351 // count number of non-empty rows
352 Index num_nzrs = Index(0);
353 for(Index i(0); i < num_rows; ++i)
354 {
355 num_nzrs += Index(dom_ptr[i] < dom_ptr[i+1] ? 1 : 0);
356 }
357
358 // allocate output matrix
359 this->_alloc(num_rows, num_cols, num_nzrs, num_nzes);
360
361 // get matrix arrays
362 Memory::TypedView<IT_> trow_ptr = this->row_ptr_view_w();
363 Memory::TypedView<IT_> trow_idx = this->row_idx_view_w();
364 Memory::TypedView<IT_> tcol_idx = this->col_idx_view_w();
365
366 // fill arrays
367 trow_ptr[0] = IndexType(dom_ptr[0]);
368 for(Index i(0), j(0); i < num_rows; ++i)
369 {
370 if(dom_ptr[i] < dom_ptr[i + 1])
371 {
372 ASSERT(j < num_nzrs);
373 trow_idx[ j] = IndexType(i);
374 trow_ptr[++j] = IndexType(dom_ptr[i+1]);
375 }
376 }
377
378 for(Index k(0); k < num_nzes; ++k)
379 tcol_idx[k] = IndexType(img_idx[k]);
380 }

470 template <typename MT_>
471 void convert(const MT_ & a)
472 {
473 XASSERT((void*)this != (void*)&a);
474
475 this->clear();
476
477 const Index num_rows_a = a.num_rows_raw();
478 const Index num_cols_a = a.num_cols_raw();
479 const Index num_nzes_a = a.num_nzes_raw();
480 if(num_nzes_a <= Index(0))
481 {
482 this->_alloc(num_rows_a, num_cols_a, Index(0), Index(0));
483 return;
484 }
485
486 Index num_nz_rows_a(0);
487 for (Index i(0); i < num_rows_a; ++i)
488 {
489 num_nz_rows_a += Index(a.row_degree(i) > Index(0) ? 1 : 0);
490 }
491
492 this->_alloc(num_rows_a, num_cols_a, num_nz_rows_a, num_nzes_a);
493
494 Memory::TypedView<IT_> row_ptr = this->row_ptr_view_w();
495 Memory::TypedView<IT_> row_idx = this->row_idx_view_w();
496 Memory::TypedView<IT_> col_idx = this->col_idx_view_w();
497 Memory::TypedView<DT_> val = this->val_view_w();
498
499 row_ptr[0] = Index(0);
500 for(Index row(0), k(0); row < num_rows_a; ++row)
501 {
502 Index nz = a.row_degree(row);
503 if(nz > Index(0))
504 {
505 a.get_row_col_indices(row, &col_idx[row_ptr[k]], IT_(0));
506 a.get_row_values(row, &val[row_ptr[k]]);
507 row_idx[k] = IT_(row);
508 row_ptr[k+1] = row_ptr[k] + IT_(nz);
509 ++k;
510 }
511 }
512 }

926 void axpy(const SparseMatrixCSCR & x, const DT_ alpha = DT_(1))
927 {
928 XASSERTM(x.num_rows() == this->num_rows(), "Matrix rows do not match!");
929 XASSERTM(x.num_cols() == this->num_cols(), "Matrix columns do not match!");
930 XASSERTM(x.num_nzes() == this->num_nzes(), "Matrix nonzeros do not match!");
931
932 if(this->empty())
933 return;
934
935 TimeStamp ts_start;
936
937 Arch::AxpyDense::template exec<DT_>(this->val_arbiter(), alpha, x.val_arbiter(), this->num_nzes());
938
939 TimeStamp ts_stop;
940
941 Statistics::add_flops(this->num_nzes() * 2);
942 Statistics::add_time_axpy(ts_stop.elapsed(ts_start));
943 }

951 void scale(const SparseMatrixCSCR & x, const DT_ alpha)
952 {
953 XASSERTM(x.num_rows() == this->num_rows(), "Row count does not match!");
954 XASSERTM(x.num_cols() == this->num_cols(), "Column count does not match!");
955 XASSERTM(x.num_nzes() == this->num_nzes(), "Nonzero count does not match!");
956
957 if(this->empty())
958 return;
959

1020 bool same_layout(const SparseMatrixCSCR& x) const
1021 {
1022 if (this->num_rows() != x.num_rows())
1023 return false;
1024 if (this->num_cols() != x.num_cols())
1025 return false;
1026 if (this->num_nzes() != x.num_nzes())
1027 return false;
1028 if (this->num_nonzero_rows() != x.num_nonzero_rows())
1029 return false;
1030
1031 if(this->num_nzes() == Index(0))
1032 return true;
1033
1034 // check if the arbiters for row_ptr, row_idx and col_idx are the same
1035 if((this->row_ptr_arbiter() == x.row_ptr_arbiter()) &&
1036 (this->row_idx_arbiter() == x.row_idx_arbiter()) &&
1037 (this->col_idx_arbiter() == x.col_idx_arbiter()))
1038 return true;
1039
1040 const Memory::TypedView<IT_> row_ptr_a = this->row_ptr_view_r();
1041 const Memory::TypedView<IT_> row_idx_a = this->row_idx_view_r();
1042 const Memory::TypedView<IT_> col_idx_a = this->col_idx_view_r();
1043 const Memory::TypedView<IT_> row_ptr_b = x.row_ptr_view_r();
1044 const Memory::TypedView<IT_> row_idx_b = x.row_idx_view_r();
1045 const Memory::TypedView<IT_> col_idx_b = x.col_idx_view_r();
1046
1047 const Index nzr = this->num_nonzero_rows();
1048 for (Index i(0) ; i < nzr; ++i)
1049 {
1050 if(row_ptr_a[i] != row_ptr_b[i])
1051 return false;
1052 if(row_idx_a[i] != row_idx_b[i])
1053 return false;
1054 }
1055 if(row_ptr_a[nzr] != row_ptr_b[nzr])
1056 return false;
1057
1058 const Index nze = this->num_nzes();
1059 for (Index i(0) ; i < nze ; ++i)
1060 {
1061 if (col_idx_a[i] != col_idx_b[i])
1062 return false;
1063 }
1064
1065 return true;
1066 }

1282 // write leading zeros
