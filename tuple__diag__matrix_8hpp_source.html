206 void write_out(FileMode mode, String filename) const
207 {
208 std::ofstream file(filename.c_str(), std::ofstream::out);
209 if (! file.is_open())
210 XABORTM("Unable to open Matrix file " + filename);
211
212 String suffix, directory;
213 auto found = filename.rfind(".");
214 if (found != std::string::npos)
215 {
216 suffix = filename.substr(found);
217 filename.erase(found);
218 }
219 found = filename.rfind("/");
220 if (found != std::string::npos)
221 {
222 directory = filename.substr(0, found + 1);
223 filename.erase(0, found + 1);
224 }
225
226 file << "%%MatrixMarket tuplediagmatrix coordinate real general\n";
227 for (Index i(1); i <= num_row_blocks; ++i)
228 {
229 file << filename << "_td" << i << suffix << "\n";
230 }
231
232 file.close();
233
234 this->write_out_submatrices(mode, directory, filename, suffix);

623
625 void restore_from_checkpoint_data(std::vector<char> & data)
626 {
627 std::uint64_t isize = *(std::uint64_t*) data.data(); //get size of checkpointed _first
628 std::vector<char>::iterator start = std::begin(data) + sizeof(std::uint64_t);
629 std::vector<char>::iterator last_of_first = std::begin(data) + sizeof(std::uint64_t) + (int) isize;
630 std::vector<char> buffer_first(start, last_of_first);
631 _first.restore_from_checkpoint_data(buffer_first);
632
633 data.erase(std::begin(data), last_of_first);
634 _rest.restore_from_checkpoint_data(data);
635 }
636
638 std::uint64_t set_checkpoint_data(std::vector<char>& data, SerialConfig& config)
639 {
640 std::size_t old_size = data.size();
641 data.insert(std::end(data), sizeof(std::uint64_t), 0); //add placeholder
642 std::uint64_t ireal_size = _first.set_checkpoint_data(data, config); //add data of _first to the overall checkpoint and save its size
643 char* csize = reinterpret_cast<char*>(&ireal_size);
644 for(std::size_t i(0) ; i < sizeof(std::uint64_t) ; ++i) //overwrite the guessed datalength
645 {
646 data[old_size +i] = csize[i];
647 }
648
649 return sizeof(std::uint64_t) + ireal_size + _rest.set_checkpoint_data(data, config); //generate and add checkpoint data for the _rest
650 }
