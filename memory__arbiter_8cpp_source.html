165 void create(std::size_t bytes_, Location loc_, Init all_)
166 {
167 std::unique_lock<std::mutex> lock(lock_mutex());
168
169 // make sure that this block is empty
170 if(this->bytes > 0u)
171 throw MemoryException("Memory::Block::create: memory block is already allocated");
172
173 // save byte count and try to allocate
174 this->bytes = bytes_;
175 if(loc_ != Location::none)
176 this->_alloc(loc_, all_);

179 // finally, register this block with the manager

197 static bool detach(Block* block)
198 {
199 if(block == nullptr)
200 return false;
201
202 Lock lock(block->lock_mutex());
203
204 //XASSERTM(block != nullptr, "cannot detach from null block");
205 if(block->ref_count == 0u)
206 throw MemoryException("Memory::Block::detach: reference count is already zero");
207
208 if(0u < --block->ref_count)
209 return true;
210
211 // unregister this block with the manager
212 {
213 Lock manager_lock(manager_mutex);
214 manager_block_set.erase(block);
215 }
216
217 // all references have been detached, so let's free the block
218 block->_free_main();
219 block->_free_cuda();
220
221 // finally, free the block itself
222 delete block;
223
224 return false;
225 }

258 void copy_from(const Block& source, Location mem_loc, std::size_t this_offset, std::size_t src_offset, std::size_t size)
259 {
260 // first lock the mutex for this memory block
261 std::unique_lock<std::mutex> lock(lock_mutex());
262
263 // make sure that this block is not mapped
264 if(this->_is_mapped_any())
265 throw AccessException("Memory::Block::copy_from: cannot copy to mapped memory block");
266
267 // now lock the mutex for the source block, but only if it actually uses another mutex than ours
268 std::unique_lock<std::mutex> source_lock;
269 if(!this->uses_same_lock(source))
270 source_lock = source.lock_mutex();
271
272 // ensure that the source block is not mapped for writing
273 if(source._is_mapped_any_write())
274 throw AccessException("Memory::Block::copy_from: cannot copy from write-mapped memory block");
275
276 // if no target location is specified, then simply copy the existing source location
277 if(mem_loc == Location::none)
278 mem_loc = source.valid_locations;
279
280 // ensure the ranges are valid
281 if(size == full_size)
282 throw RangeException("Memory::Block::copy_from: memory size must be given explicitly");
283 if(this->bytes < this_offset + size)
284 throw RangeException("Memory::Block::copy_from: invalid destination memory range");
285 if(source.bytes < src_offset + size)
286 throw RangeException("Memory::Block::copy_from: invalid source memory range");
287
288 // perform the actual copy
289 if(size > std::size_t(0))
290 {
291 this->_copy_from(source, mem_loc, this_offset, src_offset, size);
292 }
293
294 // store our up-to-date locations
295 this->valid_locations = mem_loc;
296 }

353 void format(Location mem_loc, std::size_t this_offset, std::size_t size)
354 {
355 std::unique_lock<std::mutex> lock(lock_mutex());
356 if(this->_is_mapped_any())
357 throw AccessException("Memory::Block::format: cannot format mapped memory block");
358
359 // ensure the ranges are valid
360 if(this->bytes < this_offset)
361 throw RangeException("Memory::Block::format: invalid destination memory offset");
362 if(size == full_size)
363 size = this->bytes - this_offset;
364 if(this->bytes < this_offset + size)
365 throw RangeException("Memory::Block::format: invalid destination memory range");
366
367 // if no target location is specified, then simply format the existing location
368 if(mem_loc == Location::none)
369 mem_loc = this->valid_locations;
370
371 // if there was no existing location, then let's continue with main memory
372 if(mem_loc == Location::none)
373 mem_loc = Location::main;
374
375 // reset memory location
376 this->valid_locations = Location::none;
377
378 // format main memory?
379 if(*(mem_loc & Location::main))
380 {
381 if(this->mem_main != nullptr)

383 else
384 this->_alloc_main(Init::format_to_zero);
385 this->valid_locations = this->valid_locations | Location::main;
386 }
387
388 // format cuda memory?
389 if(*(mem_loc & Location::cuda))
390 {
391 if(this->mem_cuda != nullptr)
392 Memory::memset_cuda(reinterpret_cast<char*>(this->mem_cuda) + this_offset, 0, size);
393 else
394 this->_alloc_cuda(Init::format_to_zero);

430 View map_view(Location mem_loc, Access access, std::size_t offset, std::size_t size)
431 {
432 std::unique_lock<std::mutex> lock(lock_mutex());
433
434 // do we need to create a view for a sub-region?
435 if((offset != std::size_t(0)) || (size != Arbiter::full_size))
436 {
437 // We are trying to map a write-accessible view to a sub-region of the memory block, so we need to add the
438 // read access flag independently of whether the user asked for it, because this will ensure that the
439 // entire memory block is up-to-date at the target memory location. If the read access flag was missing and
440 // the memory block was not already up-to-date on the target memory location, then the user's write
441 // operations would only overwrite the sub-region memory, but the region outside of the mapped view would
442 // remain in an undefined state.
443 if(*(access & Access::write))
444 access = access | Access::read;
445 }
446
447 // ensure that the offset is not out-of-bounds
448 if(this->bytes < offset)
449 throw RangeException("Memory::Block::map_view: memory offset is out of bounds");
450
451 // set size to entire (rest of) memory block if desired (which may be zero, which is fine)
452 if(size == full_size)
453 size = this->bytes - offset;
454 else if(this->bytes < offset + size)
455 throw RangeException("Memory::Block::map_view: memory range is out of bounds");
456
457 // do we need write access?
458 if(*(access & Access::write))
459 {
460 // do we allow overlapping write access?
461 if(*(access & Access::overlap))
462 {
463 // yes, overlap is allowed, but we must not have any mapped exclusive write views
464 if(this->_is_mapped_any_write())
465 {
466 // we can live with overlapping access, but some other write view is already mapped to this block,
467 // so ensure that a) it also allows overlapped access and b) it is mapped to the same location
468 if(!this->write_overlap)
469 throw AccessException("Memory::Block::map_view: cannot map exclusively write-mapped memory block for overlapping write access");
470 if(this->write_location != mem_loc)
471 throw AccessException("Memory::Block::map_view: memory block is write-mapped onto a different memory location");
472 }
473 // if no write overlapping write view was opened yet, then we also must not have mapped any read views yet;
474 // note that this if is never executed if an overlapping write view was already opened, which is intentional
475 // as one might not be able to create overlapping read-write views which are used in multi-threaded assembly
476 else if(this->_is_mapped_any_read())
477 {
478 throw AccessException("Memory::Block::map_view: cannot map read-mapped memory block for overlapping write access");
479 }
480 }
481 else if(this->_is_mapped_any())
482 {
483 // we require exclusive write access, but there is some other view mapped to this block
484 throw AccessException("Memory::Block::map_view: cannot map locked memory block for exclusive write access");
485 }
486 }
487
488 // do we need read access? if so, then make sure that there is no exclusive write lock on the memory block
489 if(*(access & Access::read) && !this->write_overlap)
490 {
491 if(this->_is_mapped_any_write())
492 throw AccessException("Memory::Block::map_view: cannot map write-locked memory block for read access");
493 }
494
495 // zero size? create an empty view in this case
496 if(size == std::size_t(0))
497 return View(this, mem_loc, access, size, nullptr);
498
499 // ensure that the target memory location is allocated
500 this->_alloc(mem_loc, Init::dont_care);
501
502 // read access requested?
503 if(*(access & Access::read))
504 {
505 // do we have an up-to-date memory location?
506 if(this->valid_locations == Location::none)
507 {
508 // emit a warning
509 std::cerr << "\n>>> MEMORY ARBITER: map_view: trying to map uninitialized memory for reading!\n";
510 //XABORTM("\n>>> MEMORY ARBITER: map_view: trying to map uninitialized memory for reading!\n");
511#ifdef _WIN32
512 Windows::dump_call_stack_to_stderr();
513#endif
514 }
515 // check whether the location that we want to map is out-of-date
516 else if(!*(this->valid_locations & mem_loc))
517 {
518 // the desired target memory location is out of date, so first make sure that
519 // main memory is up-to-date and then copy from main to target location
520 _copy_main_from_uptodate();
521 _copy_main_to(mem_loc);
522 }
523
524 // increase read lock counters
525 if(*(mem_loc & Location::main))
526 ++this->read_views_main;
527 if(*(mem_loc & Location::cuda))
528 ++this->read_views_cuda;
529 }
530
531 // write access requested?
532 if(*(access & Access::write))
533 {
534 if(this->write_views == std::size_t(0))
535 {
536 // remember whether we asked for overlapping or exclusive access
537 this->write_overlap = *(access & Access::overlap);
538
539 // remember memory write location
540 this->write_location = mem_loc;
541 }
542
543 // increase number of write views
544 ++this->write_views;
545 }
546
547 // create view for the requested memory location
548 if(mem_loc == Location::main)
549 {
550 return View(this, Location::main, access, size, reinterpret_cast<char*>(this->mem_main) + offset);
551 }
552 if(mem_loc == Location::cuda)
553 {
554 return View(this, Location::cuda, access, size, reinterpret_cast<char*>(this->mem_cuda) + offset);
555 }
556
557 // we should never arrive here
558 XABORTM("Memory::Block::map_view: invalid memory location");
559 }

561 void release_view(const View& view)
562 {
563 std::unique_lock<std::mutex> lock(lock_mutex());
564
565 // if the view had a write lock, then remove it
566 if(*(view.access() & Access::write))
567 {
568 if(--this->write_views == std::size_t(0))
569 {
570 // ok, this was the last open write view
571 this->valid_locations = this->write_location;
572 this->write_location = Memory::Location::none;
573 this->write_overlap = false;
574 }
575 }
576
577 // if the view had a read lock, then decrease the number of read views by one
578 if(*(view.access() & Access::read))
579 {
580 if(view.location() == Location::main)
581 {
582 XASSERT(this->read_views_main > 0u);
583 --this->read_views_main;
584 }
585 if(view.location() == Location::cuda)
586 {
587 XASSERT(this->read_views_cuda > 0u);
588 --this->read_views_cuda;
589 }
590 }
591 }

662 {
663 // try to allocate memory
664 if((this->mem_main = Memory::alloc_main(this->bytes)) == nullptr)
665 throw AllocException(this->bytes, "main");
666
667 // pre-format memory to ones if required
668 switch(mem_ini)
669 {
670 case Init::dont_care:
671#ifdef DEBUG
672 Memory::memset_main(this->mem_main, 0xFF, this->bytes);
673#endif
674 break;
675

677 // nothing to do here
678 break;

683 break;

1284 Arbiter Arbiter::clone(std::size_t offset, std::size_t size, bool clone_block) const
1285 {
1286 // no block assigned?
1287 if(this->_block == nullptr)
1288 {
1289 XASSERTM(offset == std::size_t(0), "arbiter clone range offset is out-of-bounds");
1290 XASSERTM((size == std::size_t(0)) || (size == full_size), "arbiter clone range size is out-of-bounds");
1291 return Arbiter();
1292 }
1293
1294 // validate range offset and size
1295 if(this->_block->bytes < offset)
1296 throw RangeException("Memory::Arbiter::clone(): invalid block memory offset");
1297 if((size != full_size) && (this->_block->bytes < offset + size))
1298 throw RangeException("Memory::Arbiter::clone(): invalid block memory range");
1299
1300 // create new block and copy content
1301 if(clone_block)
1302 {
1303 // clone entire block
1304 Arbiter arb(this->_block->bytes);
1305 arb._block->copy_from(*this->_block, Memory::Location::none, std::size_t(0), std::size_t(0), this->_block->bytes);
1306 arb._offset = this->_offset + offset;
1307 arb._size = size;
1308 return arb;
1309 }
1310 else
1311 {
1312 // clone selected memory range
1313 if(size == full_size)
1314 {
1315 if(this->_size == full_size)
1316 size = this->_block->bytes - offset;
1317 else
1318 size = this->_size - offset;
1319 }
1320 Arbiter arb(size);
1321 arb._block->copy_from(*this->_block, Memory::Location::none, std::size_t(0), this->_offset + offset, size);
1322 return arb;
1323 }
1324 }
