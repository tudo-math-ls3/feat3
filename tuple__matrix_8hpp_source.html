376 void restore_from_checkpoint_data(std::vector<char> & data)
377 {
378 std::uint64_t isize = *(std::uint64_t*) data.data(); //get size of checkpointed _first
379 std::vector<char>::iterator start = std::begin(data) + sizeof(std::uint64_t);
380 std::vector<char>::iterator last_of_first = std::begin(data) + sizeof(std::uint64_t) + (int) isize;
381 std::vector<char> buffer_first(start, last_of_first);
382 _first.restore_from_checkpoint_data(buffer_first);
383
384 data.erase(std::begin(data), last_of_first);
385 _rest.restore_from_checkpoint_data(data);
386 }
387
389 std::uint64_t set_checkpoint_data(std::vector<char>& data, SerialConfig& config)
390 {
391 std::size_t old_size = data.size();
392 data.insert(std::end(data), sizeof(std::uint64_t), 0); //add placeholder
393 std::uint64_t ireal_size = _first.set_checkpoint_data(data, config); //add data of _first to the overall checkpoint and save its size
394 char* csize = reinterpret_cast<char*>(&ireal_size);
395 for(std::size_t i(0) ; i < sizeof(std::uint64_t) ; ++i) //overwrite the guessed datalength
396 {
397 data[old_size + i] = csize[i];
398 }
399
400 return sizeof(std::uint64_t) + ireal_size + _rest.set_checkpoint_data(data, config); //generate and add checkpoint data for the _rest
401 }
