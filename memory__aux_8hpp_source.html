119 template<typename Tdst_, typename Tsrc_>
120 inline void convert_main(Tdst_* dst, const Tsrc_* src, const Index count)
121 {
122 if(count <= Index(0))
123 return;
124
125 XASSERT(dst != nullptr);
126 XASSERT(src != nullptr);
127
128 // check for overlapping memory regions
129 {
130 const char* vdst = reinterpret_cast<const char*>(dst);
131 const char* vsrc = reinterpret_cast<const char*>(src);
132 XASSERTM((vdst < vsrc) || (vsrc + std::size_t(count)*sizeof(Tsrc_) <= vdst), "source and destination memory regions overlap");
133 XASSERTM((vsrc < vdst) || (vdst + std::size_t(count)*sizeof(Tdst_) <= vsrc), "source and destination memory regions overlap");
134 }
135
136 for(Index i = 0u; i < count; ++i)
137 dst[i] = Tdst_(src[i]);
138 }
