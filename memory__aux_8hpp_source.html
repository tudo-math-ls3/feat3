120 template<typename Tdst_, typename Tsrc_>
121 inline void convert_main(Tdst_* dst, const Tsrc_* src, const Index count)
122 {
123 if(count <= Index(0))
124 return;
125
126 XASSERT(dst != nullptr);
127 XASSERT(src != nullptr);
128
129 // check for overlapping memory regions
130 {
131 const char* vdst = reinterpret_cast<const char*>(dst);
132 const char* vsrc = reinterpret_cast<const char*>(src);
133 XASSERTM((vdst < vsrc) || (vsrc + std::size_t(count)*sizeof(Tsrc_) <= vdst), "source and destination memory regions overlap");
134 XASSERTM((vsrc < vdst) || (vdst + std::size_t(count)*sizeof(Tdst_) <= vsrc), "source and destination memory regions overlap");
135 }
136
137 for(Index i = 0u; i < count; ++i)
138 dst[i] = Tdst_(src[i]);
139 }
