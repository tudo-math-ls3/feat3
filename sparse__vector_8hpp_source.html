92 explicit SparseVector(Index size_in, const std::map<IT_, DT_>& map)
93 {
94 Index nnze_in = map.size();
95 this->_scalar_index.push_back(size_in);
96 this->_scalar_index.push_back(nnze_in);
97 this->_elements.push_back(Memory::Arbiter(sizeof(DT_) * nnze_in));
98 this->_elements_size.push_back(nnze_in);
99 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * nnze_in));
100 this->_indices_size.push_back(nnze_in);
101
102 if(nnze_in == 0u)
103 return;
104
105 auto ind_view = this->indices_view_w();
106 auto ele_view = this->elements_view_w();
107 auto* id_p = ind_view.get_w();
108 auto* ele_p = ele_view.get_w();
109
110 for(auto mp = map.cbegin(); mp != map.cend(); ++mp, ++id_p, ++ele_p)
111 {
112 *id_p = mp->first;
113 *ele_p = mp->second;
114 }
115 }

137 this->_elements_size.push_back(elements_in.size());
138 this->_indices.push_back(indices_in.elements_arbiter().attach());
139 this->_indices_size.push_back(indices_in.size());
140 }

535 void read_from(FileMode mode, const String& filename)
536 {
537 std::ios_base::openmode bin = std::ifstream::in | std::ifstream::binary;
538 if(mode == FileMode::fm_mtx)
539 bin = std::ifstream::in;
540 std::ifstream file(filename.c_str(), bin);
541 if (! file.is_open())
542 XABORTM("Unable to open Vector file " + filename);
543 read_from(mode, file);
544 file.close();
545 }
546
553 void read_from(FileMode mode, std::istream& file)
554 {
555 switch(mode)
556 {
557 case FileMode::fm_binary:
558 [[fallthrough]];
559
560 case FileMode::fm_sv:
561 this->template _deserialize<double, std::uint64_t>(FileMode::fm_sv, file);
562 break;
563
564 case FileMode::fm_mtx:
565 {
566 this->clear();
567
568 Index rows;
569 Index nnz;
570 String line;
571 std::getline(file, line);
572 if (line.find("%%MatrixMarket matrix coordinate real general") == String::npos)
573 {
574 XABORTM("Input-file is not a compatible mtx-file");
575 }
576 while(!file.eof())
577 {
578 std::getline(file,line);
579 if (file.eof())
580 XABORTM("Input-file is empty");
581
582 String::size_type begin(line.find_first_not_of(" "));
583 if (line.at(begin) != '%')
584 break;
585 }
586 {
587 String::size_type begin(line.find_first_not_of(" "));
588 line.erase(0, begin);
589 String::size_type end(line.find_first_of(" "));
590 String srows(line, 0, end);
591 rows = (Index)atol(srows.c_str());
592 line.erase(0, end);
593
594 begin = line.find_first_not_of(" ");
595 line.erase(0, begin);
596 end = line.find_first_of(" ");
597 String scols(line, 0, end);
598 Index cols((Index)atol(scols.c_str()));
599 line.erase(0, end);
600 if (cols != 1)
601 XABORTM("Input-file is no sparse-vector-file");
602
603 begin = line.find_first_not_of(" ");
604 line.erase(0, begin);
605 end = line.find_first_of(" ");
606 String snnz(line, 0, end);
607 nnz = (Index)atol(snnz.c_str());
608 line.erase(0, end);
609 }
610
611 this->move(SparseVector<DT_, IT_>(rows, nnz));
612
613 Memory::TypedView<IT_> idx_view(this->indices_view_w());
614 Memory::TypedView<DT_> val_view(this->elements_view_w());
615
616 IT_ * pind(idx_view.get_w());
617 DT_ * pval(val_view.get_w());
618
619 while(!file.eof())
620 {
621 std::getline(file, line);
622 if (file.eof())
623 break;
624
625 String::size_type begin(line.find_first_not_of(" "));
626 line.erase(0, begin);
627 String::size_type end(line.find_first_of(" "));
628 String srow(line, 0, end);
629 IT_ row((IT_)atol(srow.c_str()));
630 --row;
631 line.erase(0, end);
632
633 begin = line.find_first_not_of(" ");
634 line.erase(0, begin);
635 end = line.find_first_of(" ");
636 line.erase(0, end);
637
638 begin = line.find_first_not_of(" ");
639 line.erase(0, begin);
640 end = line.find_first_of(" ");
641 String sval(line, 0, end);
642 DT_ tval((DT_)atof(sval.c_str()));
643
644 *pval = tval;
645 *pind = row;
646 ++pval;
647 ++pind;
648 }
649 }
650 break;
651
652 default:
653 XABORTM("Filemode not supported!");
654 }
655 }

689 void write_out(FileMode mode, std::ostream& file) const
690 {
691 switch(mode)
692 {
693 case FileMode::fm_binary:
694 [[fallthrough]];
695
696 case FileMode::fm_sv:
697 this->template _serialize<double, std::uint64_t>(FileMode::fm_sv, file);
698 break;
699
700 case FileMode::fm_mtx:
701 {
702
703 file << "%%MatrixMarket matrix coordinate real general\n";
704 file << this->size() << " " << 1 << " " << this->num_nzes() << "\n";
705 file << std::scientific << std::setprecision(Type::Traits<DT_>::format_precision);
706
707 const Index u_elem(this->num_nzes());
708 const Memory::TypedView<IT_> idx_view = this->indices_view_r();
709 const Memory::TypedView<DT_> val_view = this->elements_view_r();
710 for (Index i(0) ; i < u_elem ; ++i)
711 {
712 file << (idx_view[i]+1) << " " << 1 << " " << val_view[i] << "\n";
713 }
714 }
715 break;
716
717 default:
718 XABORTM("Filemode not supported!");
719 }

723 void permute(Adjacency::Permutation & perm)
724 {
725 if (perm.size() == 0)
726 return;
727
728 std::map<IT_, DT_> xm;
729
730 auto ip = perm.inverse();
731 const Index * const inv_pos(ip.get_perm_pos());
732
733 Memory::TypedView<DT_> xv(this->elements_view_rw());
734 Memory::TypedView<IT_> xi(this->indices_view_rw());
735 for (Index i(0) ; i < this->num_nzes() ; ++i)
736 {
737 xm.emplace(IT_(inv_pos[xi[i]]), xv[i]);
738 }
739
740 Index k(0);
741 for(const auto& x : xm)
742 {
743 xi[k] = x.first;
744 xv[k] = x.second;
745 ++k;
746 }
747 }

768 bool same_layout(const SparseVector& other) const
769 {
770 if(this->size() != other.size())
771 return false;
772 if(this->num_nzes() != other.num_nzes())
773 return false;
774
775 if(this->num_nzes() == Index(0))
776 return true;
777
778 // check if the indices arbiters are the same
779 if(this->indices_arbiter() == other.indices_arbiter())
780 return true;
781
782 const Memory::TypedView<IT_> idx_a = this->indices_view_r();
783 const Memory::TypedView<IT_> idx_b = other.indices_view_r();
784
785 // check all array entries
786 const Index n = this->num_nzes();
787 for(Index i(0); i < n; ++i)
788 {
789 if(idx_a[i] != idx_b[i])
790 return false;
791 }
792
793 // okay, arrays are identical
794 return true;
795 }

806 void print(std::ostream & os, bool print_dense) const
807 {
808 if(this->empty())
809 {
810 os << "[]";
811 return;
812 }
813 const Memory::TypedView<IT_> idx_view(this->indices_view_r());
814 const Memory::TypedView<DT_> val_view(this->elements_view_r());
815 const Index n = this->size();
816 const Index nze = this->num_nzes();
817 os << "[";
818 Index k = 0;
819 for (Index i(0) ; i < nze ; ++i)
820 {
821 if(print_dense)
822 {
823 for(; k < idx_view[i]; ++k)
824 os << " " << DT_(0);
825 }
826 else
827 os << " " << idx_view[i] << ':';
828
829 os << " " << val_view[i];
830 ++k;
831 }
832 if(print_dense)
833 {
834 for(; k < n; ++k)
835 os << " " << DT_(0);
836 }
837 os << "]";
838 }
