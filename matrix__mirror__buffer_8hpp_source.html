75 explicit MatrixMirrorBuffer(Index rows_in, Index columns_in, Index used_elements_in, Index entries_per_nonzero_in)
76 {
77 this->_scalar_index.push_back(rows_in);
78 this->_scalar_index.push_back(columns_in);
79 this->_scalar_index.push_back(used_elements_in);
80 this->_scalar_index.push_back(entries_per_nonzero_in);
81
82 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * (num_rows() + 1)));
83 this->_indices_size.push_back(num_rows() + 1);
84
85 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * num_nzes()));
86 this->_indices_size.push_back(num_nzes());
87
88 this->_elements.push_back(Memory::Arbiter(sizeof(DT_) * (num_nzes() * entries_per_nonzero())));
89 this->_elements_size.push_back(num_nzes() * entries_per_nonzero());
90 }

99 explicit MatrixMirrorBuffer(const Adjacency::Graph & graph, Index entries_per_nonzero_in) :
100 MatrixMirrorBuffer(graph.get_num_nodes_domain(), graph.get_num_nodes_image(), graph.get_num_indices(), entries_per_nonzero_in)
101 {
102 const Index num_nnze = graph.get_num_indices();
103 const Index num_rows = graph.get_num_nodes_domain();
104 const Index * dom_ptr(graph.get_domain_ptr());
105 const Index * img_idx(graph.get_image_idx());
106
107 Memory::TypedView<IT_> prow_ptr = this->row_ptr_view_w();
108 Memory::TypedView<IT_> pcol_idx = this->col_idx_view_w();
109
110 FEAT_PRAGMA_OMP(parallel for)
111 for(Index i = 0; i <= num_rows; ++i)
112 prow_ptr[i] = IT_(dom_ptr[i]);
113
114 FEAT_PRAGMA_OMP(parallel for)
115 for(Index i = 0; i < num_nnze; ++i)
116 pcol_idx[i] = IT_(img_idx[i]);
117 }

132 explicit MatrixMirrorBuffer(const Index rows_in, const Index columns_in, Index entries_per_nonzero_in,
133 DenseVector<IT_, IT_> & col_idx_in, DenseVector<DT_, IT_> & val_in, DenseVector<IT_, IT_> & row_ptr_in)
134 {
135 XASSERT(val_in.size() % entries_per_nonzero_in == 0);
136 XASSERT(val_in.size() == col_idx_in.size() * entries_per_nonzero_in);
137
138 this->_scalar_index.push_back(rows_in);
139 this->_scalar_index.push_back(columns_in);
140 this->_scalar_index.push_back(col_idx_in.size());
141 this->_scalar_index.push_back(entries_per_nonzero_in);
142
143 this->_indices.push_back(row_ptr_in.elements_arbiter().attach());
144 this->_indices_size.push_back(row_ptr_in.size());
145 this->_indices.push_back(col_idx_in.elements_arbiter().attach());
146 this->_indices_size.push_back(col_idx_in.size());
147 this->_elements.push_back(val_in.elements_arbiter().attach());
148 this->_elements_size.push_back(val_in.size());
149 }
