155 explicit GMRES(const String& section_name, const PropertyMap* section,
156 const MatrixType& matrix, const FilterType& filter, std::shared_ptr<PrecondType> precond = nullptr) :
157 BaseClass("GMRES", section_name, section, precond),
158 _system_matrix(matrix),
159 _system_filter(filter),
160 _inner_res_scale(DataType(0))
161 {
162 // we always need to compute defect norms
163 this->_force_def_norm_calc = true;
164
165 // set communicator by system matrix
166 this->_set_comm_by_matrix(matrix);
167
168 // Set _inner_res_scale by parameter or use default value
169 auto inner_res_scale_p = section->query("inner_res_scale");
170 if(inner_res_scale_p.second && (!inner_res_scale_p.first.parse(this->_inner_res_scale) || (this->_inner_res_scale < DataType(0))))
171 throw ParseError(section_name + ".inner_res_scale", inner_res_scale_p.first, "a non-negative float");
172
173 // Check if we have set _krylov_vim
174 auto krylov_dim_p = section->query("krylov_dim");
175 if(!krylov_dim_p.second)
176 throw ParseError("GMRES config section is missing the mandatory krylov_dim!");
177
178 if(!krylov_dim_p.first.parse(this->_krylov_dim) || (this->_krylov_dim <= Index(0)))
179 throw ParseError(section_name + ".krylov_dim", krylov_dim_p.first, "a positive integer");
180
181 // Set the orthogonalization strategy by parameter or use the default (modified Gram-Schmidt)
182 auto ortho_p = section->query("orthogonalizer");
183 if(ortho_p.second)
184 {
185 if(ortho_p.first == "mgs")
186 this->_ortho_type = KrylovOrthogonalizerType::modified_gram_schmidt;
187 else if(ortho_p.first == "cgs")
188 this->_ortho_type = KrylovOrthogonalizerType::classical_gram_schmidt;
189 else if(ortho_p.first == "cgs2")
190 this->_ortho_type = KrylovOrthogonalizerType::cgs2;
191 else
192 throw ParseError(section_name + ".orthogonalizer", ortho_p.first, "one of: mgs, cgs, cgs2");
193 }
194
195 this->set_plot_name("GMRES("+stringify(_krylov_dim)+")");
196 }

314 virtual Status _apply_intern(VectorType& vec_sol, const VectorType& vec_rhs)
315 {
316 IterationStats pre_iter(*this);
317 Statistics::add_solver_expression(std::make_shared<ExpressionStartSolve>(this->name()));
318 const MatrixType& matrix(this->_system_matrix);
319 const FilterType& filter(this->_system_filter);
320
321 // compute initial defect
322 Status status = this->_set_initial_defect(this->_subspace->at(0), vec_sol);
323
324 pre_iter.destroy();
325
326 const Index krylov_dim = this->_subspace->get_krylov_dim();
327
328 // outer GMRES loop
329 while(status == Status::progress)
330 {
331 IterationStats stat(*this);
332
333 _q.clear();
334 _s.clear();
335 _c.clear();
336 _q.push_back(this->_def_cur);
337
338 // normalize v[0]
339 this->_subspace->at(0).scale(this->_subspace->at(0), DataType(1) / _q.back());
340
341 // inner GMRES loop
342 Index i(0);
343 while(i < krylov_dim)
344 {
345 // apply preconditioner
346 if(!this->_apply_precond(this->_vec_w, this->_subspace->at(i), filter))
347 {
348 stat.destroy();
349 Statistics::add_solver_expression(std::make_shared<ExpressionEndSolve>(this->name(), Status::aborted, this->get_num_iter()));
350 return Status::aborted;
351 }
352 //filter.filter_cor(this->_vec_z.at(i));
353
354 // v[i+1] := A*w
355 matrix.apply(this->_subspace->at(i+1), this->_vec_w);
356 filter.filter_def(this->_subspace->at(i+1));
357
358 // orthogonalize v_{i+1}
359 DataType alpha = this->_subspace->orthogonalize(this->_h.at(i).data(), i+1);
360
361 // and now normalize
362 this->_subspace->at(i+1).scale(this->_subspace->at(i+1), DataType(1) / alpha);
363
364 // apply Givens rotations
365 for(Index k(0); k < i; ++k)
366 {
367 DataType t(this->_h.at(i).at(k));
368 this->_h.at(i).at(k ) = this->_c.at(k) * t + this->_s.at(k) * this->_h.at(i).at(k+1);
369 this->_h.at(i).at(k+1) = this->_s.at(k) * t - this->_c.at(k) * this->_h.at(i).at(k+1);
370 }
371
372 // compute beta
373 DataType beta = Math::sqrt(Math::sqr(this->_h.at(i).at(i)) + Math::sqr(alpha));
374
375 // compute next plane rotation
376 _s.push_back(alpha / beta);
377 _c.push_back(this->_h.at(i).at(i) / beta);
378
379 this->_h.at(i).at(i) = beta;
380 this->_q.push_back(this->_s.back() * this->_q.at(i));
381 this->_q.at(i) *= this->_c.back();
382
383 // push our new defect
384 if(++i < this->_krylov_dim)
385 {
386 // get the absolute defect
387 DataType def_cur = Math::abs(this->_q.back());
388
389 // did we diverge?
390 if((def_cur > this->_div_abs) || (def_cur > (this->_div_rel * this->_def_init)))
391 break;
392
393 // minimum number of iterations performed?
394 if(!(this->_num_iter < this->_min_iter))
395 {
396 // did we converge?
397 if((def_cur <= _inner_res_scale * this->_tol_abs) &&
398 ((def_cur <= _inner_res_scale * (this->_tol_rel * this->_def_init)) || (def_cur <= _inner_res_scale * this->_tol_abs_low) ))
399 break;
400
401 // maximum number of iterations performed?
402 if(this->_num_iter >= this->_max_iter)
403 break;
404 }
405
406 // set our pseudo defect
407 // increase iteration count
408 ++this->_num_iter;
409
410 // compute new defect
411 this->_def_cur = def_cur;
412
413 // plot?
414 if(this->_plot_iter())
415 {
416 String msg = this->_plot_name
417 + "* " + stringify(this->_num_iter).pad_front(this->_iter_digits)
418 + " : " + stringify_fp_sci(this->_def_cur);
419 this->_print_line(msg);
420 }
421 }
422 }
423
424 if(this->_plot_iter())
425 {
426 /* DEBUG */
427 String msg = this->_plot_name
428 + "* " + stringify(this->_num_iter).pad_front(this->_iter_digits)
429 + " : " + stringify_fp_sci(Math::abs(this->_q.back()))
430 + " <= " + stringify_fp_sci(this->_inner_res_scale * this->_tol_abs)
431 + " && " + stringify_fp_sci(this->_inner_res_scale * this->_tol_rel);
432 this->_print_line(msg);
433 }
434
435 Index n = Math::min(i, krylov_dim);
436
437 // solve H*q = q
438 for(Index k(n); k > 0;)
439 {
440 --k;
441 this->_q.at(k) /= this->_h.at(k).at(k);
442 for(Index j(k); j > 0;)
443 {
444 --j;
445 this->_q.at(j) -= this->_h.at(k).at(j) * this->_q.at(k);
446 }
447 }
448
449 // update solution
450 this->_vec_w.format();
451
452 // dispatch vector correction to subspace (fused kernel possible)
453 this->_subspace->correct_vector(this->_vec_w, this->_q.data(), n, DataType(1));
454
455 // re-use v[0] for preconditioned correction
456 if(!this->_apply_precond(this->_subspace->at(0), this->_vec_w, filter))
457 {
458 stat.destroy();
459 Statistics::add_solver_expression(std::make_shared<ExpressionEndSolve>(this->name(), Status::aborted, this->get_num_iter()));
460 return Status::aborted;
461 }
462 vec_sol.axpy(this->_subspace->at(0));
463
464 // compute "real" residual
465 matrix.apply(this->_subspace->at(0), vec_sol, vec_rhs, -DataType(1));
466 filter.filter_def(this->_subspace->at(0));
467
468 // set the current defect
469 status = this->_set_new_defect(this->_subspace->at(0), vec_sol);
470
471 }
472
473 // finished
474 Statistics::add_solver_expression(std::make_shared<ExpressionEndSolve>(this->name(), status, this->get_num_iter()));
475 return status;
476 }

478 std::unique_ptr<OrthoType> _create_orthogonalizer(const VectorType& vec_w)
479 {
480 switch(_ortho_type)
481 {
482 case KrylovOrthogonalizerType::classical_gram_schmidt:
483 return std::make_unique<ClassicGramSchmidtOrthogonalizer<VectorType>>(vec_w, this->_krylov_dim);
484 case KrylovOrthogonalizerType::cgs2:
485 return std::make_unique<CGS2Orthogonalizer<VectorType>>(vec_w, this->_krylov_dim);
486 case KrylovOrthogonalizerType::modified_gram_schmidt:
487 default:
488 return std::make_unique<ModifiedGramSchmidtOrthogonalizer<VectorType>>(vec_w, this->_krylov_dim);
489 }
490 }
