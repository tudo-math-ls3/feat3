376#ifdef FEAT_HAVE_MPI
378 template<typename T_>
379 const Datatype& get_type()
380 {
381 // NOTE(mmuegge): The offsetof macro is only defined for "standard layout types".
382 // Any other way to take offsets is probably highly compiler dependent or UB.
383 static_assert(std::is_standard_layout_v<T_>, "MPI typemaps require standard-layout types!");
384
385 std::scoped_lock<std::recursive_mutex> lock(_mutex);
386
387 const std::type_index idx(typeid(T_));
388
389 const auto it = _types.find(idx);
390 if(it == _types.end())
391 {
392 // Type does not exist yet. Create it.
393 const Typemap desc = TypeDescription<T_>::typemap();
394
395 XASSERTM(desc.count() > 0, "TypeRegistry::get_type: empty typemap!");
396
397 // Unpack wrapped mpi types
398 std::vector<MPI_Datatype> mpi_types;
399 mpi_types.reserve(desc.count());
400 std::transform(
401 desc._types.begin(),
402 desc._types.end(),
403 std::back_inserter(mpi_types),
404 [](const Datatype& d) { return d.dt; });
405
406 // Cast displacements to correct type
407 std::vector<MPI_Aint> displacements;
408 displacements.reserve(desc.count());
409 std::transform(
410 desc._displacements.begin(),
411 desc._displacements.end(),
412 std::back_inserter(displacements),
413 [](const std::size_t d) { return static_cast<MPI_Aint>(d); });
414
415 // Create temporary type
416 MPI_Datatype tmptype;
417 MPI_Type_create_struct(static_cast<int>(desc.count()), desc._blocklengths.data(), displacements.data(), mpi_types.data(), &tmptype);
418
419 // The temporary might be smaller than the actual type T_, because there might be padding at the end of T_.
420 // That would lead to issues when sending multiple T_s.
421 // Resize the type to include padding.
422 MPI_Datatype newtype;
423 MPI_Type_create_resized(tmptype, 0, sizeof(T_), &newtype);
424
425 // Commit type
426 MPI_Type_commit(&newtype);
427
428 // Free temporary type
429 MPI_Type_free(&tmptype);
430
431 // Store type for later
432 const auto [iter, _] = _types.try_emplace(idx, Datatype(newtype, sizeof(T_)));
433
434 return iter->second;
435 }
436 else
437 {
438 return it->second;
439 }
440 }
441
445 void free()
446 {
447 std::scoped_lock<std::recursive_mutex> lock(_mutex);
448
449 for(auto& [_, dt] : _types)
450 {
451 MPI_Type_free(&dt.dt);
452 }
453 _types.clear();
454 }
455#else
457 template<typename T_>
458 const Datatype& get_type()
459 {
460 std::scoped_lock<std::recursive_mutex> lock(_mutex);
461
462 const std::type_index idx(typeid(T_));
463
464 auto it = _types.find(idx);
465 if(it == _types.end())
466 {
467 const auto [iter, _] = _types.try_emplace(idx, Datatype(next_mock_id++, sizeof(T_)));
468 return iter->second;
469 }
470 else
471 {
472 return it->second;
473 }
474
475 return _types.at(idx);
476 }
477
481 void free()
482 {
483 std::scoped_lock<std::recursive_mutex> lock(_mutex);
484 _types.clear();
485 }
486#endif
