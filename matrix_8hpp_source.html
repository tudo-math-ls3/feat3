423 void apply(VectorTypeL& r, const VectorTypeR& x, const VectorTypeL& y, const DataType alpha = DataType(1)) const
424 {
425 // copy y to r
426 r.copy(y);
427
428 // convert from type-1 to type-0
429 r.from_1_to_0();
430
431 // r <- r + alpha*A*x

633 LocalMatrix_ convert_to_1() const
634 {
635 ASSERTM(_col_gate, "Column gate is not set!");
636 ASSERTM(_row_gate, "Row gate is not set!");
637 LocalMatrix_ locmat = _matrix.clone(LAFEM::CloneMode::Weak);
638 SynchMatrix<LocalMatrix_, RowMirror_> synch(*_row_gate->_comm, _row_gate->_ranks, _row_gate->_mirrors, _col_gate->_mirrors);
639 synch.init(locmat);
640 synch.exec(locmat);
641 return locmat;
642 }

672 void convert_to_1(LocalMatrix_& matrix_sync_1, bool full_copy = true) const
673 {

676 {
677 SynchMatrix<LocalMatrix_, RowMirror_> synch(*_row_gate->_comm, _row_gate->_ranks, _row_gate->_mirrors, _col_gate->_mirrors);
678 synch.init(matrix_sync_1);
679 synch.exec(matrix_sync_1);
680 }
