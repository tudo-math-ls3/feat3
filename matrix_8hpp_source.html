433 void apply(VectorTypeL& r, const VectorTypeR& x, const VectorTypeL& y, const DataType alpha = DataType(1)) const
434 {
435 // copy y to r
436 r.copy(y);
437
438 // convert from type-1 to type-0
439 r.from_1_to_0();
440
441 // r <- r + alpha*A*x
442 _matrix.apply(r.local(), x.local(), r.local(), alpha);

444 // synchronize r

643 LocalMatrix_ convert_to_1() const
644 {
645 ASSERTM(_col_gate, "Column gate is not set!");
646 ASSERTM(_row_gate, "Row gate is not set!");
647 LocalMatrix_ locmat = _matrix.clone(LAFEM::CloneMode::Weak);
648 SynchMatrix<LocalMatrix_, RowMirror_> synch(*_row_gate->_comm, _row_gate->_ranks, _row_gate->_mirrors, _col_gate->_mirrors);
649 synch.init(locmat);
650 synch.exec(locmat);
651 return locmat;
652 }
