91 template<typename DT_, typename IT_>
92 class MirrorHelper<LAFEM::VectorMirror<DT_, IT_>>
93 {
94 public:
95 typedef LAFEM::VectorMirror<DT_, IT_> InMirrorType;
96
97 template<typename OutMirror_, typename Vec_>
98 static std::pair<Index, Index> fill_mirror(OutMirror_& mirror, const InMirrorType& input, const Vec_& vec, std::pair<Index, Index> offsets = {0u, 0u})
99 {
100 // get indices (host-side views into the mirror index arrays)
101 const auto indices_view = input.indices_view_r();
102 const auto* indices = indices_view.get_r();
103 const Index& dof_offset = offsets.first;
104 const Index& ptr_offset = offsets.second;
105 auto target_view = mirror.indices_view_w();
106 auto* target = target_view.get_w() + ptr_offset;
107 constexpr int dim = dim_vector<Vec_>;
108 for(Index k = 0; k < input.num_indices(); ++k)
109 {
110 for(int d = 0; d < dim; ++d)
111 {
112 target[k*dim + Index(d)] = IT_(indices[k]*dim + Index(d) + dof_offset);
113 }
114 }
115 offsets.first += vec.size_raw();
116 offsets.second += input.num_indices()*dim;
117 return offsets;
118 }

413 explicit AmaVankaGMRES(const MatrixType& matrix, const FilterType& filter, std::shared_ptr<VankaType> vanka, Index krylov_dim,
414 DataType inner_res_scale = DataType(0)) :
415 BaseClass("AmaVankaGMRES(" + stringify(krylov_dim) + ")"),
416 _system_matrix(matrix),
417 _system_filter(filter),
418 _comm(_system_matrix.get_comm()),

428 _use_cuda_gmres(false),
429 _prefetch_data(false),
430 _always_use_parallel_kernel(false)
431 {
432 if(_comm)
433 _gate.set_comm(_comm);
434 // we always need to compute defect norms
435 this->_force_def_norm_calc = true;
436
437 // set communicator by system matrix
438 this->_set_comm_by_matrix(matrix);
439 set_inner_res_scale(inner_res_scale);
440
441 const Index local_mat_rows = _system_matrix.local().num_rows_raw();
442 const Index local_mat_cols = _system_matrix.local().num_cols_raw();
443
444 // construct our internal gate
445 if(matrix.get_row_gate())
446 {
447 const auto& row_gate = *matrix.get_row_gate();
448 const auto& ranks = row_gate.get_ranks();
449 const auto& mirrors = row_gate.get_mirrors();
450 const auto& freqs = row_gate.get_freqs();
451 for(Index k = 0; k < ranks.size(); ++k)
452 {
453 //construct mirror
454 const Index num_indices = mirrors.at(k).buffer_size(freqs);
455 MirrorType mirror(local_mat_rows, num_indices);
456 auto offsets = Intern::MirrorHelper<MirrorRow_>::fill_mirror(mirror, mirrors.at(k), freqs);
457 XASSERTM(offsets.first == local_mat_rows, "Offset does not match overall mirror size");
458 XASSERTM(offsets.second == num_indices, "Offset does not match overall mirror indices");
459
460 this->_gate.push(ranks.at(k), std::move(mirror));
461 this->_gate.set_sync_location(row_gate.get_sync_location());
462 }
463 // convert frequency
464 VecType tmp_freq;
465 tmp_freq.convert(freqs);
466 this->_gate.compile(std::move(tmp_freq));
467
468 // our internal global matrix uses the new gate
469 this->_internal_system_mat = GlobalMatType(&this->_gate, &this->_gate, local_mat_rows, local_mat_cols);
470 this->_internal_filter = FilType(local_mat_rows, 0u);
471 }

793 DataType _orthogonalize_cpu(const Index i)
794 {
795 VecType& target = this->_vec_v.at(i+1).local();
796 const Index num_basis = i + 1u;
797 const bool distributed = this->_comm && (this->_comm->size() > 1);
798
799 // first classical Gram-Schmidt sweep: batched projection dots + appended self-dot
800 DataType self_dot = distributed
801 ? _calc_triple_dots_s(this->_h.at(i), target, this->_vec_v_elements, this->_gate.get_freqs(), num_basis)
802 : _calc_dots_s(this->_h.at(i), target, this->_vec_v_elements, num_basis);
803 _internal_flop_counter += (num_basis + 1u) * 2u * _global_vec_size;
804
805 DataType sum_sqr(0);
806 for(Index k(0); k < num_basis; ++k)
807 sum_sqr += Math::sqr(this->_h.at(i).at(k));
808
809 // subtract the projection in a single batched cumulative axpy: target -= sum_k h[k] * v[k]
810 _add_vecs_s(target, this->_vec_v_elements, this->_h.at(i), num_basis, DataType(-1));
811 _internal_flop_counter += num_basis * 2u * _global_vec_size;
812
813 DataType nrm_sqr = self_dot - sum_sqr;
814
815 // conditional corrective reorthogonalization sweep (cgs2); the second batched reduction
816 // supplies both the corrections and the updated self-dot, so no extra reduction is needed
817 if((this->_ortho_type == KrylovOrthogonalizerType::cgs2)
818 && (this->_force_reortho || (nrm_sqr < Math::sqr(this->_reortho_tol) * self_dot)))
819 {
820 DataType self_dot2 = distributed
821 ? _calc_triple_dots_s(this->_corr, target, this->_vec_v_elements, this->_gate.get_freqs(), num_basis)
822 : _calc_dots_s(this->_corr, target, this->_vec_v_elements, num_basis);
823 _internal_flop_counter += (num_basis + 1u) * 2u * _global_vec_size;
824 DataType sum_sqr2(0);
825 for(Index k(0); k < num_basis; ++k)
826 {
827 this->_h.at(i).at(k) += this->_corr.at(k);
828 sum_sqr2 += Math::sqr(this->_corr.at(k));
829 }
830 _add_vecs_s(target, this->_vec_v_elements, this->_corr, num_basis, DataType(-1));
831 _internal_flop_counter += num_basis * 2u * _global_vec_size;
832 nrm_sqr = self_dot2 - sum_sqr2;
833 }
834
835 // recover the norm algebraically; fall back to an explicit norm2 on heavy cancellation
836 // (guarantees a positive result so the subsequent normalization cannot produce a NaN)
837 if(nrm_sqr <= Math::Limits<DataType>::epsilon() * DataType(100))
838 {
839 _internal_flop_counter += 2u * _global_vec_size;
840 return this->_vec_v.at(i+1).norm2();
841 }
842 return Math::sqrt(nrm_sqr);
843 }

858 Status _set_initial_defectp(const GlobalVecType& vec_def, const GlobalVecType&)
859 {
860 // reset iteration variables
861 this->_def_init = this->_def_prev = this->_def_cur = DataType(0);
862 this->_num_iter = Index(0);
863 this->_num_stag_iter = Index(0);
864
865 // first, let's see if we have to compute the defect at all
866 bool calc_def = this->_force_def_norm_calc
867 || (this->_min_iter < this->_max_iter)
868 || (this->_plot_iter() || this->_plot_summary())
869 || (this->_min_stag_iter > Index(0));
870
871 // if we don't have to compute the initial defect, we can exit here
872 if(!calc_def)
873 return Status::progress;
874
875 // compute new initial defect
876 this->_def_init = this->_def_cur = this->_def_prev = vec_def.norm2();
877 Statistics::add_solver_expression(std::make_shared<ExpressionDefect>(this->name(), this->_def_init, this->get_num_iter()));
878 _internal_flop_counter += 2u * _global_vec_size;
879
880 // plot iteration line?
881 if(this->_plot_iter())
882 this->_plot_iter_line(Index(0), this->_def_init, this->_def_init);
883
884 // ensure that the defect is neither NaN nor infinity
885 if(!Math::isfinite(this->_def_init))
886 return Status::aborted;
887
888 // check against low absolute tolerance; this may be fulfilled if we have only a handful of dofs
889 if(this->_def_init < this->_tol_abs_low)
890 return Status::success;
891
892 // check if the initial defect is zero; we test against eps^2 here
893 if(this->_def_init <= Math::sqr(Math::eps<DataType>()))
894 return Status::success;
895
896 // continue iterating
897 return Status::progress;
898 }

1015 virtual Status _apply_intern(GlobalVecType& vec_sol, const GlobalVecType& vec_rhs)
1016 {
1017 FEAT::Util::ScopedNvtxRange _nvtx_total("Internal Apply");
1018 IterationStats pre_iter(*this);
1019 Statistics::add_solver_expression(std::make_shared<ExpressionStartSolve>(this->name()));
1020 const GlobalMatType& matrix(this->_internal_system_mat);
1021 const FilType& filter(this->_internal_filter);
1022
1023 // compute initial defect
1024 Status status = this->_set_initial_defectp(this->_vec_v.at(0), vec_sol);
1025
1026 pre_iter.destroy();
1027
1028
1029
1030 // outer GMRES loop
1031 while(status == Status::progress)
1032 {
1033 IterationStats stat(*this);
1034
1035 _q.clear();
1036 _s.clear();
1037 _c.clear();
1038 _q.push_back(this->_def_cur);
1039
1040 // normalize v[0]
1041 this->_vec_v.at(0).scale(this->_vec_v.at(0), DataType(1) / _q.back());
1042 _internal_flop_counter += 2u*this->_vec_v.at(0).size_raw();
1043
1044 // inner GMRES loop
1045 Index i(0);
1046 { // "Inner GMRES Loop" scope (i is declared outside so it survives for the Hessenberg solve)
1047 FEAT::Util::ScopedNvtxRange _nvtx_inner("Inner GMRES Loop");
1048 while(i < this->_krylov_dim)
1049 {
1050 // if(i == 0)
1051 // {
1052 // std::cout << "Rank " << _comm->rank() << " :" << this->_vec_v[0].local() << "\n-------------\n";
1053 // }
1054 // apply vanka matrix
1055 {
1056 FEAT::Util::ScopedNvtxRange _nvtx("Solve Vanka");
1057 _apply_vanka(this->_vec_w, this->_vec_v.at(i));
1058 }
1059 // // sync by calculating mean
1060 // if(i == 0)
1061 // {
1062 // std::cout << "Rank " << _comm->rank() << " :" << this->_vec_w.local() << "\n-------------\n";
1063 // }
1064
1065 // v[i+1] := A*w
1066 {
1067 FEAT::Util::ScopedNvtxRange _nvtx("A*w");
1068 matrix.apply(this->_vec_v.at(i+1), this->_vec_w);
1069 filter.filter_def(this->_vec_v.at(i+1).local());
1070 _internal_flop_counter += 2u*_global_used_elements;
1071 }
1072
1073 // Gram-Schmidt process (classical / conditional CGS2), with the self-dot fused into the
1074 // batched reduction so the orthogonalized norm is recovered without a separate norm2
1075 const DataType alpha = _orthogonalize_cpu(i);
1076
1077 // normalize v[i+1]
1078 {
1079 FEAT::Util::ScopedNvtxRange _nvtx("Norm scale");
1080 this->_vec_v.at(i+1).scale(this->_vec_v.at(i+1), DataType(1) / alpha);
1081 _internal_flop_counter += 2u * _global_vec_size;
1082 }
1083
1084 {
1085 FEAT::Util::ScopedNvtxRange _nvtx("Givens Rotation");
1086 // apply Givens rotations
1087 for(Index k(0); k < i; ++k)
1088 {
1089 DataType t(this->_h.at(i).at(k));
1090 this->_h.at(i).at(k ) = this->_c.at(k) * t + this->_s.at(k) * this->_h.at(i).at(k+1);
1091 this->_h.at(i).at(k+1) = this->_s.at(k) * t - this->_c.at(k) * this->_h.at(i).at(k+1);
1092 }
1093 }
1094
1095
1096 {
1097 FEAT::Util::ScopedNvtxRange _nvtx("Push defect");
1098 // compute beta
1099 const DataType beta = Math::sqrt(Math::sqr(this->_h.at(i).at(i)) + Math::sqr(alpha));
1100
1101 // compute next plane rotation
1102 _s.push_back(alpha / beta);
1103 _c.push_back(this->_h.at(i).at(i) / beta);
1104
1105 this->_h.at(i).at(i) = beta;
1106 this->_q.push_back(this->_s.back() * this->_q.at(i));
1107 this->_q.at(i) *= this->_c.back();
1108 }
1109 // push our new defect
1110 if(++i < this->_krylov_dim)
1111 {
1112 // get the absolute defect
1113 DataType def_cur = Math::abs(this->_q.back());
1114
1115 // did we diverge?
1116 if((def_cur > this->_div_abs) || (def_cur > (this->_div_rel * this->_def_init)))
1117 break;
1118
1119 // minimum number of iterations performed?
1120 if(!(this->_num_iter < this->_min_iter))
1121 {
1122 // did we converge?
1123 if((def_cur <= _inner_res_scale * this->_tol_abs) &&
1124 ((def_cur <= _inner_res_scale * (this->_tol_rel * this->_def_init)) || (def_cur <= _inner_res_scale * this->_tol_abs_low) ))
1125 break;
1126
1127 // maximum number of iterations performed?
1128 if(this->_num_iter >= this->_max_iter)
1129 break;
1130 }
1131
1132 // set our pseudo defect
1133 // increase iteration count
1134 ++this->_num_iter;
1135
1136 // compute new defect
1137 this->_def_cur = def_cur;
1138
1139 // plot?
1140 if(this->_plot_iter())
1141 {
1142 String msg = this->_plot_name
1143 + "* " + stringify(this->_num_iter).pad_front(this->_iter_digits)
1144 + " : " + stringify_fp_sci(this->_def_cur);
1145 this->_print_line(msg);
1146 }
1147 }
1148 }
1149
1150 } // close "Inner GMRES Loop" scope
1151
1152
1153 {
1154 const Index n = Math::min(i, this->_krylov_dim);
1155 {
1156 FEAT::Util::ScopedNvtxRange _nvtx("Solve Hessenberg");
1157 // solve H*q = q
1158 for(Index k(n); k > 0;)
1159 {
1160 --k;
1161 this->_q.at(k) /= this->_h.at(k).at(k);
1162 for(Index j(k); j > 0;)
1163 {
1164 --j;
1165 this->_q.at(j) -= this->_h.at(k).at(j) * this->_q.at(k);
1166 }
1167 }
1168 } // "Solve Hessenberg"
1169
1170 // Runtime::SyncGuard sync_guard;
1171 // update solution
1172 this->_vec_w.format();
1173 // _add_vecs(this->_vec_w, this->_vec_v, this->_q, n, DataType(1));
1174 _add_vecs_s(this->_vec_w.local(), this->_vec_v_elements, this->_q, n, DataType(1));
1175 _internal_flop_counter += n * 2u * _global_vec_size;
1176 }
1177
1178 {
1179 FEAT::Util::ScopedNvtxRange _nvtx("Solve Vanka");
1180 _apply_vanka(this->_vec_v.front(), this->_vec_w);
1181 vec_sol.axpy(this->_vec_v.front());
1182 _internal_flop_counter += _global_vec_size;
1183 }
1184
1185 // compute "real" residual
1186 {
1187 FEAT::Util::ScopedNvtxRange _nvtx("Residual Calc");
1188 matrix.apply(this->_vec_v.at(0), vec_sol, vec_rhs, -DataType(1));
1189 filter.filter_def(this->_vec_v.at(0).local());
1190 _internal_flop_counter += 2u*_global_used_elements;
1191 }
1192
1193 // set the current defect
1194 status = this->_set_new_defectp(this->_vec_v.at(0), vec_sol);
1195 }
1196
1197 // finished
1198 Statistics::add_solver_expression(std::make_shared<ExpressionEndSolve>(this->name(), status, this->get_num_iter()));
1199 return status;
1200 }
