393 template <int block_size_>
394 DenseVectorBlocked<DT_, IT_, block_size_> inflate_to_blocks()
395 {
396 const Index n = this->size();
397 DenseVectorBlocked<DT_, IT_, block_size_> result(this->size());
398
399 auto vb = result.elements_view_w();
400 const auto vx = this->elements_view_r();
401 for(Index i(0); i < n; ++i)
402 vb[i] = vx(i);
403
404 return result;

414 void read_from(FileMode mode, const String& filename)
415 {
416 std::ios_base::openmode bin = std::ifstream::in | std::ifstream::binary;
417 if (mode == FileMode::fm_mtx)
418 bin = std::ifstream::in;
419 std::ifstream file(filename.c_str(), bin);
420 if (! file.is_open())
421 XABORTM("Unable to open Vector file " + filename);
422 read_from(mode, file);
423 file.close();
424 }

432 void read_from(FileMode mode, std::istream & file)
433 {
434 switch (mode)
435 {
436 case FileMode::fm_mtx:
437 {
438 Index rows;
439 String line;
440 std::getline(file, line);
441 if (line.find("%%MatrixMarket matrix array real general") == String::npos)
442 {
443 XABORTM("Input-file is not a compatible mtx-vector-file");
444 }
445 while (! file.eof())
446 {
447 std::getline(file, line);
448 if (file.eof())
449 XABORTM("Input-file is empty");
450
451 String::size_type begin(line.find_first_not_of(" "));
452 if (line.at(begin) != '%')
453 break;
454 }
455 {
456 String::size_type begin(line.find_first_not_of(" "));
457 line.erase(0, begin);
458 String::size_type end(line.find_first_of(" "));
459 String srows(line, 0, end);
460 rows = (Index)atol(srows.c_str());
461 line.erase(0, end);
462
463 begin = line.find_first_not_of(" ");
464 line.erase(0, begin);
465 end = line.find_first_of(" ");
466 String scols(line, 0, end);
467 Index cols((Index)atol(scols.c_str()));
468 line.erase(0, end);
469 XASSERTM(cols == 1, "Input-file is no dense-vector-file");
470 }
471
472 DenseVector<DT_, IT_> tmp(rows);
473 Memory::TypedView<DT_> tmp_view(tmp.elements_view_w());
474 DT_* pval = tmp_view.get_w();
475
476 while (! file.eof())
477 {
478 std::getline(file, line);
479 if (file.eof())
480 break;
481
482 String::size_type begin(line.find_first_not_of(" "));
483 line.erase(0, begin);
484 String::size_type end(line.find_first_of(" "));
485 String sval(line, 0, end);
486 DT_ tval((DT_)atof(sval.c_str()));
487
488 *pval = tval;
489 ++pval;
490 }
491 tmp_view.release();
492 this->move(std::move(tmp));
493 break;
494 }
495 case FileMode::fm_exp:
496 {
497 std::vector<DT_> data;
498
499 while (! file.eof())
500 {
501 std::string line;
502 std::getline(file, line);
503 if (line.find("#", 0) < line.npos)
504 continue;
505 if (file.eof())
506 break;
507
508 std::string n_z_s;
509
510 std::string::size_type first_digit(line.find_first_not_of(" "));
511 line.erase(0, first_digit);
512 std::string::size_type eol(line.length());
513 for (std::string::size_type i(0); i < eol; ++i)
514 {
515 n_z_s.append(1, line[i]);
516 }
517
518 DT_ n_z((DT_)atof(n_z_s.c_str()));
519
520 data.push_back(n_z);
521 }
522
523 DenseVector<DT_, IT_> tmp(data.size());
524 {
525 Memory::TypedView<DT_> tmp_view(tmp.elements_view_w());
526 tmp_view.convert_from(data.data());
527 }
528
529 this->move(std::move(tmp));
530 break;
531 }
532
533 case FileMode::fm_dv:
534 [[fallthrough]];
535
536 case FileMode::fm_binary:
537 this->template _deserialize<double, std::uint64_t>(FileMode::fm_dv, file);
538 break;
539
540 default:
541 XABORTM("Filemode not supported!");
542 }
543 }

551 void write_out(FileMode mode, const String& filename) const
552 {
553 std::ios_base::openmode bin = std::ofstream::out | std::ofstream::binary;
554 if (mode == FileMode::fm_mtx || mode == FileMode::fm_exp)
555 bin = std::ofstream::out;
556 std::ofstream file;
557 char* buff = nullptr;
558 if(mode == FileMode::fm_mtx)
559 {
560 buff = new char[LAFEM::file_out_stream_buffer_size];
561 file.rdbuf()->pubsetbuf(buff, LAFEM::file_out_stream_buffer_size);
562 }
563 file.open(filename.c_str(), bin);
564 if(! file.is_open())
565 XABORTM("Unable to open Matrix file " + filename);
566 write_out(mode, file);
567 file.close();
568 delete[] buff;
569 }

579 switch (mode)

738 {
739 XASSERTM(x.size() == this->size(), "Vector size does not match!");
740

751
752 Statistics::add_flops(this->size() * 2);
753 Statistics::add_time_axpy(ts_stop.elapsed(ts_start));
754 }
