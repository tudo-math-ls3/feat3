248 template<typename MonitorFn_>
249 MeshDeformerQuality deform(Geometry::RootMeshNode<MeshType_>& mesh_node, const MonitorFn_& monitor_fn)
250 {
251 const Real blend_delta = Real(1) / Real(_blend_steps);
252
253 // Prepare for deformation
254 MeshType& mesh = *mesh_node.get_mesh();
255 TrafoType trafo(mesh);
256 SpaceType space(trafo);
257
258 // Determine mesh boundary for handling boundary values
259 Geometry::GlobalMaskedBoundaryFactory<MeshType> boundary_factory(mesh);
260 for(auto& [rank, halo] : mesh_node.get_halo_map())
261 {
262 boundary_factory.add_halo(rank, *halo);
263 }
264 boundary_factory.compile(_comm);
265 Geometry::MeshPart<MeshType> boundary(boundary_factory);
266
267 // Determine surface features we want to maintain
268 Geometry::MeshPart<MeshType> features = Assembly::FeatureMeshPartFactory<MeshType>::make(mesh, boundary);
269
270 // Assemble projection filter to enforce strong boundary conditions
271 ProjectionFilterType projection_filter;
272 if(_keep_boundary_fixed)
273 {
274 const auto& tgs = boundary.template get_target_set<0>();
275 const auto& vertices = mesh.get_vertex_set();
276 for(Index i(0); i < boundary.get_num_entities(0); i++)
277 {
278 projection_filter.add_fixed_dof(tgs[i], vertices[tgs[i]]);
279 }
280 }
281 else
282 {
283 // Normal boundary conditions
284 // Project boundary back onto boundary or onto closest feature
285 Assembly::ProjectionFilterAssembler<MeshType> projection_filter_assembler;
286 projection_filter_assembler.add_mesh_part(boundary);
287 projection_filter_assembler.add_feature_mesh_part(features);
288 projection_filter_assembler.assemble(projection_filter, space, _gate);
289 }
290
291 Assembly::DomainAssembler<TrafoType> domain_assembler(trafo);
292 domain_assembler.compile_all_elements();
293
294 // Iteratively deform
295
296 MeshDeformerQuality current_quality{};
297
298 // Blend loop. Interpolate between current area function and target monitor function.
299 // Loop i blends with factor (i + 1) * blend_delta
300 for(Index i(0); i < _blend_steps; i++)
301 {
302 // Iterative loop
303 // Re-assemble the Poisson problem on the already deformed mesh to get a better solution on the next solve.
304 for(Index j(0); j < _max_iters_per_blend_step; j++)
305 {
306 // Deform the mesh
307 _deform(
308 mesh,
309 trafo,
310 space,
311 domain_assembler,
312 projection_filter,
313 monitor_fn,
314 DataType(i + 1) * blend_delta);
315
316 // Determine quality of deformed mesh
317 current_quality = _quality(mesh, space, domain_assembler, monitor_fn);
318
319 // Exit early upon reaching sufficient quality
320 // Value of 0.05 is taken from the paper
321 if(current_quality.l2norm < 0.05)
322 {
323 break;
324 }
325 }
326 }
327
328 return current_quality;
329 }

336 template<typename MonitorFn_>
337 MeshDeformerQuality quality(Geometry::RootMeshNode<MeshType>& mesh_node, const MonitorFn_& monitor_fn)
338 {
339 // Set up trafo, space, and domain assembler
340 MeshType& mesh = *mesh_node.get_mesh();
341
342 TrafoType trafo(mesh);
343 SpaceType space(trafo);
344
345 Assembly::DomainAssembler<TrafoType> domain_assembler(trafo);
346 domain_assembler.compile_all_elements();
347
348 // Calculate quality
349 return _quality(mesh, space, domain_assembler, monitor_fn);

354 template<typename MonitorFn_>
355 void _deform(
356 MeshType& mesh,
357 const TrafoType& trafo,
358 const SpaceType& space,
359 Assembly::DomainAssembler<TrafoType>& domain_assembler,
360 const ProjectionFilterType& projection_filter,
361 const MonitorFn_& monitor_lambda,
362 Real blend_factor = 1.0)
363 {
364 // Compute area function g and monitor function f
365 VectorType area_fn = _area_function(mesh);
366 VectorType monitor_fn(_gate, space.get_num_dofs());
367 {
368 const auto& vertices = mesh.get_vertex_set();
369 Memory::TypedView<DataType> view = monitor_fn.local().elements_view_w();
370 for(Index i(0); i < space.get_num_dofs(); i++)
371 {
372 view[i] = monitor_lambda(vertices[i]);
373 }
374 }
375
376 // We assume that the monitor_function will produce the same value for the same point on all ranks,
377 // so no synchronization of monitor_fn is necessary.
378
379 // Create inverse functions
380 VectorType inv_area_fn(_gate, space.get_num_dofs());
381 VectorType inv_monitor_fn(_gate, space.get_num_dofs());
382 {
383 if(blend_factor != DataType(1))
384 {
385 scale_to_same_integral_mean(domain_assembler, monitor_fn, area_fn, space);
386
387 // Blend
388 monitor_fn.scale(monitor_fn, blend_factor);
389 monitor_fn.axpy(area_fn, DataType(1) - blend_factor);
390 }
391
392 // Invert
393 inv_monitor_fn.component_invert(monitor_fn);
394 inv_area_fn.component_invert(area_fn);
395
396 scale_to_same_integral_mean(domain_assembler, inv_monitor_fn, inv_area_fn, space);
397 }
398
399 // Compute rhs (inv_monitor_fn - inv_area_fn)
400 VectorType rhs(_gate, space.get_num_dofs());
401 rhs.format();
402 {
403 VectorType tmp = inv_monitor_fn.clone();
404 tmp.axpy(inv_area_fn, DataType(-1));
405
406 Assembly::Common::IdentityOperator op;
407 Assembly::assemble_bilinear_operator_apply_vector_1(
408 domain_assembler,
409 rhs.local(),
410 tmp.local(),
411 op,
412 space,
413 "auto-degree:5");
414 rhs.sync_0();
415 }
416
417 // Solve -laplace w = rhs with integral mean 0
418 VectorType solution(_gate, space.get_num_dofs());
419 {
420 solution.local().format();
421
422 MatrixType matrix(_gate, _gate);
423 Assembly::SymbolicAssembler::assemble_matrix_std1(matrix.local(), space);
424 matrix.local().format();
425
426 Assembly::Common::LaplaceOperator laplace_operator;
427 Assembly::assemble_bilinear_operator_matrix_1(
428 domain_assembler,
429 matrix.local(),
430 laplace_operator,
431 space,
432 "auto-degree:3");
433
434 Assembly::MeanFilterAssembler mean_assembler;
435 FilterType mean_filter;
436 mean_assembler.assemble(mean_filter.local(), space, "auto-degree:3", DataType(0.0));
437
438 mean_filter.local().filter_mat(matrix.local());
439 mean_filter.filter_rhs(rhs);
440 mean_filter.filter_sol(solution);
441
442 auto solver = _solver_factory(matrix, mean_filter);
443
444 solver->init();
445 Solver::solve(*solver, solution, rhs, matrix, mean_filter);
446 solver->done();
447 }
448
449 LocalBlockedVectorType velocity(space.get_num_dofs());
450 velocity.format();
451 Assembly::DiscreteVertexProjector::project_gradient(velocity, solution.local(), space);
452
453 // Strictly enforce boundary conditions by projecting velocity onto the boundary
454 projection_filter.filter_def(velocity);
455
456 LocalInverseMappingType inv_mapping(trafo, 0.1, 1E-2);
457 InverseMappingType global_unmapping(_comm, inv_mapping);
458
459 // Solve ODEs
460 {
461 // Copy vertices into blocked vector so that we can do math on them
462 LocalBlockedVectorType vertices(mesh.get_num_vertices());
463 {
464 const auto& mesh_vertices = mesh.get_vertex_set();
465 Memory::TypedView<VertexType> view = vertices.elements_view_w();
466 for(Index i(0); i < mesh.get_num_vertices(); i++)
467 {
468 view[i] = mesh_vertices[i];
469 }
470 }
471
472 auto moved_vertices = runge_kutta_3(
473 vertices,
474 [&](DataType t, const LocalBlockedVectorType& vs)
475 {
476 LocalBlockedVectorType ode_rhs(mesh.get_num_vertices());
477 Memory::TypedView<VertexType> view_out = ode_rhs.elements_view_w();
478
479 // TODO: Abstract the following into a Global::DiscreteEvaluator class
480
481 // Unmap all points
482 std::vector<Global::GlobalInverseMappingData<DataType, dim>> unmapping =
483 global_unmapping.unmap_points(ViewRange<VertexType>{vs.elements_view_r()});
484
485 // Eval f, g, v
486 struct EvalResult
487 {
488 DataType f;
489 DataType g;
490 VertexType v;
491
492 static Dist::Typemap typemap()
493 {
494 return Dist::Typemap{}
495 .template add_entry<DataType>(offsetof(EvalResult, f))
496 .template add_entry<DataType>(offsetof(EvalResult, g))
497 .template add_entry<VertexType>(offsetof(EvalResult, v));
498 }
499 };
500
501 std::vector<int> ranks;
502 std::vector<Index> indices;
503 std::vector<std::tuple<VertexType, Index, int>> args;
504
505 for(Index i(0); i < unmapping.size(); i++)
506 {
507 if(unmapping[i].size() == 0)
508 {
509 // Unmapping for this point failed.
510 // We call the lambda with the index of the vertex as a fallback
511 ranks.push_back(_comm.rank());
512 indices.push_back(i);
513 args.emplace_back(VertexType(0.0), i, 0);
514 }
515 else
516 {
517 ranks.push_back(unmapping[i].ranks[0]);
518 indices.push_back(i);
519 args.emplace_back(unmapping[i].dom_points[0], unmapping[i].cells[0], 1);
520 }
521 }
522
523 Global::ScalarRemoteLambda<EvalResult(VertexType, Index, int)> remote_eval(_comm);
524 std::vector<EvalResult> evals = remote_eval.call(
525 ranks,
526 indices,
527 args,
528 [&](VertexType dom_point, Index cell, int success)
529 {
530 if(success == 0)
531 {
532 const DataType f = inv_monitor_fn.local().elements_view_r()(cell);
533 const DataType g = inv_area_fn.local().elements_view_r()(cell);
534 const VertexType v = velocity.elements_view_r()(cell);
535
536 return EvalResult{f, g, v};
537 }
538 else
539 {
540 Trafo::InverseMappingData<DataType, dim> inv_map_data;
541 inv_map_data.cells.push_back(cell);
542 inv_map_data.dom_points.push_back(dom_point);
543
544 const DataType f =
545 Assembly::DiscreteEvaluator::eval_fe_function(inv_map_data, inv_monitor_fn.local(), space)
546 .mean_value();
547 const DataType g =
548 Assembly::DiscreteEvaluator::eval_fe_function(inv_map_data, inv_area_fn.local(), space)
549 .mean_value();
550 const VertexType v =
551 Assembly::DiscreteEvaluator::eval_fe_function(inv_map_data, velocity, space).mean_value();
552
553 return EvalResult{f, g, v};
554 }
555 });
556
557 FEAT_PRAGMA_OMP(parallel for)
558 for(Index i = 0; i < mesh.get_num_vertices(); i++)
559 {
560 const DataType f = evals[i].f;
561 const DataType g = evals[i].g;
562 const VertexType v = evals[i].v;
563 const DataType denom = t * f + (DataType(1.0) - t) * g;
564
565 view_out[i] = (DataType(1.0) / denom) * v;
566 }
567
568 return std::move(ode_rhs);
569 });
570
571 // Copy moved vertices back into mesh
572 {
573 auto& mesh_vertices = mesh.get_vertex_set();
574 Memory::TypedView<VertexType> view = moved_vertices.elements_view_r();
575 for(Index i(0); i < mesh.get_num_vertices(); i++)
576 {
577 mesh_vertices[i] = view(i);
578 }
579 }
580 }
581 }

583 template<typename MonitorFn_>
584 MeshDeformerQuality _quality(
585 MeshType& mesh,
586 const SpaceType& space,
587 Assembly::DomainAssembler<TrafoType>& assembler,
588 const MonitorFn_& monitor_fn)
589 {
590 VectorType area = _area_function(mesh);
591 VectorType monitor(_gate, mesh.get_num_vertices());
592 {
593 const auto& vertices = mesh.get_vertex_set();
594
595 Memory::TypedView<DataType> monitor_view = monitor.local().elements_view_w();
596
597 for(Index i(0); i < mesh.get_num_vertices(); i++)
598 {
599 monitor_view[i] = monitor_fn(vertices[i]);
600 }
601 }
602
603 // Scale deformed_area_function
604 scale_to_same_integral_mean(assembler, monitor, area, space);
605
606 area.component_invert(area);
607 VectorType quality_function(_gate, space.get_num_dofs());
608 quality_function.component_product(area, monitor);
609
610 double quality = Math::sqrt(
611 Assembly::integrate_error_function<0>(
612 assembler,
613 Analytic::Common::ConstantFunction<dim>(1.0),
614 quality_function.local(),
615 space,
616 "auto-degree:5")
617 .norm_h0_sqr);
618 double quality_max = Math::abs(quality_function.max_abs_element() - 1.0);
619
620 _comm.allreduce(&quality, &quality, 1, Dist::op_sum);
621 _comm.allreduce(&quality_max, &quality_max, 1, Dist::op_max);
622
623 return {quality, quality_max};
624 }

638 void scale_to_same_integral_mean(
639 Assembly::DomainAssembler<TrafoType>& domain_assembler,
640 const VectorType& a,
641 VectorType& b,
642 const SpaceType& space)
643 {
644 auto int_a = Assembly::integrate_discrete_function<0>(domain_assembler, a.local(), space, "auto-degree:5").value;
645 auto int_b = Assembly::integrate_discrete_function<0>(domain_assembler, b.local(), space, "auto-degree:5").value;
646
647 // Add up integrals on all ranks
648 _comm.allreduce(&int_a, &int_a, 1, Dist::op_sum);
649 _comm.allreduce(&int_b, &int_b, 1, Dist::op_sum);
650
651 // Choose alpha such that alpha * int_b = int_a
652 b.scale(b, int_a / int_b);
653 }
654
656 VectorType _area_function(MeshType_& mesh)
657 {
658 TrafoType trafo(mesh);
659 VectorType area_fn(_gate, mesh.get_num_vertices(), 0);
660
661 // Precompute cell volumes
662 std::vector<DataType> cell_volumes(mesh.get_num_entities(dim));
663 for(Index i(0); i < mesh.get_num_entities(dim); i++)
664 {
665 cell_volumes[i] = trafo.compute_vol(i);
666 }
667
668 // Render cells@vertices
669 const Adjacency::Graph c_at_v(Adjacency::RenderType::transpose, mesh.template get_index_set<dim, 0>());
670
671 // Create frequency vector
672 VectorType freqs(_gate, mesh.get_num_vertices());
673 {
674 Memory::TypedView<DataType> view = freqs.local().elements_view_w();
675 for(Index i(0); i < mesh.get_num_vertices(); i++)
676 {
677 view[i] = DataType(c_at_v.degree(i));
678 }
679 }
680
681 // Sync number of adjacent cells
682 freqs.sync_0();
683
684 {
685 Memory::TypedView<DataType> view = area_fn.local().elements_view_w();
686 Memory::TypedView<DataType> freq_view = freqs.local().elements_view_r();
687 for(Index i(0); i < mesh.get_num_vertices(); i++)
688 {
689 // Compute average cell area
690 DataType area(0.0);
691 for(auto it = c_at_v.image_begin(i); it != c_at_v.image_end(i); ++it)
692 {
693 area += cell_volumes[*it];
694 }
695
696 view[i] = area / freq_view(i);
697 }
698 }
699
700 area_fn.sync_0();
701
702 return area_fn;
703 }

706 template<typename Rhs_>
707 static LocalBlockedVectorType runge_kutta_3(const LocalBlockedVectorType& initial, const Rhs_& rhs)
708 {
709 const Real t_start(0);
710 const Real t_end(1.0);
711 const Real stepsize(0.05);
712
713 LocalBlockedVectorType x = initial.clone();
714
715 // Perform 100 steps of Runge-Kutta 3
716 for(Real t(t_start); t < t_end; t += stepsize)
717 {
718 // std::cerr << "x = " << x << "\n";
719 const LocalBlockedVectorType k1 = rhs(t, x);
720 // std::cerr << "k1 = " << k1 << "\n";
721 LocalBlockedVectorType x_tmp = x.clone();
722 x_tmp.axpy(k1, stepsize / DataType(2));
723 const LocalBlockedVectorType k2 = rhs(t + (stepsize / Real(2)), x_tmp);
724 // std::cerr << "k2 = " << k2 << "\n";
725 x_tmp = x.clone();
726 x_tmp.axpy(k1, -stepsize);
727 x_tmp.axpy(k2, DataType(2) * stepsize);
728 const LocalBlockedVectorType k3 = rhs(t + stepsize, x_tmp);
729 // std::cerr << "k3 = " << k3 << "\n";
730
731 x.axpy(k1, stepsize / DataType(6));
732 x.axpy(k2, stepsize / DataType(6) * 4);
733 x.axpy(k3, stepsize / DataType(6));
734 // std::cerr << "x = " << x << "\n";
735 }
736
737 return x;
738 }
