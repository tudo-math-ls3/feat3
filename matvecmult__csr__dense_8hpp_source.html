98 template <typename DT_, typename IT_>
99 static void exec(Memory::Arbiter& r, const DT_ alpha, const Memory::Arbiter& x, const Memory::Arbiter& y, const Memory::Arbiter& a, const Memory::Arbiter& col_idx, const Memory::Arbiter& row_ptr, const Index rows, const Index cols, const Index nonzeros, const bool transposed)
100 {
101 XASSERT(r != x);
102
103 switch(Backend::get_preferred_backend())
104 {
105 case PreferredBackend::generic:
106 exec_generic<DT_, IT_>(r, alpha, x, y, a, col_idx, row_ptr, rows, cols, nonzeros, transposed);
107 break;
108
109 case PreferredBackend::mkl:
110#ifdef FEAT_HAVE_MKL
111 if constexpr(sizeof(IT_) == sizeof(Index))
112 exec_mkl<DT_>(r, alpha, x, y, a, col_idx, row_ptr, rows, cols, nonzeros, transposed);
113 else
114 {
115 Backend::warn_missing("LAFEM::Arch::MatVecMultCSRDense::exec: MKL backend not available for IT_ != Index");
116 exec_generic<DT_, IT_>(r, alpha, x, y, a, col_idx, row_ptr, rows, cols, nonzeros, transposed);
117 }
118#else
119 XABORTM("LAFEM::Arch::MatVecMultCSRDense::exec: MKL backend not available!");
120#endif
121 break;
122
123 case PreferredBackend::cuda:
124#ifdef FEAT_HAVE_CUDA
125 exec_cuda<DT_, IT_>(r, alpha, x, y, a, col_idx, row_ptr, rows, cols, nonzeros, transposed);
126#else
127 XABORTM("LAFEM::Arch::MatVecMultCSRDense::exec: CUDA backend not available!");
128#endif
129 break;
130
131 default:
132 XABORTM("LAFEM::Arch::MatVecMultCSRDense::exec: unknown backend!");
133 break;
134 }
135 }
