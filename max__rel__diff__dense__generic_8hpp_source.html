17 DT_ MaxRelDiffDense::exec_generic_impl(const DT_ * const x, const DT_ * const y, const Index size)
18 {
19 DT_ r(0);
20 bool has_nan = false;
21 bool has_inf = false;
22 for (Index i(0) ; i < size ; ++i)
23 {
24 has_nan = has_nan || Math::isnan(x[i]) || Math::isnan(y[i]);
25 has_inf = has_inf || Math::isinf(x[i]) || Math::isinf(y[i]);
26 r = Math::max(r, Math::abs(x[i] - y[i]) / (Math::abs(x[i]) + Math::abs(y[i]) + DT_(1)));
27 }
28
29 // if we found a NaN or an infinity, we'll return a NaN here
30 return (has_nan || has_inf ? Math::nan<DT_>() : r);
31 }
