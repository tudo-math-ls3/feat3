111 explicit SparseVectorBlocked(Index size_in, DenseVectorBlocked<DT_, IT_, block_size_> & elements_in,
112 DenseVector<IT_, IT_> & indices_in)
113 {
114 XASSERT(size_in != Index(0));
115 XASSERTM(indices_in.size() == elements_in.size(), "Vector size mismatch!");

118 this->_scalar_index.push_back(elements_in.size());

120 this->_elements.push_back(elements_in.elements_arbiter().attach());
121 this->_elements_size.push_back(elements_in.size() * Index(block_size_));

424 {
425 std::ios_base::openmode bin = std::ofstream::out | std::ofstream::binary;
426 if(mode == FileMode::fm_mtx)

428 std::ofstream file;

430 if(mode == FileMode::fm_mtx)
431 {

433 file.rdbuf()->pubsetbuf(buff, LAFEM::file_out_stream_buffer_size);
434 }
435 file.open(filename.c_str(), bin);

449 void write_out(FileMode mode, std::ostream& file) const
450 {
451 switch(mode)
452 {
453 case FileMode::fm_binary:
454 [[fallthrough]];
455
456 case FileMode::fm_svb:
457 this->template _serialize<double, std::uint64_t>(FileMode::fm_svb, file);
458 break;
459
460 case FileMode::fm_mtx:
461 {
462
463 file << "%%MatrixMarket matrix coordinate real general\n";
464 file << this->size()*block_size << " " << 1 << " " << this->num_nzes()*block_size << "\n";
465 file << std::scientific << std::setprecision(Type::Traits<DT_>::format_precision);
466
467 const Index u_elem(this->num_nzes());
468 const Memory::TypedView<IT_> idx_view = this->indices_view_r();
469 const Memory::TypedView<ValueType> val_view = this->elements_view_r();
470 for (Index i(0) ; i < u_elem ; ++i)
471 {
472 for(int k(0); k < block_size; ++k)
473 file << (idx_view[i]*Index(block_size+k+1)) << " " << 1 << " " << val_view[i][k] << "\n";
474 }
475 break;
476 }
477
478 default:
479 XABORTM("Filemode not supported!");
480 }

586 void permute(Adjacency::Permutation & perm)
587 {
588 if (perm.size() == 0)
589 return;
590
591 std::map<IT_, ValueType> xm;
592
593 auto ip = perm.inverse();
594 const Index * const inv_pos(ip.get_perm_pos());
595
596 Memory::TypedView<ValueType> xv(this->elements_view_rw());
597 Memory::TypedView<IT_> xi(this->indices_view_rw());
598 for (Index i(0) ; i < this->num_nzes() ; ++i)
599 {
600 xm.emplace(IT_(inv_pos[xi[i]]), xv[i]);
601 }
602
603 Index k(0);
604 for(const auto& x : xm)
605 {
606 xi[k] = x.first;
607 xv[k] = x.second;
608 ++k;
609 }
610 }

631 bool same_layout(const SparseVectorBlocked& other) const
632 {
633 if(this->size() != other.size())
634 return false;
635 if(this->num_nzes() != other.num_nzes())
636 return false;
637
638 if(this->num_nzes() == Index(0))
639 return true;
640
641 // check if the indices arbiters are the same
642 if(this->indices_arbiter() == other.indices_arbiter())
643 return true;
644
645 const Memory::TypedView<IT_> idx_a = this->indices_view_r();
646 const Memory::TypedView<IT_> idx_b = other.indices_view_r();
647
648 // check all array entries
649 const Index n = this->num_nzes();
650 for(Index i(0); i < n; ++i)
651 {
652 if(idx_a[i] != idx_b[i])
653 return false;
654 }
655
656 // okay, arrays are identical
657 return true;
658 }

666 friend std::ostream & operator<< (std::ostream & lhs, const SparseVectorBlocked & b)
667 {
668 const Memory::TypedView<IT_> idx_view(b.indices_view_r());
669 const Memory::TypedView<ValueType> val_view(b.elements_view_r());
670 const Index n = b.size();
671 const Index nze = b.num_nzes();
672 lhs << "[";
673 Index k = 0;
674 for (Index i(0) ; i < nze ; ++i)
675 {
676 for(; k < idx_view[i]; ++k)
677 {
678 for(int d = 0; d < block_size_; ++d)
679 lhs << " " << DT_(0);
680 }
681
682 for(int d = 0; d < block_size_; ++d)
683 lhs << " " << val_view[i][d];
684 ++k;
685 }
686 for(; k < n; ++k)
687 for(int d = 0; d < block_size_; ++d)
688 lhs << " " << DT_(0);
689 lhs << "]";
690
691 return lhs;
692 }
