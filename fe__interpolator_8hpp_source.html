58 struct Lagrange1To2DofAtEntity

84 template<typename DataType_, typename ValueType_>
85 static void recurse(Memory::TypedView<ValueType_>& vtc, const Memory::TypedView<ValueType_>& vfc, const ToSpace& to_space)
86 {
87 const auto& mesh = to_space.get_trafo().get_mesh();
88
89 // We use this to identify the vertex DoF at our shape
90 const auto& from_vertex_at_shape_idx = mesh.template get_index_set<shape_dim_, 0>();
91
92 // This is for mapping to the correct Lagrange2 DoF
93 typename ToSpace::template DofAssignment<shape_dim_, DataType_>::Type to_dof_assignment(to_space);
94
95 for(Index k(0); k < mesh.get_num_entities(shape_dim_); ++k)
96 {
97 to_dof_assignment.prepare(k);
98
99 // There is only one Lagrange2 DoF at each entity, but this is easier to understand for future reference
100 for(int j(0); j < to_dof_assignment.get_num_assigned_dofs(); ++j)
101 {
102 ValueType_ tmp(0);
103
104 for(int i(0); i < from_vertex_at_shape_idx.get_num_indices(); ++i)
105 {
106 tmp += vfc(from_vertex_at_shape_idx(k,i));
107 }
108
109 // Linear interpolation
110 tmp *= ( DataType_(1)/DataType_(from_vertex_at_shape_idx.get_num_indices()) );
111
112 // Emplacement operator
113 vtc[to_dof_assignment.get_index(j)] = tmp;
114 }
115 }
116
117 // recurse down
118 Lagrange1To2DofAtEntity<Trafo_, shape_dim_-1>::template recurse<DataType_>( vtc, vfc, to_space);
119 }
120 };
