237 template<typename LVT2_, typename MT2_>
238 void convert(const Gate<LVT2_, MT2_>& other)
239 {
240 if((void*)this == (void*)&other)
241 return;
242
243 this->_comm = other._comm;
244 this->_ranks = other._ranks;
245
246 const std::size_t size = other._mirrors.size();
247
248 this->_mirrors.resize(size);
249 this->_send_buffers.clear();
250 this->_send_buffers.reserve(size);
251 this->_recv_buffers.clear();
252 this->_recv_buffers.reserve(size);
253 for(std::size_t i(0); i < size; ++i)
254 {
255 this->_mirrors.at(i).convert(other._mirrors.at(i));
256 this->_send_buffers.emplace_back(_mirrors.at(i).create_buffer(LocalVector_()));
257 this->_recv_buffers.emplace_back(_mirrors.at(i).create_buffer(LocalVector_()));
258 }
259 this->_send_reqs.clear();
260 this->_recv_reqs.clear();
261 this->_send_reqs.reserve(size);
262 this->_recv_reqs.reserve(size);
263 this->_send_views.resize(size);
264 this->_recv_views.resize(size);
265
266 this->_freqs.convert(other._freqs);
267 }

288 template<typename LVT2_>
289 void convert(const Gate<LVT2_, Mirror_>& other, LocalVector_&& vector, LAFEM::CloneMode mode = LAFEM::CloneMode::Shallow)
290 {
291 if((void*)this == (void*)&other)
292 return;
293
294 const std::size_t size = other._mirrors.size();
295 this->_ranks.clear();
296 this->_mirrors.clear();
297 this->_send_buffers.clear(); // will be rebuild by compile function
298 this->_recv_buffers.clear(); // will be rebuild by compile function
299 this->_mirrors.resize(size);
300 this->_send_buffers.reserve(size);
301 this->_recv_buffers.reserve(size);
302 this->_freqs.clear(); // will be rebuild by compile function
303
304 this->_comm = other._comm;
305 this->_ranks = other._ranks;
306
307 // shallow-clone mirrors
308 for(std::size_t i(0); i < _mirrors.size(); ++i)
309 {
310 _mirrors.at(i).clone(other._mirrors.at(i), mode);
311 }
312
313 // compile this gate
314 compile(std::forward<LocalVector_>(vector));
315 }

370 void compile(LocalVector_&& vector)
371 {
372 const std::size_t size = _mirrors.size();
373 // initialize frequency vector
374 _freqs = std::move(vector);
375 _freqs.format(DataType(1));
376
377 _send_buffers.clear();
378 _send_buffers.reserve(size);
379 _recv_buffers.clear();
380 _recv_buffers.reserve(size);
381
382 // loop over all mirrors and create the buffers
383 for(std::size_t i(0); i < size; ++i)
384 {
385 _send_buffers.emplace_back(_mirrors.at(i).create_buffer(_freqs));
386 _recv_buffers.emplace_back(_mirrors.at(i).create_buffer(_freqs));
387 }
388
389 this->_send_reqs.clear();
390 this->_recv_reqs.clear();
391 this->_send_reqs.reserve(size);
392 this->_recv_reqs.reserve(size);
393 this->_send_views.resize(size);
394 this->_recv_views.resize(size);
395
396 // loop over all mirrors
397 for(std::size_t i(0); i < _mirrors.size(); ++i)
398 {
399 // skip empty mirrors
400 if(_mirrors.at(i).hollow())
401 continue;
402
403 // sum up number of ranks per frequency entry, listed in different mirrors
404 auto& temp = _send_buffers.at(i);
405 temp.format(DataType(1));
406
407 // gather-axpy into frequency vector
408 _mirrors.at(i).scatter_axpy(_freqs, temp);
409 }
410
411 // invert frequencies
412 _freqs.component_invert(_freqs);
413 }

443 Index get_num_global_dofs() const
444 {
445 XASSERT(this->_comm != nullptr);
446 if(this->_comm->size() <= 1)
447 return this->get_num_local_dofs();
448
449 // get my rank
450 const int my_rank = this->_comm->rank();
451
452 // get local number of DOFs
453 const Index loc_dofs = this->get_num_local_dofs();
454
455 // create a local mask vector and format it to 1
456 std::vector<int> mask(std::size_t(loc_dofs), 1);
457
458 // set all DOFs, which are shared with a lower rank neighbor, to 0
459 for(std::size_t i(0); i < _mirrors.size(); ++i)
460 {
461 if(this->_ranks.at(i) < my_rank)
462 this->_mirrors.at(i).mask_scatter(this->_freqs, mask, 0);
463 }
464
465 // count the number of DOFs that are still 1 and thus not owned by a lower rank neighbor
466 Index owned_dofs(0u);
467 for(const auto& k : mask)
468 owned_dofs += Index(k);
469
470 // now sum up the number of owned DOFs over all processes
471 Index global_dofs(0u);
472 this->_comm->allreduce(&owned_dofs, &global_dofs, std::size_t(1), Dist::op_sum);
473
474 // done!
475 return global_dofs;
476 }

840 void _sync_vec(LocalVector_& vector) const
841 {
842 if(_ranks.empty())
843 {
844 return;
845 }
846
847 const PreferredBackend sync_backend = Backend::get_preferred_backend();
848 const bool flip_sync_order = sync_backend == PreferredBackend::cuda;
849
850 const std::size_t n = this->_mirrors.size();
851 // post receives
852 for(std::size_t i(0); i < n; ++i)
853 {
854 // skip hollow mirrors
855 if(this->_mirrors.at(i).hollow())
856 continue;
857
858 // post receive
859 _recv_views.at(i) = _recv_buffers.at(i).elements_view_w(Memory::to_location(sync_backend));
860 _recv_reqs.push_back(_comm->irecv(_recv_views.at(i).get_w(), _recv_buffers.at(i).size(), _ranks.at(i)));
861 }
862
863 if(flip_sync_order)
864 {
865 // post sends
866 for(std::size_t i(0); i < n; ++i)
867 {
868 // skip hollow mirrors
869 if(_mirrors.at(i).hollow())
870 continue;
871
872 // gather from mirror
873 _mirrors.at(i).gather(_send_buffers.at(i), vector);
874 }
875
876 Runtime::synchronize_devices();
877 }
878 // post sends
879 for(std::size_t i(0); i < n; ++i)
880 {
881 // skip hollow mirrors
882 if(_mirrors.at(i).hollow())
883 continue;
884
885 // gather from mirror
886 if(!flip_sync_order)
887 _mirrors.at(i).gather(_send_buffers.at(i), vector);
888
889 // post send
890 _send_views.at(i) = _send_buffers.at(i).elements_view_r(Memory::to_location(sync_backend));
891 _send_reqs.push_back(_comm->isend(_send_views.at(i).get_r(), _send_buffers.at(i).size(), _ranks.at(i)));
892 }
893
894 // process all pending receives
895 for(std::size_t idx(0u); _recv_reqs.wait_any(idx); )
896 {
897 // scatter the receive buffer
898 _recv_views.at(idx).release();
899 _mirrors.at(idx).scatter_axpy(vector, _recv_buffers.at(idx));
900 }
901
902 _send_reqs.wait_all();
903 // release send views
904 for(auto& v : _send_views)
905 {
906 v.release();
907 }
908 _send_reqs.clear();
909 _recv_reqs.clear();
910 }
