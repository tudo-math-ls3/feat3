68 template<std::size_t i = 0, typename... Ts_>
69 void pack_tuple(
70 const Dist::Comm& comm,
71 const std::tuple<Ts_...>& tuple,
72 void* outbuf,
73 std::size_t outsize,
74 int* position)
75 {
76 using T = std::decay_t<std::remove_reference_t<decltype(std::get<i>(tuple))>>;
77
78 const T& t = std::get<i>(tuple);
79 comm.pack<T>(&t, 1, outbuf, outsize, position);
80
81 if constexpr(i + 1 != sizeof...(Ts_))
82 {
83 pack_tuple<i + 1>(comm, tuple, outbuf, outsize, position);
84 }
85 }

231 template<typename RankRange_, typename IndexRange_, typename ArgsRange_, typename Lambda_>
232 std::vector<ReturnType>
233 call(const RankRange_& ranks, const IndexRange_& indices, const ArgsRange_& args, Lambda_ lambda)
234 {
235 XASSERTM(
236 std::distance(ranks.begin(), ranks.end()) == std::distance(indices.begin(), indices.end()),
237 "size mismatch between ranks and indices");
238
239 const auto num_ranks = (std::size_t)_comm.size();
240 const auto num_calls = (std::size_t)std::distance(ranks.begin(), ranks.end());
241 const auto pack_size = (std::size_t)Intern::tuple_pack_size<Args_...>(_comm);
242
244 // Phase 1 - Exchange number of messages
246
247 std::vector<int> receivecounts(num_ranks, 0);
248 {
249 std::vector<int> sendcounts(num_ranks, 0);
250 for(auto i : ranks)
251 {
252 sendcounts[(std::size_t)i] += 1;
253 }
254 _comm.alltoall(sendcounts.data(), 1, receivecounts.data(), 1);
255 }
256
258 // Phase 2 - Send args and receive results
260
261 Dist::RequestVector arg_sends;
262 Dist::RequestVector result_receives;
263 std::vector<char> arg_sendbuffer(num_calls * pack_size, 0);
264 std::vector<ReturnType> return_receivebuffer(num_calls);
265 {
266 auto rank_iter = ranks.begin();
267 auto index_iter = indices.begin();
268
269 std::size_t send_pos(0);
270 std::size_t receive_pos(0);
271 while(rank_iter != ranks.end())
272 {
273 const Index idx = *index_iter++;
274 const int rank = *rank_iter++;
275 const ArgumentTuple& arg = Intern::nth_of_range(args, idx);
276
277 // NOTE(mmuegge): We need to tag arguments and return types differently,
278 // because MPI will match arg-sends and return-receives if the return type
279 // and the packed tuple happen to have the same size.
280
281 // Post receive for final result
282 result_receives.push_back(_comm.irecv(&return_receivebuffer[receive_pos], 1, rank, tag_return));
283
284 // Pack tuple for sending arg
285 int pos = (int)send_pos;
286 Intern::pack_tuple(_comm, arg, arg_sendbuffer.data(), num_calls * pack_size, &pos);
287
288 // Post send for sending argument tuple
289 arg_sends.push_back(_comm.isend(&arg_sendbuffer[send_pos], pack_size, Dist::dt_packed, rank, tag_argument));
290
291 // Increment buffer positions
292 receive_pos++;
293 send_pos += pack_size;
294 }
295 }
296
298 // Phase 3 - Receive args and send results
300
301 const auto num_args = (std::size_t)std::accumulate(receivecounts.begin(), receivecounts.end(), 0);
302 Dist::RequestVector arg_receives;
303 std::vector<char> arg_receivebuffer(num_args * pack_size, 0);
304 {
305 std::size_t pos(0);
306 for(std::size_t rank(0); rank < num_ranks; rank++)
307 {
308 const int receive_count = receivecounts[rank];
309
310 for(int i(0); i < receive_count; i++)
311 {
312 arg_receives.push_back(
313 _comm.irecv(&arg_receivebuffer[pos], pack_size, Dist::dt_packed, (int)rank, tag_argument));
314 pos += pack_size;
315 }
316 }
317 }
318
319 Dist::RequestVector result_sends;
320 std::vector<ReturnType> return_sendbuffer(num_args);
321 {
322 std::size_t idx(0);
323 Dist::Status status;
324
325 while(arg_receives.wait_any(idx, status))
326 {
327 ArgumentTuple t;
328 int arg_pos = (int)(idx * pack_size);
329 Intern::unpack_tuple(_comm, t, arg_receivebuffer.data(), num_args * pack_size, &arg_pos);
330
331 return_sendbuffer[idx] = std::apply(lambda, t);
332 result_sends.push_back(_comm.isend(&return_sendbuffer[idx], 1, status.source(), tag_return));
333 }
334 }
335
336 arg_sends.wait_all();
337 arg_receives.wait_all();
338 result_sends.wait_all();
339 result_receives.wait_all();
340
341 return return_receivebuffer;
342 }
343 };

437 template<typename RankRange_, typename IndexRange_, typename ArgsRange_, typename Lambda_>
438 std::vector<ReturnType>
439 call(const RankRange_& ranks, const IndexRange_& indices, const ArgsRange_& args, Lambda_ lambda)
440 {
441 // Inner type of ranges returned by lambda
442 using ValueType = std::decay_t<std::remove_reference_t<decltype(*std::declval<ReturnType>().begin())>>;
443
444 XASSERTM(
445 std::distance(ranks.begin(), ranks.end()) == std::distance(indices.begin(), indices.end()),
446 "size mismatch between ranks and indices");
447
448 const auto num_ranks = (std::size_t)_comm.size();
449 const auto num_calls = (std::size_t)std::distance(ranks.begin(), ranks.end());
450 const auto pack_size = (std::size_t)Intern::tuple_pack_size<Args_...>(_comm);
451
453 // Phase 1 - Exchange number of calls
455
456 // Each rank needs to know how many arguments it is to receive from each rank
457 // so that it can post the necessary sends and receives.
458
459 std::vector<int> receivecounts(num_ranks, 0);
460 {
461 std::vector<int> sendcounts(num_ranks, 0);
462 for(auto i : ranks)
463 {
464 sendcounts[(std::size_t)i] += 1;
465 }
466 _comm.alltoall(sendcounts.data(), 1, receivecounts.data(), 1);
467 }
468
470 // Phase 2 - Send args and receive sizes of ranges
472
473 // Sends and receives are posted in the same order as given by the user
474 Dist::RequestVector size_receives;
475 Dist::RequestVector arg_sends;
476 std::vector<std::size_t> range_message_sizes(num_calls, 0);
477 std::vector<char> arg_sendbuffer(num_calls * pack_size, 0);
478 {
479 auto rank_iter = ranks.begin();
480 auto index_iter = indices.begin();
481
482 std::size_t send_pos(0);
483 std::size_t receive_pos(0);
484 while(rank_iter != ranks.end())
485 {
486 const Index idx = *index_iter++;
487 const int rank = *rank_iter++;
488 const ArgumentTuple& arg = Intern::nth_of_range(args, idx);
489
490 // Pack tuple for sending arg
491 int pos = (int)send_pos;
492 Intern::pack_tuple(_comm, arg, arg_sendbuffer.data(), num_calls * pack_size, &pos);
493
494 // Post send for sending argument tuple
495 arg_sends.push_back(_comm.isend(&arg_sendbuffer[send_pos], pack_size, Dist::dt_packed, rank, tag_argument));
496
497 // Post receive for size of resulting range
498 size_receives.push_back(_comm.irecv(&range_message_sizes[receive_pos], 1, rank, tag_size));
499
500 // Increment buffer positions
501 send_pos += pack_size;
502 receive_pos++;
503 }
504 }
505
507 // Phase 3 - Receive args and send range sizes and ranges
509
510 const auto num_args = (std::size_t)std::accumulate(receivecounts.begin(), receivecounts.end(), 0);
511 Dist::RequestVector arg_receives;
512 std::vector<char> arg_receivebuffer(num_args * pack_size, 0);
513 {
514 std::size_t pos(0);
515 for(std::size_t rank(0); rank < num_ranks; rank++)
516 {
517 const int receive_count = receivecounts[rank];
518
519 for(int i(0); i < receive_count; i++)
520 {
521 // Post receive for argument
522 arg_receives.push_back(
523 _comm.irecv(&arg_receivebuffer[pos], pack_size, Dist::dt_packed, (int)rank, tag_argument));
524 pos += pack_size;
525 }
526 }
527 }
528
529 Dist::RequestVector size_sends;
530 Dist::RequestVector range_sends;
531 std::vector<ReturnType> ranges(num_args);
532 std::vector<std::size_t> range_sizes(num_args, 0);
533 std::vector<std::size_t> range_buffer_sizes(num_ranks, 0);
534 {
535 std::size_t idx(0);
536 Dist::Status status;
537
538 while(arg_receives.wait_any(idx, status))
539 {
540 int arg_pos = (int)(idx * pack_size);
541
542 ArgumentTuple t;
543 Intern::unpack_tuple(_comm, t, arg_receivebuffer.data(), num_args * pack_size, &arg_pos);
544
545 ranges[idx] = std::apply(lambda, t);
546 range_sizes[idx] = (std::size_t)std::distance(ranges[idx].begin(), ranges[idx].end());
547 range_buffer_sizes[std::size_t(status.source())] += range_sizes[idx];
548
549 // Post send for sizes
550 size_sends.push_back(_comm.isend(&range_sizes[idx], 1, status.source(), tag_size));
551
552 // Pointer to beginning of range, or nullptr if range is empty
553 // NOTE: This assumes the range is contiguous in memory
554 const void* range_pointer = range_sizes[idx] > 0 ? &*ranges[idx].begin() : nullptr;
555
556 // Post send for range
557 range_sends.push_back(
558 _comm.isend(range_pointer, range_sizes[idx], Dist::autotype<ValueType>(), status.source(), tag_return));
559 }
560 }
561
563 // Phase 4 - Exchange buffer sizes for all ranks
565
566 // At this point each ranks knows how many values it will send to each other rank.
567 // Exchange these values so that reach ranks can allocate its range_buffer before receiving individual ranges.
568
569 // TODO: Is there a smarter way than doing _comm.size() reduces?
570 std::size_t range_buffer_size(0);
571 for(std::size_t rank(0); rank < num_ranks; rank++)
572 {
573 _comm.reduce(&range_buffer_sizes[rank], &range_buffer_size, 1, Dist::op_sum, (int)rank);
574 }
575
577 // Phase 5 - Receive range sizes and ranges
579
580 size_sends.wait_all();
581
582 Dist::RequestVector range_receives;
583 std::vector<ValueType> range_buffer(range_buffer_size);
584 std::vector<std::size_t> range_buffer_offsets;
585 std::exclusive_scan(
586 range_message_sizes.begin(),
587 range_message_sizes.end(),
588 std::back_inserter(range_buffer_offsets),
589 std::size_t(0));
590 {
591 std::size_t idx(0);
592 Dist::Status status;
593
594 // For each message containing a range size, post the corresponding range receive
595 while(size_receives.wait_any(idx, status))
596 {
597 std::size_t offset = range_buffer_offsets[idx];
598 std::size_t size = range_message_sizes[idx];
599 void* range_pointer = size > 0 ? &range_buffer[offset] : nullptr;
600 range_receives.push_back(
601 _comm.irecv(range_pointer, size, Dist::autotype<ValueType>(), status.source(), tag_return));
602 }
603 }
604
606 // Phase 6 - Reconstruct ranges
608
609 arg_sends.wait_all();
610 arg_receives.wait_all();
611 range_receives.wait_all();
612 range_sends.wait_all();
613
614 std::vector<ReturnType> result;
615
616 // NOTE: We are keeping the same message order throughout all of the above code,
617 // so we can just walk linearly over the range_buffer and the reconstructed ranges
618 // will have the same order as the input parameters.
619 std::size_t pos(0);
620 for(std::size_t size : range_message_sizes)
621 {
622 auto begin = range_buffer.data() + pos;
623 auto end = begin + size;
624 result.emplace_back(begin, end);
625 pos += size;
626 }
627
628 return result;
629 }
630 };
