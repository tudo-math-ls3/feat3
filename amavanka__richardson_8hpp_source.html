218 virtual Status _apply_intern(VectorType& vec_sol, const VectorType& vec_rhs)
219 {
220 Statistics::add_solver_expression(std::make_shared<ExpressionStartSolve>(this->name()));
221
222 VectorType& vec_def(this->_vec_def);
223 VectorType& vec_cor(this->_vec_cor);
224 const auto& vanka_mat = this->_vanka->get_vanka_matrix();
225
226 const Index steps = Math::max(this->_max_iter, Index(1));
227 for(Index step(0); step < steps; ++step)
228 {
229 // d = rhs - A*sol (global apply includes the type-0 synchronization)
230 this->_system_matrix.apply(vec_def, vec_sol, vec_rhs, -DataType(1));
231 this->_system_filter.filter_def(vec_def);
232
233 // cor = V*d (local AmaVanka apply) + Schwarz sync_1 + correction filter
234 vanka_mat.apply(vec_cor.local(), vec_def.local());
235 vec_cor.sync_1();
236 this->_system_filter.filter_cor(vec_cor);
237
238 // u += omega * cor
239 vec_sol.axpy(vec_cor, this->_omega);
240 }
241
242 this->_num_iter = steps;
243 this->_def_cur = this->_def_init = DataType(0);
244 Statistics::add_solver_expression(std::make_shared<ExpressionEndSolve>(this->name(), Status::success, this->get_num_iter()));
245 return Status::success;
246 }

250 template<int bh_, int bw_>
251 static void _fill_block(RichardsonHelper::MetaBlock<DataType, IndexType>& mb,
252 const LAFEM::SparseMatrixBCSR<DataType, IndexType, bh_, bw_>& m)
253 {
254 if(m.num_nzes() == Index(0))
255 return; // leave the block empty
256 // ensure this block owns a persistent cuSPARSE SpMV workspace (built once, reused across
257 // applies; required by the cuSPARSE BSR SpMV path on CUDA >= 13) and hand it to the driver
258 if(!m.get_workspace(PreferredBackend::cuda))
259 m.init_workspace();
260 mb.workspace = m.get_workspace(PreferredBackend::cuda).get();
261 mb.val = &m.val_arbiter();
262 mb.row = &m.row_ptr_arbiter();
263 mb.col = &m.col_idx_arbiter();
264 mb.block_height = bh_;
265 mb.block_width = bw_;
266 mb.num_rows = m.num_rows();
267 mb.num_cols = m.num_cols();
268 mb.num_nnz = m.num_nzes();
269 }

286 const LAFEM::TupleMatrix<Row0_, Row1_>& mat)

296 template<int C_, typename LocGate_>
297 void _fill_components(RichardsonHelper::RichardsonApplyInfo<DataType, IndexType, num_comp>& info,
298 VectorType& vec_sol, const VectorType& vec_rhs, const LocGate_* gate)
299 {
300 if constexpr(C_ < num_comp)
301 {
302 auto& sc = vec_sol.local().template at<C_>();
303 const auto& rc = vec_rhs.local().template at<C_>();
304 auto& dc = this->_vec_def.local().template at<C_>();
305 auto& cc = this->_vec_cor.local().template at<C_>();
306 constexpr int bs = Intern::ricci_dim_vector<std::decay_t<decltype(sc)>>;
307
308 info.components[C_] = RichardsonHelper::MetaVectorInfo{bs, sc.size()};
309 info.vec_sol[C_] = &sc.elements_arbiter();
310 info.vec_rhs[C_] = &rc.elements_arbiter();
311 info.vec_def[C_] = &dc.elements_arbiter();
312 info.vec_cor[C_] = &cc.elements_arbiter();
313
314 // gate for this component (empty if no row gate)
315 auto& gw = info.gates[C_];
316 gw.comm = const_cast<Dist::Comm*>(this->_comm);
317 if(gate != nullptr)
318 {
319 const auto& ranks = gate->get_ranks();
320 const auto& mirrors = gate->get_mirrors();
321 gw.ranks.assign(ranks.begin(), ranks.end());
322 gw.freqs = &gate->get_freqs().template at<C_>().elements_arbiter();
323 for(std::size_t k(0); k < ranks.size(); ++k)
324 {
325 const auto& sub = mirrors[k].template at<C_>();
326 gw.mirror_idx.push_back(&sub.indices_arbiter());
327 gw.mirror_size.push_back(sub.num_indices());
328 }
329 }
330 _fill_components<C_ + 1>(info, vec_sol, vec_rhs, gate);
331 }
332 }

334 virtual Status _cuda_apply_intern(VectorType& vec_sol, const VectorType& vec_rhs)
335 {
336 Statistics::add_solver_expression(std::make_shared<ExpressionStartSolve>(this->name()));
337
338 RichardsonHelper::RichardsonApplyInfo<DataType, IndexType, num_comp> info;
339 _extract_matrix(info.mat_sys, this->_system_matrix.local());
340 _extract_matrix(info.mat_vanka, this->_vanka->get_vanka_matrix());
341
342 // per-component unit-defect filters
343 _extract_filters<0>(info);
344
345 // per-component vectors + gates
346 const auto* gate = this->_system_matrix.get_row_gate();
347 _fill_components<0>(info, vec_sol, vec_rhs, gate);
348
349 info.omega = this->_omega;
350 info.min_iter = this->_min_iter;
351 info.max_iter = Math::max(this->_max_iter, Index(1));
352 info.always_use_parallel_kernel = this->_always_use_parallel_kernel;
353
354 RichardsonHelper::RichardsonStatus<DataType> stat;
355 if((this->_comm == nullptr || this->_comm->size() == 1) && !this->_always_use_parallel_kernel)
356 this->_status = VankaRichardson::cuda_apply_amavanka_richardson_seq(info, stat);
357 else
358 this->_status = VankaRichardson::cuda_apply_amavanka_richardson_parallel(info, stat);
359
360 this->_num_iter = stat.num_iter;
361 this->_def_cur = stat.def_norm;
362 this->_def_init = stat.def_init;
363 Statistics::add_solver_expression(std::make_shared<ExpressionEndSolve>(this->name(), this->_status, stat.num_iter));
364 return this->_status;
365 }

368 template<int C_>
369 void _extract_filters(RichardsonHelper::RichardsonApplyInfo<DataType, IndexType, num_comp>& info)
370 {
371 if constexpr(C_ < num_comp)
372 {
373 auto& vc = this->_vec_def.local().template at<C_>();
374 std::map<IndexType, DataType> fmap;
375 Intern::FilterHelper<std::decay_t<decltype(this->_system_filter.local().template at<C_>())>>::add_filter(
376 fmap, this->_system_filter.local().template at<C_>(), vc);
377 // stash the flattened unit filter so its arbiters stay alive for the whole apply
378 auto& uf = _unit_filters[C_];
379 uf = LAFEM::UnitFilter<DataType, IndexType>(vc.size_raw(), fmap);
380 info.def_filters[C_] = RichardsonHelper::UnitFilterInfo<DataType, IndexType>{
381 &uf.elements_arbiter(), &uf.indices_arbiter(), uf.num_nzes_raw()};
382 _extract_filters<C_ + 1>(info);
383 }
384 }
385
387 std::array<LAFEM::UnitFilter<DataType, IndexType>, std::size_t(num_comp)> _unit_filters;
388#endif // FEAT_HAVE_CUDA
