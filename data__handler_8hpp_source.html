78 template<typename ITX_>
79 bool _contains_common_element(const ITX_* a, const ITX_* ae, const ITX_* b, const ITX_* be) const
80 {
81 std::vector<ITX_> intersection;
82
83 std::set<ITX_> aa { a, ae };
84 std::set<ITX_> bb { b, be };
85
86 std::set_intersection(aa.begin(), aa.end(),
87 bb.begin(), bb.end(),
88 std::inserter(intersection, intersection.end()));
89
90 return !intersection.empty();
91 }

181 template<typename ColoringType_>
182 explicit LagrangeDataHandler(const SpaceType& space, const ColoringType_& coloring, int hint = -1) :
183 _cell_to_dof(),
184 _cell_to_dof_sorter(),
185 _cell_to_dof_size(Index(0)),
186 _nodes(),
187 _nodes_size(Index(0)),
188 time_off_load_mesh(0.),
189 time_off_load_color(0.),
190 time_init_mesh(0.),
191 time_init_color(0.)
192 {
193 TimeStamp stamp1;
194 if constexpr(std::is_same<ColoringType_, Adjacency::Coloring>::value)
195 {
196 ASSERTM(space.get_mesh().get_num_entities(dim) == coloring.get_num_nodes(), "Coloring and space do not fit!");
197 }
198 else
199 {
200 ASSERTM(space.get_mesh().get_num_entities(dim) == coloring.size(), "Coloring and space do not fit!");
201 }
202 _nodes_size = space.get_mesh().get_vertex_set().get_num_vertices();
203 _nodes = Memory::Arbiter(sizeof(Tiny::Vector<DataType, dim>) * _nodes_size);
204 // copy the internal nodes array
205 const auto* vertex_begin = (const typename SpaceType::MeshType::VertexSetType::CoordType*)space.get_mesh().get_vertex_set().begin();
206 std::transform(vertex_begin, vertex_begin + _nodes_size*Index(dim),
207 _nodes.template typed_view<DataType>(Memory::Location::main, Memory::Access::write).get_w(),
208 [](const auto& a) ->DataType {return DataType(a);});
209
210 // define our _cell_to_dof
211 _cell_to_dof_size = space.get_mesh().get_num_entities(dim) * SpaceType::DofMappingType::dof_count;
212 _cell_to_dof = Memory::Arbiter(sizeof(IndexType) * std::size_t(_cell_to_dof_size));
213 _cell_to_dof_sorter = Memory::Arbiter(sizeof(IndexType) * std::size_t(_cell_to_dof_size));
214 // for this iterate through our target_sets and parse them in
215 {
216 Memory::TypedView<IndexType> ctdv(_cell_to_dof.view(Memory::Location::main, Memory::Access::write));
217 Memory::TypedView<IndexType> ctdsv(_cell_to_dof_sorter.view(Memory::Location::main, Memory::Access::write));
218 VoxelAssembly::fill_cell_to_dof(ctdv.get_w(), space);
219 VoxelAssembly::fill_sorter(ctdsv.get_w(), ctdv.get_r(), space);
220 }
221 TimeStamp stamp2;
222 time_init_mesh = stamp2.elapsed(stamp1);
