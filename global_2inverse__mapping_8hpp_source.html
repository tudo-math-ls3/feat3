185 template<typename PointRange_>
186 std::vector<InvMapDataType> unmap_points(const PointRange_& points)
187 {
188 const int total_ranks = _comm.size();
189
190 // Exchange bounding boxes with other ranks
191 std::vector<BoundingBoxType> bbs(static_cast<std::size_t>(total_ranks));
192 bbs.at(std::size_t(_comm.rank())) = _local_bb;
193 _comm.allgather(bbs.data(), 1, bbs.data(), 1);
194
195 // Build structure for exchanging points
196 std::vector<int> ranks;
197 std::vector<Index> indices;
198
199 Index point_idx(0);
200 for(auto it = points.begin(); it != points.end(); ++it)
201 {
202 const ImagePointType& p = *it;
203 for(int i(0); i < total_ranks; i++)
204 {
205 const BoundingBoxType& bb = bbs[std::size_t(i)];
206
207 if(is_in_bb(bb, p))
208 {
209 ranks.push_back(i);
210 indices.push_back(point_idx);
211 }
212 }
213 point_idx++;
214 }
215
216 VectorRemoteLambda<std::vector<UnmappingResult>(const ImagePointType&)> rl(_comm);
217 std::vector<std::vector<UnmappingResult>> result_pairs = rl.call(ranks, indices, points, [this](const ImagePointType& p)
218 {
219 // NOTE(mmuegge): ignore_failure is always set to true, because the RemoteLambda call is a collective operations
220 // and one process throwing an exception will cause a deadlock.
221 auto local_result = _mapping.get().unmap_point(p, true);
222
223 // Convert local result to result pairs for mpi sending
224 std::vector<UnmappingResult> result;
225 result.reserve(local_result.size());
226 for(Index i(0); i < local_result.size(); i++)
227 {
228 result.push_back({local_result.cells[i], local_result.dom_points[i]});
229 }
230
231 return result;
232 });
233
234
235 // Assemble final result from ResultPairs
236 std::vector<InvMapDataType> result(std::size_t(std::distance(points.begin(), points.end())));
237 for(std::size_t i(0); i < ranks.size(); i++)
238 {
239 const int rank = ranks[i];
240 const Index idx = indices[i];
241 auto iter = points.begin();
242 std::advance(iter, idx);
243 const ImagePointType& image_point = *iter;
244
245 std::vector<UnmappingResult>& pairs = result_pairs[i];
246
247 result[idx].img_point = image_point;
248 for(const UnmappingResult& pair : pairs)
249 {
250 result[idx].cells.push_back(pair.cell);
251 result[idx].dom_points.push_back(pair.dom_point);
252 result[idx].ranks.push_back(rank);
253 }
254 }
255
256 return result;
257 }
258
259 private:
261 static bool is_in_bb(const BoundingBoxType& bb, const ImagePointType& p)
262 {
263 constexpr DataType bb_eps(1E-2);
264 for(int i(0); i < world_dim; i++)
265 {
266 if(p[i] > (bb[1][i] + bb_eps) || p[i] < (bb[0][i] - bb_eps))
267 {
268 return false;
269 }
270 }
271 return true;
272 }
273
275 BoundingBoxType get_local_bb()
276 {
277 const auto& vertex_set = _mapping.get().get_trafo().get_mesh().get_vertex_set();
278
279 if(vertex_set.get_num_vertices() == 0)
280 {
281 return BoundingBoxType{0};
282 }
283
284 // Determine local bounding box
285 BoundingBoxType local_bb;
286
287 local_bb[0] = vertex_set[0];
288 local_bb[1] = vertex_set[0];
289 for(Index vtx_idx(0); vtx_idx < vertex_set.get_num_vertices(); vtx_idx++)
290 {
291 const auto& v = vertex_set[vtx_idx];
292
293 for(int i(0); i < world_dim; i++)
294 {
295 Math::mini(local_bb[0][i], v[i]);
296 Math::maxi(local_bb[1][i], v[i]);
297 }
298 }
299
300 return local_bb;
301 }
