83 std::string_view next_line()
84 {
85 // Calling this at the end of the buffer is an error
86 XASSERT(current < end);
87
88 // If this is called, we assume that the next part of the data is valid ascii
89 // We can thus safely skip to the next non-whitespace char
90 while(current < end && (std::isspace(static_cast<unsigned char>(*current)) != 0))
91 {
92 current++;
93 }
94
95 // Search for next newline
96 const char* eol = static_cast<const char*>(std::memchr(current, '\n', std::size_t(end - current)));
97
98 if(eol == nullptr)
99 {
100 eol = end;
101 }
102
103 // String views are constructed via a (start, len) pair
104 // So convert from end-of-line pointer to length
105 const auto line_len = std::size_t(eol - current);
106
107 const std::string_view line{current, line_len};
108
109 // Advance past line. +1 to skip over newline
110 current = eol + 1 < end ? eol + 1 : end;
111
112 // Check we are still in bounds
113 XASSERT(current <= end);
114
115 return line;
116 }
117
119 void advance(std::size_t len)
120 {
121 XASSERT(current + len <= end);
122 current += len;
123 }
124
126 template<typename... Args>
127 void parse_line(Args&... args)
128 {
129 const std::string_view line = next_line();
130 std::istringstream stream((std::string(line)));
131
132 ((stream >> args), ...);
133
134 XASSERT(!stream.fail());
135 }
136
138 template<typename T>
139 void memcpy(T* t)
140 {
141 XASSERT(current + sizeof(T) <= end);
142 std::memcpy(t, current, sizeof(T));
143 advance(sizeof(T));
144 }

297 void read_root_markup()
298 {
299 // NOTE(mmuegge): Unlike FEATs .xml format, .msh allows meshes with mixed
300 // elements. In ASCII meshes each element is tagged with its type, in
301 // binary meshes elements are split into chunks of elements of the same
302 // type. We thus need to scan through the whole file to ensure it contains
303 // a single element type that is representable by FEAT.
304
305 FileView view(_file_contents);
306
307 XASSERT(view.next_line() == "$MeshFormat");
308
309 // Next line is:
310 // version-number file-type data-size
311 view.parse_line(_version, _is_binary, _data_size);
312 XASSERTM(_version == 2.2, "Unsupported .msh file format version!");
313 XASSERTM(_data_size == 8, "Unsupported .msh data size!");
314
315 if(_is_binary)
316 {
317 int one(0);
318 view.memcpy(&one);
319
320 XASSERTM(one == 1, ".msh files with non-native endianess are not supported!");
321 }
322
323 XASSERT(view.next_line() == "$EndMeshFormat");
324
325 while(!view.done())
326 {
327 std::string_view line = view.next_line();
328
329 if(line == "$Nodes")
330 {
331 scan_nodes(view);
332 }
333 else if(line == "$Elements")
334 {
335 scan_elements(view);
336 }
337 }
338
339 XASSERTM(_element_type != -1, ".msh file with no elements");
340 _shape_type = element_type_to_shape_type(_element_type);
341 _shape_dim = element_type_to_shape_dim(_element_type);
342 _mesh_type_string = "conformal:" + stringify(_shape_type) + ":" + stringify(_shape_dim) + ":3";
343 _have_root_markup = true;
344 }

359 template<typename MeshType_>
360 std::unique_ptr<RootMeshNode<MeshType_>> parse(MeshAtlas<MeshType_>& mesh_atlas, PartitionSet* part_set = nullptr)
361 {
362 using MeshShapeType = typename MeshType_::ShapeType;
363 using DataType = typename MeshType_::CoordType;
364 using VertexType = typename MeshType_::VertexType;
365
366 static constexpr std::size_t num_verts = Shape::FaceTraits<MeshShapeType, 0>::count;
367 static constexpr int shape_dim = MeshType_::shape_dim;
368
369 if(!_have_root_markup)
370 {
371 read_root_markup();
372 }
373
374 XASSERT(num_verts == nodes_per_element(_element_type));
375
376 // Validate that the chosen mesh type fits to the data
377 if(sizeof(DataType) != _data_size)
378 {
379 XABORTM("CoordType of MeshType_ does not match data type of .msh file");
380 }
381
382 XASSERT(MeshShapeType::dimension == _shape_dim);
383
384 if(_shape_type == ParsedShapeType::hypercube && _shape_dim == 2)
385 {
386 XASSERT((std::is_same_v<MeshShapeType, Shape::Hypercube<2>>));
387 }
388 else if(_shape_type == ParsedShapeType::hypercube && _shape_dim == 3)
389 {
390 XASSERT((std::is_same_v<MeshShapeType, Shape::Hypercube<3>>));
391 }
392 else if(_shape_type == ParsedShapeType::simplex && _shape_dim == 2)
393 {
394 XASSERT((std::is_same_v<MeshShapeType, Shape::Simplex<2>>));
395 }
396 else if(_shape_type == ParsedShapeType::simplex && _shape_dim == 3)
397 {
398 XASSERT((std::is_same_v<MeshShapeType, Shape::Simplex<3>>));
399 }
400
401 // part_set is only for compatibility
402 (void)part_set;
403
404 // Create mesh
405 std::array<Index, shape_dim + 1> size;
406 size.fill(0);
407 size[0] = _num_nodes;
408 size[shape_dim] = _num_elements;
409
410 auto mesh = std::make_unique<MeshType_>(size.data());
411
412 // Parse nodes
413 auto& vertex_set = mesh->get_vertex_set();
414 FileView node_view(_file_contents, _nodes_start);
415 if(_is_binary)
416 {
417 // Safety: If we reached this point, then scan_nodes succeeded,
418 // which means there are the expected number of bytes between $Nodes and $EndNodes.
419 // The following memory accesses are thus safe.
420 for(std::size_t node(0); node < _num_nodes; node++)
421 {
422 // Read node index
423 int vertex_index(0);
424 node_view.memcpy(&vertex_index);
425
426 // This might occur in meshes with non-contiguous indices for vertices.
427 // We currently do not support those.
428 XASSERT(std::size_t(vertex_index - 1) < _num_nodes);
429
430 // Copy coords
431 for(int j(0); j < get_world_dim(); j++)
432 {
433 // NOTE: .msh files are 1-indexed, hence vertex_index - 1 to transform to 0-based indices
434 node_view.memcpy(&vertex_set[Index(vertex_index - 1)][j]);
435 }
436 }
437 }
438 else
439 {
440 // Safety: Cast to int is safe because _num_nodes is stored as an int as per .msh docs.
441 for(int i(0); i < (int)_num_nodes; i++)
442 {
443 Index vertex_index(~Index(0));
444 double x(0.0);
445 double y(0.0);
446 double z(0.0);
447
448 node_view.parse_line(vertex_index, x, y, z);
449
450 // This might occur in meshes with non-contiguous indices for vertices.
451 // We currently do not support those.
452 XASSERT(std::size_t(vertex_index - 1) < _num_nodes);
453
454 // NOTE: .msh files are 1-indexed, hence vertex_index - 1 to transform to 0-based indices
455 vertex_set[vertex_index - 1] = VertexType{x, y, z};
456 }
457 }
458
459 // Parse elements
460 auto& element_index_set = mesh->template get_index_set<shape_dim, 0>();
461 if(_is_binary)
462 {
463 for(BinaryElementsHeader& header : _element_headers)
464 {
465 FileView chunk_view(_file_contents, header.elements_start);
466
467 // Safety: If we reached this point, then scan_elements succeeded,
468 // which means there are the expected number of bytes between $Elements and $EndElements.
469 // The following memory accesses are thus safe.
470 for(std::size_t e(0); e < std::size_t(header.num_elements); e++)
471 {
472 // Determine element index
473 int element_index(0);
474 chunk_view.memcpy(&element_index);
475
476 // This might occur in meshes with non-contiguous indices for elements.
477 // We currently do not support those.
478 XASSERT(std::size_t(element_index - 1) < _num_elements);
479
480 // Skip past tags
481 chunk_view.advance(std::size_t(header.tags_per_element) * sizeof(int));
482
483 for(int n(0); n < (int)num_verts; n++)
484 {
485 int node_index(0);
486 chunk_view.memcpy(&node_index);
487
488 // NOTE: .msh indexing is one-based
489 element_index_set(Index(element_index - 1), msh_to_feat_order(header.type, n)) = Index(node_index - 1);
490 }
491 }
492 }
493 }
494 else
495 {
496 FileView view(_file_contents, _elements_start);
497
498 for(int i(0); i < (int)_num_elements; i++)
499 {
500 std::string_view line = view.next_line();
501 std::istringstream stream((std::string(line)));
502
503 int element_index(0);
504 int type(0);
505 int num_tags(0);
506
507 stream >> element_index >> type >> num_tags;
508
509 // This might occur in meshes with non-contiguous indices for elements.
510 // We currently do not support those.
511 XASSERT(std::size_t(element_index - 1) < _num_elements);
512
513 // Skip over tags
514 for(int tag(0); tag < num_tags; tag++)
515 {
516 int dummy(0);
517 stream >> dummy;
518 }
519
520 for(int n(0); n < (int)num_verts; n++)
521 {
522 int node_number(0);
523 stream >> node_number;
524
525 element_index_set(Index(element_index - 1), msh_to_feat_order(type, n)) = Index(node_number - 1);
526 }
527
528 XASSERT(!stream.fail());
529 }
530 }
531
532 // Validate element orientation
533
534 mesh->deduct_topology_from_top();
535 return std::make_unique<RootMeshNode<MeshType_>>(std::move(mesh), &mesh_atlas);
536 }

540 static std::size_t binary_element_chunk_size(const BinaryElementsHeader& header)
541 {
542 // 4 bytes for element index, 4 bytes per tag, 4 bytes per vertex index
543 return std::size_t(header.num_elements) * (sizeof(int) + (sizeof(int) * std::size_t(header.tags_per_element)) +
544 (sizeof(int) * std::size_t(nodes_per_element(header.type))));
545 }

548 static ParsedShapeType element_type_to_shape_type(int type)
549 {
550 switch(type)
551 {
552 case 2: return ParsedShapeType::simplex; // Triangle
553 case 3: return ParsedShapeType::hypercube; // Quadrangle
554 case 4: return ParsedShapeType::simplex; // 4-point tetrahedron
555 case 5: return ParsedShapeType::hypercube; // 8-point hexahedron
556 default: XABORTM("Unsupported .msh element type!");
557 }
558 }
559
561 static int element_type_to_shape_dim(int type)
562 {
563 switch(type)
564 {
565 case 2: return 2; // Triangle
566 case 3: return 2; // Quadrangle
567 case 4: return 3; // 4-point tetrahedron
568 case 5: return 3; // 8-point hexahedron
569 default: XABORTM("Unsupported .msh element type!");
570 }
571 }
572
574 static int nodes_per_element(int type)
575 {
576 switch(type)
577 {
578 case 2: return 3; // Triangle
579 case 3: return 4; // Quadrangle
580 case 4: return 4; // 4-point tetrahedron
581 case 5: return 8; // 8-point hexahedron
582 default: XABORTM("Unsupported .msh element type!");
583 }
584 }

593 static int msh_to_feat_order(int type, int node)
594 {
595 // Re-ordering for quadrangles
596 // .msh are ordered counter-clockwise, FEAT in a zig-zag pattern
597 static constexpr std::array<int, 4> quadrangle_ordering{0, 1, 3, 2};
598
599 // Re-ordering for hexahedrons
600 static constexpr std::array<int, 8> hexahedron_ordering{0, 1, 3, 2, 4, 5, 7, 6};
601
602 // Re-ordering
603 switch(type)
604 {
605 case 2: return node; // triangles are ordered the same
606 case 3: return quadrangle_ordering.at(std::size_t(node));
607 case 4: return node; // tetrahedrons are ordered the same
608 case 5: return hexahedron_ordering.at(std::size_t(node));
609 default: XABORTM("Unsupported .msh element type!");
610 }
611 }

614 void scan_nodes(FileView& view)
615 {
616 // Next line contains number of nodes
617 view.parse_line(_num_nodes);
618
619 // Skip past nodes
620 _nodes_start = view.position();
621 if(_is_binary)
622 {
623 // Each node consists of an integer index and 3 coordinates
624 view.advance(_num_nodes * (sizeof(int) + (3 * _data_size)));
625 }
626 else
627 {
628 for(Index i(0); i < _num_nodes; i++)
629 {
630 view.next_line();
631 }
632 }
633
634 XASSERT(view.next_line() == "$EndNodes");
635 }
636
638 void scan_elements(FileView& view)
639 {
640 view.parse_line(_num_elements);
641
642 _elements_start = view.position();
643 if(_is_binary)
644 {
645 // Each chunk has a header containing:
646 // - the element type,
647 // - the number of elements of the chunk,
648 // - the number of tags per element of the chunk
649 // We handle each chunk one by one until we have seen all elements
650 std::size_t elements_handled(0);
651 while(elements_handled < _num_elements)
652 {
653 BinaryElementsHeader header{};
654
655 // Read header values
656 view.memcpy(&header.type);
657 view.memcpy(&header.num_elements);
658 view.memcpy(&header.tags_per_element);
659
660 // Record current positions as start of elements for this header
661 header.elements_start = view.position();
662
663 // Skip past actual element entries
664 view.advance(binary_element_chunk_size(header));
665
666 if(_element_type == -1)
667 {
668 _element_type = header.type;
669 }
670 else
671 {
672 XASSERT(_element_type == header.type);
673 }
674
675 elements_handled += std::size_t(header.num_elements);
676 _element_headers.push_back(header);
677 }
678 }
679 else
680 {
681 for(Index i(0); i < _num_elements; i++)
682 {
683 int number(0);
684 int type(0);
685
686 view.parse_line(number, type);
687
688 if(_element_type == -1)
689 {
690 _element_type = type;
691 }
692 else
693 {
694 XASSERT(_element_type == type);
695 }
696 }
697 }
698
699 XASSERT(view.next_line() == "$EndElements");
700 }
