97 void add(const SparseVector<DT_, IT_>& vec)
98 {
99 XASSERTM(vec.size() == this->size(), "invalid vector size");
100 const Memory::TypedView<IT_> v_idx = vec.indices_view_r();
101 const Memory::TypedView<DT_> v_val = vec.elements_view_r();
102 Index nze = vec.num_nzes();
103 for(Index i = 0; i < nze; ++i)
104 {
105 auto ib = _entries.emplace(v_idx[i], BlockedValueType(v_val[i]));
106 if(!ib.second)
107 ib.first->second = BlockedValueType(v_val[i]);
108 }
