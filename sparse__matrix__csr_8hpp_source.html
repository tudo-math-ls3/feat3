386 template <typename DT2_, typename IT2_>
387 void convert(const SparseMatrixBanded<DT2_, IT2_> & other)
388 {
389 this->clear();
390
391 this->_scalar_index.push_back(other.num_rows());
392 this->_scalar_index.push_back(other.num_cols());
393 this->_scalar_index.push_back(other.num_nzes());
394
395 if (other.num_nzes() == 0)
396 return;
397
398 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * (_rows() + 1)));
399 this->_indices_size.push_back(_rows() + 1);
400 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * _nzes()));
401 this->_indices_size.push_back(_nzes());
402 this->_elements.push_back(Memory::Arbiter(sizeof(DT_) * _nzes()));
403 this->_elements_size.push_back(_nzes());
404
405 Memory::TypedView<IT_> row_ptr = this->row_ptr_view_w();
406 Memory::TypedView<IT_> col_idx = this->col_idx_view_w();
407 Memory::TypedView<DT_> val = this->val_view_w();
408
409 const Memory::TypedView<DT_> cval = other.val_view_r();
410 const Memory::TypedView<IT_> coffsets = other.offsets_view_r();
411 const Index nbands(other.num_bands());
412 const Index nrows(other.num_rows());
413 const Index ncols(other.num_cols());
414 //const Index nnzes(other.num_nzes());
415
416 row_ptr[0] = 0;
417
418 // Search first offset of the upper triangular matrix
419 Index k(0);
420 while (k < nbands && coffsets[k] + 1 < nrows)
421 {
422 ++k;
423 }
424
425 IT_ ue(0);
426 // iteration over all offsets of the lower triangular matrix
427 for (Index i(k + 1); i > 0;)
428 {
429 --i;
430
431 // iteration over all offsets of the upper triangular matrix
432 for (Index j(nbands + 1); j > 0;)
433 {
434 --j;
435
436 // iteration over all rows which contain the offsets between offset i and offset j
437 Index so1 = Index(0);
438 Index so2 = nrows;
439 Index eo1 = Index(0);
440 Index eo2 = nrows;
441 if(i < nbands)
442 so1 = Math::max(ncols + Index(1), nrows + ncols - Index(coffsets[i])) - ncols - Index(1);
443 if(j < nbands)
444 eo1 = Math::min(nrows, ncols + nrows - Index(coffsets[j]) - Index(1));
445 if(i > 0)
446 so2 = Math::max(ncols + Index(1), nrows + ncols - Index(coffsets[i-1])) - ncols - Index(1);
447 if(j > 0)
448 eo2 = Math::min(nrows, ncols + nrows - Index(coffsets[j-1]) - Index(1));
449
450 const Index start = Math::max(so1, eo1);
451 const Index end = Math::min(so2, eo2);
452
453 for (Index l(start); l < end; ++l)
454 {
455 for (Index a(i); a < j; ++a)
456 {
457 val[ue] = cval[a * nrows + l];
458 col_idx[ue] = IT_(l + coffsets[a] + 1 - nrows);
459 ++ue;
460 }
461 row_ptr[l + 1] = ue;
462 }
463 }
464 }
465 }

474 template <typename DT2_, typename IT2_, int block_height_, int block_width_>
475 void convert(const SparseMatrixBCSR<DT2_, IT2_, block_height_, block_width_> & other)
476 {
477 this->clear();
478
479 this->_scalar_index.push_back(other.num_rows_raw());
480 this->_scalar_index.push_back(other.num_cols_raw());
481 this->_scalar_index.push_back(other.num_nzes_raw());
482
483 if (other.num_nzes_raw() == 0)
484 return;
485
486 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * (_rows() + 1)));
487 this->_indices_size.push_back(_rows() + 1);
488 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * _nzes()));
489 this->_indices_size.push_back(_nzes());
490 this->_elements.push_back(Memory::Arbiter(sizeof(DT_) * _nzes()));
491 this->_elements_size.push_back(_nzes());
492
493 Memory::TypedView<IT_> row_ptr = this->row_ptr_view_w();
494 Memory::TypedView<IT_> col_idx = this->col_idx_view_w();
495 Memory::TypedView<DT_> val = this->val_view_w();
496
497 const Memory::TypedView<IT_> brow_ptr = other.row_ptr_view_r();
498 const Memory::TypedView<IT_> bcol_idx = other.col_idx_view_r();
499 const Memory::TypedView<Tiny::Matrix<DT_, block_height_, block_width_>> bval = other.val_view_r();
500 const Index brows = other.num_rows();
501
502 row_ptr[0] = IT_(0);
503 for (Index orow(0), ait(0) ; orow < brows ; ++orow)
504 {
505 for (int row(0) ; row < block_height_ ; ++row)
506 {
507 for(Index ocol(brow_ptr[orow]) ; ocol < brow_ptr[orow + 1] ; ++ocol)
508 {
509 for (int col(0) ; col < block_width_ ; ++col, ++ait)
510 {
511 val[ait] = bval[ocol](row,col);
512 col_idx[ait] = bcol_idx[ocol] * IT_(block_width_) + IT_(col);
513 }
514 }
515 row_ptr[orow * Index(block_height_) + Index(row) + 1] = IT_(ait);
516 }
517 }

527 template <typename MT_>
528 void convert(const MT_ & a)
529 {
530 XASSERT(a.num_nzes_raw() > 0);
531
532 const Index arows(a.num_rows_raw());
533 const Index acols(a.num_cols_raw());
534 const Index anzes(a.num_nzes_raw());
535
536 SparseMatrixCSR<DT_, IT_> csr(arows, acols, anzes);
537
538 Memory::TypedView<IT_> row_ptr = csr.row_ptr_view_w();
539 Memory::TypedView<IT_> col_idx = csr.col_idx_view_w();
540 Memory::TypedView<DT_> val = csr.val_view_w();
541
542 row_ptr[0] = IT_(0);
543 for (Index i(0); i < arows; ++i)
544 {
545 row_ptr[i + 1] = row_ptr[i] + IT_(a.row_degree(i));
546 }
547
548 for (Index i(0); i < arows; ++i)
549 {
550 a.get_row_col_indices(i, &col_idx[row_ptr[i]], IT_(0));
551 a.get_row_values(i, &val[row_ptr[i]]);
552 }
553
554 val.release();
555 col_idx.release();
556 row_ptr.release();
557
558 this->move(std::move(csr));
559 }

696 void read_from(FileMode mode, std::istream& file)
697 {
698 switch(mode)
699 {
700 case FileMode::fm_mtx:
701 {
702 this->clear();
703 this->_scalar_index.push_back(0);
704 this->_scalar_index.push_back(0);
705 this->_scalar_index.push_back(0);
706
707 std::map<IT_, std::map<IT_, DT_> > entries; // map<row, map<column, value> >
708
709 Index ue(0);
710 String line;
711 std::getline(file, line);
712 const bool general((line.find("%%MatrixMarket matrix coordinate real general") != String::npos) ? true : false);
713 const bool symmetric((line.find("%%MatrixMarket matrix coordinate real symmetric") != String::npos) ? true : false);
714
715 if (symmetric == false && general == false)
716 {
717 XABORTM("Input-file is not a compatible mtx-file");
718 }
719
720 while(!file.eof())
721 {
722 std::getline(file,line);
723 if (file.eof())
724 XABORTM("Input-file is empty");
725
726 String::size_type begin(line.find_first_not_of(" "));
727 if (line.at(begin) != '%')
728 break;
729 }
730 {
731 String::size_type begin(line.find_first_not_of(" "));
732 line.erase(0, begin);
733 String::size_type end(line.find_first_of(" "));
734 String srow(line, 0, end);
735 Index row((Index)atol(srow.c_str()));
736 line.erase(0, end);
737
738 begin = line.find_first_not_of(" ");
739 line.erase(0, begin);
740 end = line.find_first_of(" ");
741 String scol(line, 0, end);
742 Index col((Index)atol(scol.c_str()));
743 line.erase(0, end);
744 _rows() = row;
745 _cols() = col;
746 }
747
748 while(!file.eof())
749 {
750 std::getline(file, line);
751 if (file.eof())
752 break;
753
754 String::size_type begin(line.find_first_not_of(" "));
755 line.erase(0, begin);
756 String::size_type end(line.find_first_of(" "));
757 String srow(line, 0, end);
758 IT_ row((IT_)atol(srow.c_str()));
759 --row;
760 line.erase(0, end);
761
762 begin = line.find_first_not_of(" ");
763 line.erase(0, begin);
764 end = line.find_first_of(" ");
765 String scol(line, 0, end);
766 IT_ col((IT_)atol(scol.c_str()));
767 --col;
768 line.erase(0, end);
769
770 begin = line.find_first_not_of(" ");
771 line.erase(0, begin);
772 end = line.find_first_of(" ");
773 String sval(line, 0, end);
774 DT_ tval((DT_)atof(sval.c_str()));
775
776 entries[IT_(row)].insert(std::pair<IT_, DT_>(col, tval));
777 ++ue;
778 if (symmetric == true && row != col)
779 {
780 entries[IT_(col)].insert(std::pair<IT_, DT_>(row, tval));
781 ++ue;
782 }
783 }
784 _nzes() = ue;
785
786 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * (_rows() + 1)));
787 this->_indices_size.push_back(num_rows() + 1);
788 this->_indices.push_back(Memory::Arbiter(sizeof(IT_) * _nzes()));
789 this->_indices_size.push_back(_nzes());
790 this->_elements.push_back(Memory::Arbiter(sizeof(DT_) * _nzes()));
791 this->_elements_size.push_back(_nzes());
792
793 Memory::TypedView<IT_> trow_ptr(this->row_ptr_view_w());
794 Memory::TypedView<IT_> tcol_idx(this->col_idx_view_w());
795 Memory::TypedView<DT_> tval(this->val_view_w());
796
797 IT_ idx(0);
798 Index row_idx(0);
799 for (auto row : entries)
800 {
801 trow_ptr[row_idx] = idx;
802 for (auto col : row.second)
803 {
804 tcol_idx[idx] = col.first;
805 tval[idx] = col.second;
806 ++idx;
807 }
808 row.second.clear();
809 ++row_idx;
810 }
811 trow_ptr[num_rows()] = IT_(ue);
812 entries.clear();
813
814 break;
815 }
816
817 case FileMode::fm_csr:
818 [[fallthrough]];
819
820 case FileMode::fm_binary:
821 this->template _deserialize<double, std::uint64_t>(FileMode::fm_csr, file);
822 break;
823
824 default:
825 XABORTM("Filemode not supported!");
826 }
827 }

869 void write_out(FileMode mode, std::ostream& file, bool symmetric = false) const
870 {
871 switch(mode)
872 {
873 case FileMode::fm_csr:
874 [[fallthrough]];
875
876 case FileMode::fm_binary:
877 this->template _serialize<double, std::uint64_t>(FileMode::fm_csr, file);
878 break;
879
880 case FileMode::fm_mtx:
881 {
882 const Memory::TypedView<IT_> trow_ptr = this->row_ptr_view_r();
883 const Memory::TypedView<IT_> tcol_idx = this->col_idx_view_r();
884 const Memory::TypedView<DT_> tval = this->val_view_r();
885
886 if (symmetric)
887 {
888 file << "%%MatrixMarket matrix coordinate real symmetric\n";
889 std::vector<IT_> rowv;
890 std::vector<IT_> colv;
891 std::vector<DT_> valv;
892
893 for (Index row(0) ; row < num_rows() ; ++row)
894 {
895 const IT_ end(trow_ptr[row + 1]);
896 for (IT_ i(trow_ptr[row]) ; i < end ; ++i)
897 {
898 const IT_ col(tcol_idx[i]);
899 if (row >= col)
900 {
901 rowv.push_back(IT_(row + 1));
902 colv.push_back(col + 1);
903 valv.push_back(tval[i]);
904 }
905 }
906 }
907 file << this->num_rows() << " " << this->num_cols() << " " << valv.size() << "\n";

1523 SparseMatrixCSR transpose() const
1524 {
1525 const Index txrows(this->num_rows());
1526 const Index txcolumns(this->num_cols());
1527 const Index txnonzeros(this->num_nzes());
1528
1529 if(txnonzeros <= Index(0))
1530 return SparseMatrixCSR(txcolumns, txrows);
1531
1532 SparseMatrixCSR<DT_, IT_> trans_mat(txcolumns, txrows, txnonzeros);
1533
1534 {
1535 const Memory::TypedView<IT_> ptxcol_idx(this->col_idx_view_r());
1536 const Memory::TypedView<IT_> ptxrow_ptr(this->row_ptr_view_r());
1537 const Memory::TypedView<DT_> ptxval(this->val_view_r());
1538
1539 Memory::TypedView<IT_> ptrow_ptr(trans_mat.row_ptr_view_w());
1540 Memory::TypedView<IT_> ptcol_idx(trans_mat.col_idx_view_w());
1541 Memory::TypedView<DT_> ptval(trans_mat.val_view_w());
1542
1543 for (Index i(0); i <= txcolumns; ++i)
1544 {
1545 ptrow_ptr[i] = IT_(0);
1546 }
1547
1548 for (Index i(0); i < txnonzeros; ++i)
1549 {
1550 ++ptrow_ptr[ptxcol_idx[i] + 1];
1551 }
1552
1553 for (Index i(1); i < txcolumns; ++i)
1554 {
1555 ptrow_ptr[i + 1] += ptrow_ptr[i];
1556 }
1557
1558 for (Index i(0); i < txrows; ++i)
1559 {
1560 for (IT_ k(ptxrow_ptr[i]); k < ptxrow_ptr[i+1]; ++k)
1561 {
1562 const IT_ l(ptxcol_idx[k]);
1563 const IT_ j(ptrow_ptr[l]);
1564 ptval[j] = ptxval[k];
1565 ptcol_idx[j] = IT_(i);
1566 ++ptrow_ptr[l];
1567 }
1568 }
1569
1570 for (Index i(txcolumns); i > 0; --i)
1571 {
1572 ptrow_ptr[i] = ptrow_ptr[i - 1];
1573 }
1574 ptrow_ptr[0] = 0;
1575 }
1576
1577 return trans_mat;
1578 }

1934 Memory::TypedView<DT_> data_x = this->val_view_rw();
1935 const Memory::TypedView<DT_> data_d = d.val_view_r();
1936 const Memory::TypedView<DT_> data_a = a.val_view_r();
1937 const Memory::TypedView<DT_> data_b = b.val_view_r();

1944 const Memory::TypedView<IT_> row_ptr_b = b.row_ptr_view_r();
1945 const Memory::TypedView<IT_> col_idx_b = b.col_idx_view_r();

1947 // loop over all rows of D and X, resp.
1948 for(IT_ i(0); i < IT_(this->num_rows()); ++i)

1951 for(IT_ ik(row_ptr_d[i]); ik < row_ptr_d[i+1]; ++ik)

1953 // get column index k
1954 const IT_ k = col_idx_d[ik];

1957 for(IT_ kl(row_ptr_a[k]); kl < row_ptr_a[k+1]; ++kl)
1958 {

1966 // loop over all non-zeros X_ij in row i of X and

2419 void permute(const Adjacency::Permutation & perm_row, const Adjacency::Permutation & perm_col)
2420 {
2421 if (perm_row.size() == 0 && perm_col.size() == 0)
2422 return;
2423
2424 XASSERTM(perm_row.size() == this->num_rows(), "Container rows does not match permutation size");
2425 XASSERTM(perm_col.size() == this->num_cols(), "Container columns does not match permutation size");
2426
2427 // http://de.mathworks.com/help/matlab/math/sparse-matrix-operations.html#f6-13070
2428 IT_ * temp_row_ptr = new IT_[num_rows() + 1];
2429 IT_ * temp_col_idx = new IT_[num_nzes()];
2430 DT_ * temp_val = new DT_[num_nzes()];
2431
2432 Memory::TypedView<IT_> row_ptr = this->row_ptr_view_rw();
2433 Memory::TypedView<IT_> col_idx = this->col_idx_view_rw();
2434 Memory::TypedView<DT_> val = this->val_view_rw();
2435
2436 const Index * perm_pos = perm_row.get_perm_pos();
2437
2438 //permute rows from source to temp_*
2439 Index new_start(0);
2440 temp_row_ptr[0] = 0;
2441 for (Index row(0) ; row < this->num_rows() ; ++row)
2442 {
2443 Index row_size(row_ptr[perm_pos[row] + 1] - row_ptr[perm_pos[row]]);
2444
2445 //iterate over all elements in single one new and old row
2446 for (Index i(new_start), j(row_ptr[perm_pos[row]]) ; i < new_start + row_size ; ++i, ++j)
2447 {
2448 temp_col_idx[i] = col_idx[j];
2449 temp_val[i] = val[j];
2450 }
2451
2452 new_start += row_size;
2453 temp_row_ptr[row+1] = (IT_)new_start;
2454 }
2455
2456 //use inverse col permutation as lookup table: i -> new location of i
2457 Adjacency::Permutation perm_col_inv = perm_col.inverse();
2458 perm_pos = perm_col_inv.get_perm_pos();
2459
2460 //permute columns from temp_* to result
2461 Memory::memcopy_main(row_ptr.get_w(), temp_row_ptr, (num_rows() + 1) * sizeof(IT_));
2462 Memory::memcopy_main(val.get_w(), temp_val, num_nzes() * sizeof(DT_));
2463 for (Index i(0) ; i < num_nzes() ; ++i)
2464 {
2465 col_idx[i] = (IT_)perm_pos[temp_col_idx[i]];
2466 }
2467
2468 delete[] temp_row_ptr;
2469 delete[] temp_col_idx;
2470 delete[] temp_val;
2471
2472 //sort columns in every row by column index
2473 IT_ swap_key;
2474 DT_ swap_val;
2475 for (Index row(0) ; row < num_rows() ; ++row)
2476 {
2477 Index offset(row_ptr[row]);
2478 Index row_size(row_ptr[row+1] - row_ptr[row]);
2479 for (Index i(1), j ; i < row_size ; ++i)
2480 {
