122 template <typename ITX_>
123 inline Index _start_offset(const Index i, const ITX_ * const offsets,
124 const Index rows_in, const Index columns_in, const Index noo) const
125 {
126 if (i == Index(-1))
127 {
128 return rows_in;
129 }
130 else if (i == noo)
131 {
132 return Index(0);
133 }
134 else
135 {
136 return Math::max(columns_in + Index(1), rows_in + columns_in - Index(offsets[i])) - columns_in - Index(1);
137 }
138 }
139
141 template <typename ITX_>
142 inline Index _end_offset(const Index i, const ITX_ * const offsets,
143 const Index rows_in, const Index columns_in, const Index noo) const
144 {
145 if (i == Index (-1))
146 {
147 return rows_in - 1;
148 }
149 else if (i == noo)
150 {
151 return Index(-1);
152 }
153 else
154 {
155 return Math::min(rows_in, columns_in + rows_in - Index(offsets[i]) - Index(1)) - Index(1);
156 }
157 }

206 this->_scalar_index.assign(layout_in._scalar_index.begin(), layout_in._scalar_index.end());

226 explicit SparseMatrixBanded(const Index rows_in, const Index columns_in,
227 DenseVector<DT_, IT_> & val_in,
228 DenseVector<IT_, IT_> & bands_in)
229 {
230 if (val_in.size() != rows_in * bands_in.size())
231 {
232 XABORTM("Size of values does not match to number of offsets and row count!");
233 }
234
235 this->_scalar_index.push_back(rows_in);
236 this->_scalar_index.push_back(columns_in);
237
238 Index tnonzeros(0);
239
240 Memory::TypedView<IT_> vbands(bands_in.elements_view_r());
241 for (Index i(0); i < bands_in.size(); ++i)
242 {
243 const Index toffset(Index(vbands(i)));
244
245 if (toffset + Index(2) > rows_in + columns_in)
246 {
247 XABORTM("Offset out of matrix!");
248 }
249
250 tnonzeros += columns_in + Math::min(rows_in, columns_in + rows_in - toffset - 1) - Math::max(columns_in + rows_in - toffset - 1, columns_in);
251 }
252 vbands.release();
253
254 this->_scalar_index.push_back(bands_in.size());
255 this->_scalar_index.push_back(tnonzeros);
256
257 this->_elements.push_back(val_in.elements_arbiter().attach());
258 this->_elements_size.push_back(val_in.size());
259 this->_indices.push_back(bands_in.elements_arbiter().attach());
260 this->_indices_size.push_back(bands_in.size());
261 }

283 explicit SparseMatrixBanded(const Adjacency::Graph & graph)
284 {
285 Index num_rows = graph.get_num_nodes_domain();
286 Index num_cols = graph.get_num_nodes_image();
287
288 const Index * dom_ptr(graph.get_domain_ptr());
289 const Index * img_idx(graph.get_image_idx());
290
291 std::set<IT_> moffsets;
292
293 for(Index row(0); row < num_rows; ++row)
294 {
295 for(Index j(dom_ptr[row]); j < dom_ptr[row+1]; ++j)
296 {
297 moffsets.insert(IT_(num_rows - 1 + img_idx[j] - row));
298 }
299 }
300
301 DenseVector<IT_, IT_> toffsets(Index(moffsets.size()));
302 Memory::TypedView<IT_> ptoffsets = toffsets.elements_view_w();
303
304 IT_ idx(0);
305 for (auto off : moffsets)
306 {
307 ptoffsets[idx] = off;
308 ++idx;
309 }
310
311 DenseVector<DT_, IT_> tval(Index(moffsets.size()) * num_rows, DT_(0));
312
313 this->move(SparseMatrixBanded(num_rows, num_cols, tval, toffsets));
314 }

450 template <typename DT2_, typename IT2_>
451 void convert(const SparseMatrixCSR<DT2_, IT2_> & csr)
452 {
453 XASSERT(csr.num_nzes() > 0);
454
455 const Memory::TypedView<IT_> row_ptr = csr.row_ptr_view_r();
456 const Memory::TypedView<IT_> col_idx = csr.col_idx_view_r();
457 const Memory::TypedView<DT_> csr_val = csr.val_view_r();
458
459 std::set<IT_> offset_set;
460 Index nrows(csr.num_rows());
461 Index ncolumns(csr.num_cols());
462 for (Index row(0) ; row < nrows ; ++row)
463 {
464 for (Index i(row_ptr[row]) ; i < row_ptr[row+1] ; ++i)
465 {
466 //compute band offset for each non zero entry
467 IT_ off = IT_(col_idx[i] - row + nrows - 1);
468 offset_set.insert(off);
469 }
470 }
471
472 DenseVector<DT_, IT_> val_new(Index(offset_set.size()) * nrows, DT_(0));
473 Memory::TypedView<DT_ > pval = val_new.elements_view_w();
474 for (Index row(0) ; row < nrows ; ++row)
475 {
476 for (Index i(row_ptr[row]) ; i < row_ptr[row+1] ; ++i)
477 {
478 Index col = col_idx[i];
479 Index offset = (col - row + nrows - 1);
480 Index band = Index(std::distance(offset_set.begin(), offset_set.find(IT_(offset))));
481 pval[band * nrows + row] = csr_val[i];
482 }
483 }
484
485 DenseVector<IT_, IT_> offsets_new(Index(offset_set.size()));
486 Memory::TypedView<IT_> poffsets = offsets_new.elements_view_w();
487 auto it_offset = offset_set.begin();
488 for (Index i(0) ; i < offset_set.size() ; ++i, ++it_offset)
489 {
490 poffsets[i] = *it_offset;
491 }
492 offset_set.clear();
493
494 pval.release();
495 poffsets.release();
496
497 SparseMatrixBanded<DT_, IT_> temp(nrows, ncolumns, val_new, offsets_new);
498 this->move(std::move(temp));
499 }

507 void write_out(FileMode mode, const String& filename) const
508 {
509 std::ios_base::openmode bin = std::ofstream::out | std::ofstream::binary;
510 if(mode == FileMode::fm_mtx)
511 bin = std::ofstream::out;
512 std::ofstream file;
513 char* buff = nullptr;
514 if(mode == FileMode::fm_mtx)
515 {
516 buff = new char[LAFEM::file_out_stream_buffer_size];
517 file.rdbuf()->pubsetbuf(buff, LAFEM::file_out_stream_buffer_size);
518 }
519 file.open(filename.c_str(), bin);
520 if(! file.is_open())
521 XABORTM("Unable to open Matrix file " + filename);
522 write_out(mode, file);
523 file.close();
524 delete[] buff;
525 }
532 void write_out(FileMode mode, std::ostream& file) const
533 {
534 switch(mode)
535 {
536 case FileMode::fm_bm:
537 [[fallthrough]];
538
539 case FileMode::fm_binary:
540 if (! std::is_same<DT_, double>::value)
541 std::cout<<"Warning: You are writing out a banded matrix that is not double precision!\n";
542 this->template _serialize<double, std::uint64_t>(FileMode::fm_bm, file);
543 break;
544
545 default:
546 XABORTM("Filemode not supported!");
547 }
548 }

769
770 Statistics::add_flops(this->val_size() * 2);
771 Statistics::add_time_axpy(ts_stop.elapsed(ts_start));

827 void apply(DenseVector<DT_, IT_>& r, const DenseVector<DT_, IT_>& x) const
828 {
829 XASSERTM(r.size() == this->num_rows(), "Vector size of r does not match!");
830 XASSERTM(x.size() == this->num_cols(), "Vector size of x does not match!");
831
832 if(this->empty())
833 return;
834
835 if (this->num_nzes() == Index(0))
836 {
837 r.format();
838 return;
839 }
840
841 XASSERTM(r.elements_arbiter() != x.elements_arbiter(), "Vector x and r must not share the same memory!");
842
843 TimeStamp ts_start;
844
845 Arch::MatVecMultBandedDense::template exec<DT_, IT_>(
846 r.elements_arbiter(), DT_(1), x.elements_arbiter(), Memory::Arbiter(),
847 this->val_arbiter(), this->offsets_arbiter(), this->num_bands(), this->num_rows(), this->num_cols());
848
849 TimeStamp ts_stop;
850
851 Statistics::add_flops(2 * this->val_size());
852 Statistics::add_time_blas2(ts_stop.elapsed(ts_start));
853 }

896 void apply(DenseVector<DT_, IT_>& r,
897 const DenseVector<DT_, IT_>& x,
898 const DenseVector<DT_, IT_>& y,
899 const DT_ alpha = DT_(1)) const
900 {
901 XASSERTM(r.size() == this->num_rows(), "Vector size of r does not match!");
902 XASSERTM(x.size() == this->num_cols(), "Vector size of x does not match!");
903 XASSERTM(y.size() == this->num_rows(), "Vector size of y does not match!");
904
905 if(this->empty())
906 return;
907
908 XASSERTM(r.elements_arbiter() != x.elements_arbiter(), "Vector x and r must not share the same memory!");
909
910 // we check alpha strictly for equality to 0, because testing if |alpha| < eps may
911 // lead to false triggers if the matrix/vector contents are also < eps
912 if((this->num_nzes() == Index(0)) || (alpha == DT_(0)))
913 {
914 r.copy(y);
915 return;
916 }
917
918 TimeStamp ts_start;
919
920 Arch::MatVecMultBandedDense::template exec<DT_, IT_>(
921 r.elements_arbiter(), alpha, x.elements_arbiter(), y.elements_arbiter(),
922 this->val_arbiter(), this->offsets_arbiter(), this->num_bands(), this->num_rows(), this->num_cols());
923
924 TimeStamp ts_stop;
925
926 Statistics::add_flops( 2 * (this->val_size() + this->num_rows()) );
927 Statistics::add_time_blas2(ts_stop.elapsed(ts_start));
928 }

938 void apply_transposed(DenseVector<DT_, IT_>& r,
939 const DenseVector<DT_, IT_>& x,
940 const DenseVector<DT_, IT_>& y,
941 const DT_ alpha = DT_(1)) const
942 {
943 XASSERTM(r.size() == this->num_cols(), "Vector size of r does not match!");
944 XASSERTM(x.size() == this->num_rows(), "Vector size of x does not match!");
945 XASSERTM(y.size() == this->num_cols(), "Vector size of y does not match!");
946
947 if(this->empty())
948 return;
949
950 XASSERTM(r.elements_arbiter() != x.elements_arbiter(), "Vector x and r must not share the same memory!");
951
952 // we check alpha strictly for equality to 0, because testing if |alpha| < eps may
953 // lead to false triggers if the matrix/vector contents are also < eps
954 if((this->num_nzes() == Index(0)) || (alpha == DT_(0)))
955 {
956 r.copy(y);
957 return;
958 }
959
960 TimeStamp ts_start;
961 Statistics::add_flops( 2 * (this->val_size() + this->num_rows()) );
962
963 Arch::MatVecMultBandedDense::template exec<DT_, IT_>(
964 r.elements_arbiter(), alpha, x.elements_arbiter(), y.elements_arbiter(),
965 this->val_arbiter(), this->offsets_arbiter(), this->num_bands(), this->num_rows(), this->num_cols(), true);
