398 void filter_weak_matrix_rows(SparseMatrixCSR<DT_, IT_> & matrix_a, const SparseMatrixCSR<DT_, IT_>& matrix_m) const
399 {
400 if(_sv.hollow())
401 return;
402
403 XASSERTM(_sv.size() == matrix_a.num_rows(), "Matrix size does not match!");
404 XASSERTM(_sv.size() == matrix_m.num_rows(), "Matrix size does not match!");
405 XASSERTM(matrix_a.num_cols() == matrix_m.num_cols(), "matrix A and M must share their layout");
406 XASSERTM(matrix_a.num_nzes() == matrix_m.num_nzes(), "matrix A and M must share their layout");
407
408 Arch::UnitFilterDenseWeakCSR::template exec<DT_, IT_>(
409 matrix_a.val_arbiter(), matrix_m.val_arbiter(), matrix_a.row_ptr_arbiter(),
410 _sv.elements_arbiter(), _sv.indices_arbiter(), _sv.num_nzes());
411 }
