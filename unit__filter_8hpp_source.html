393 void filter_weak_matrix_rows(SparseMatrixCSR<DT_, IT_> & matrix_a, const SparseMatrixCSR<DT_, IT_>& matrix_m) const
394 {
395 if(_sv.hollow())
396 return;
397
398 XASSERTM(_sv.size() == matrix_a.num_rows(), "Matrix size does not match!");
399 XASSERTM(_sv.size() == matrix_m.num_rows(), "Matrix size does not match!");
400 XASSERTM(matrix_a.num_cols() == matrix_m.num_cols(), "matrix A and M must share their layout");
401 XASSERTM(matrix_a.num_nzes() == matrix_m.num_nzes(), "matrix A and M must share their layout");
402
403 Arch::UnitFilterDenseWeakCSR::template exec<DT_, IT_>(
404 matrix_a.val_arbiter(), matrix_m.val_arbiter(), matrix_a.row_ptr_arbiter(),
405 _sv.elements_arbiter(), _sv.indices_arbiter(), _sv.num_nzes());
406 }
