155 void fill_color(const std::vector<int>& coloring, int hint = -1)
156 {
157 int num_colors = hint;
158 if(hint < 0)
159 {
160 num_colors = *std::max_element(coloring.begin(), coloring.end()) + 1;
161 }
162 // fill tmp vector with coloring
163 std::vector<std::vector<int>> tmp_vector;
164 tmp_vector.resize(Index(num_colors));
165 for(std::size_t i = 0; i < coloring.size(); ++i)
166 {
167 tmp_vector.at(std::size_t(coloring.at(i))).push_back(int(i));
168 }
169
170 _coloring_map = Memory::Arbiter(coloring.size() * sizeof(int));
171 _coloring_map_offsets.resize(std::size_t(num_colors));
172 _coloring_map_sizes.resize(std::size_t(num_colors));
173 Memory::TypedView<int> coloring_map_view(_coloring_map.view(Memory::Location::main, Memory::Access::write));
174 Index offset(0);
175 for(Index i = 0; i < Index(num_colors); ++i)
176 {
177 Memory::memcopy_main(&coloring_map_view.get_w()[offset], tmp_vector.at(i).data(), tmp_vector.at(i).size() * sizeof(int));
178 _coloring_map_offsets.at(i) = offset;
179 _coloring_map_sizes.at(i) = tmp_vector.at(i).size();
180 offset += _coloring_map_sizes.at(i);
181 }
182 }

191 void fill_color(const Coloring& coloring, int hint = -1)
192 {
193 int num_colors = int(coloring.get_num_colors());
194 if(hint >= 0)
195 {
196 ASSERTM(num_colors == hint, "Hint and number of colors do not fit!");
