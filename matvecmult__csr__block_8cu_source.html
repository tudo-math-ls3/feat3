17 template <typename DT_, typename IT_, int block_size_>
18 __global__ void cuda_apply_csrsb(DT_ * r, const DT_ alpha, const DT_ * x, const DT_ * y, const DT_ * val, const IT_ * col_idx, const IT_ * row_ptr, const Index count)
19 {
20 Index idx = threadIdx.x + blockDim.x * blockIdx.x;
21 if (idx >= count)
22 return;
23
24 DT_ bsum[block_size_];
25 for (int j(0) ; j < block_size_ ; ++j)
26 {
27 bsum[j] = DT_(0);
28 }
29 const IT_ end(row_ptr[idx + 1]);
30 for (IT_ i(row_ptr[idx]) ; i < end ; ++i)
31 {
32 const DT_ vali(val[i]);
33 const IT_ coli(col_idx[i] * block_size_);
34 for (int j(0) ; j < block_size_ ; ++j)
35 {
36 bsum[j] += vali * x[coli + j];
37 }
38 }
39
40 if(y != nullptr)
41 {
42 for (int j(0) ; j < block_size_ ; ++j)
43 {
44 r[idx * block_size_ + j] = y[idx * block_size_ + j] + alpha * bsum[j];
45 }
46 }
47 else
48 {
49 for (int j(0) ; j < block_size_ ; ++j)
50 {
51 r[idx * block_size_ + j] = alpha * bsum[j];
52 }
53 }
54 }

56 template <typename DT_, typename IT_, int bs_>
57 void MatVecMultCSRBlock::exec_cuda_impl(DT_ * r, const DT_ alpha, const DT_ * const x, const DT_ * const y, const DT_ * const val, const IT_ * const col_idx, const IT_ * const row_ptr, const Index rows, const Index cols, const Index nonzeros, const bool transposed)
58 {
59 XASSERTM(!transposed, "transposed mat-vec-mult not implemented yet");
60 dim3 block(static_cast<unsigned int>(Util::cuda_blocksize_spmv));
61 dim3 grid((static_cast<unsigned int>(rows) + block.x - 1u) / block.x); // round up to next integer
62
63 cuda_apply_csrsb<DT_, IT_, bs_><<<grid, block>>>(r, alpha, x, y, val, col_idx, row_ptr, rows);
64
65#ifdef DEBUG
66 cudaDeviceSynchronize();
67 cudaError_t last_error(cudaGetLastError());
68 if (cudaSuccess != last_error)
69 throw InternalError(__func__, __FILE__, __LINE__, "CUDA error occurred in execution!\n" + stringify(cudaGetErrorString(last_error)));
70#endif
71 }
