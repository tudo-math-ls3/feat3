104 void set_elements_compression(CompressionModes comp)
105 {
106 CompressionModes temp = comp & CompressionModes::elements_mask;
107 switch(temp)
108 {
109 case CompressionModes::elements_off:
110 break;
111 case CompressionModes::elements_zlib:
112#ifdef FEAT_HAVE_ZLIB
113 break;
114#else
115 XABORTM("Zlib compression was enabled for elements, but zlib bibliography was not loaded. Please configure FEAT with -zlib flag!");
116#endif
117 case CompressionModes::elements_zfp:
118#ifdef FEAT_HAVE_ZFP
119 XASSERTM(tolerance > FEAT::Real(0.), "tolerance ist smaller or equall 0 while zfp is enabled!");
120 break;
121#else
122 XABORTM("Zfp compression was enabled for elements, but zfp bibliography was not loaded. Please configure FEAT with -zfp flag!");
123#endif
124 default:
125 XABORTM("No legal CompressionModes was given");
126 }
127 elements_compression = temp;
128 }
135 void set_indices_compression(CompressionModes comp)
136 {
137 CompressionModes temp = comp & CompressionModes::indices_mask;
138 switch(temp)
139 {
140 case CompressionModes::indices_off:
141 break;
142 case CompressionModes::indices_zlib:
143#ifdef FEAT_HAVE_ZLIB
144 break;
145#else
146 XABORTM("Zlib compression was enabled for indices, but zlib bibliography was not loaded. Please configure FEAT with -zlib flag!");

220 class Container

239 void _copy_content(const Container & other, bool full)
240 {
241 // avoid self-copy
242 if(this == &other)
243 return;
244
245 XASSERTM(_elements.size() == other._elements.size(), "Container size mismatch!");
246 XASSERTM(_indices.size() == other._indices.size(), "Container size mismatch!");
247 XASSERTM(_scalar_index.size() == other._scalar_index.size(), "Container size mismatch!");
248 XASSERTM(_scalar_dt.size() == other._scalar_dt.size(), "Container size mismatch!");
249
250 if (full)
251 {
252 this->_scalar_index.assign(other._scalar_index.begin(), other._scalar_index.end());
253 this->_scalar_dt.assign(other._scalar_dt.begin(), other._scalar_dt.end());
254
255 for (Index i(0) ; i < _indices.size() ; ++i)
256 {
257 XASSERTM(_indices_size.at(i) == other._indices_size.at(i), "Container size mismatch!");
258 _indices.at(i).copy(other._indices.at(i));
259 }
260 }
261
262 for (Index i(0) ; i < _elements.size() ; ++i)
263 {
264 XASSERTM(_elements_size.at(i) == other._elements_size.at(i), "Container size mismatch!");
265 _elements.at(i).copy(other._elements.at(i));
266 }
267
268 }

278 template <typename DT2_, typename IT2_>
279 void assign(const Container<DT2_, IT2_> & other)
280 {
281 this->_elements.clear();
282 this->_indices.clear();
283 this->_elements_size.clear();
284 this->_indices_size.clear();
285 this->_scalar_index.clear();
286 this->_scalar_dt.clear();
287
288 this->_elements_size.assign(other.get_elements_size().begin(), other.get_elements_size().end());
289 this->_indices_size.assign(other.get_indices_size().begin(), other.get_indices_size().end());
290 this->_scalar_index.assign(other.get_scalar_index().begin(), other.get_scalar_index().end());
291 if constexpr (std::is_same<DT_, DT2_>::value)
292 {
293 this->_scalar_dt.assign(other.get_scalar_dt().begin(), other.get_scalar_dt().end());
294 }
295 else
296 {
297 for(auto t : other.get_scalar_dt())
298 {
299 this->_scalar_dt.push_back(DT_(t));
300 }
301 }
302
303 this->_elements.resize(other._elements.size());
304
305 if constexpr (std::is_same<DT_, DT2_>::value)
306 {
307 for (Index i(0) ; i < this->_elements.size() ; ++i)
308 this->_elements.at(i).attach(other._elements.at(i));
309 }
310 else
311 {
312 for (Index i(0) ; i < this->_elements_size.size() ; ++i)
313 {
314 this->_elements.at(i) = Memory::Arbiter(sizeof(DT_) * this->_elements_size.at(i));
315 this->_elements.at(i).template convert<DT_, DT2_>(other._elements.at(i));
316 }
317 }
318
319 this->_indices.resize(other._indices.size());
320
321 if constexpr (std::is_same<IT_, IT2_>::value)
322 {
323 for (Index i(0) ; i < this->_indices.size() ; ++i)
324 this->_indices.at(i).attach(other._indices.at(i));
325 }
326 else
327 {
328 for (Index i(0) ; i < this->_indices_size.size() ; ++i)
329 {
330 this->_indices.at(i) = Memory::Arbiter(sizeof(IT_) * this->_indices_size.at(i));
331 this->_indices.at(i).template convert<IT_, IT2_>(other._indices.at(i));
332 }
333 }

345 template <typename DT2_ = DT_, typename IT2_ = IT_>
346 std::uint64_t _serialized_size(const LAFEM::SerialConfig& config = LAFEM::SerialConfig()) const
347 {
348 Container<DT2_, IT2_> tc;
349 tc.assign(*this);
350
351 std::uint64_t gsize(4 * sizeof(std::uint64_t)); //raw array size + magic number + type_index DT_ + type_index IT_
352 gsize += 7 * sizeof(std::uint64_t); // size of all seven stl containers
353 gsize += 2 * tc._elements_size.size() * sizeof(std::uint64_t); // _elements_size contents + _elements_arrays bytesize
354 gsize += 2 * tc._indices_size.size() * sizeof(std::uint64_t); // _indices_size contents + _indices_arrays bytesize
355 gsize += tc._scalar_index.size() * sizeof(std::uint64_t); // _scalar_index contents
356 gsize += tc._scalar_dt.size() * sizeof(DT2_); // _scalar_dt contents
357
358 CompressionModes compress = config.get_elements_compression() | config.get_indices_compression();
359 Pack::Type compression_type_elements = Pack::deduct_type<DT2_>();
360 Pack::Type compression_type_index = Pack::deduct_type<IT2_>();
361 if((compress & CompressionModes::elements_mask) == CompressionModes::elements_zfp) // If zfp is used for elements
362 {
363 compression_type_elements = compression_type_elements | Pack::Type::Mask_P;
364 }
365 else if((compress & CompressionModes::elements_mask) == CompressionModes::elements_zlib) //If zlib is used and zfp is not used elements
366 {
367 compression_type_elements = compression_type_elements | Pack::Type::Mask_Z;
368 }
369 if((compress & CompressionModes::indices_mask) == CompressionModes::indices_zlib) // If zlib is used for indices
370 compression_type_index = compression_type_index | Pack::Type::Mask_Z;
371
372
373 FEAT::Real tolerance = config.get_tolerance();
374
375 for (Index i(0) ; i < tc._elements_size.size() ; ++i)
376 {
377 gsize += Pack::estimate_size(tc._elements_size.at(i), compression_type_elements, (double)tolerance); // upper_bound for (compressed) elements
378 }
379
380 for (Index i(0) ; i < tc._indices_size.size() ; ++i)
381 {
382 gsize += Pack::estimate_size(tc._indices_size.at(i), compression_type_index); // upper_bound for (compressed) indices
383 }
384 gsize += 16; //padding for datatype alignment mismatch
385
386 return gsize;
387 }

428 template <typename DT2_ = DT_, typename IT2_ = IT_>
429 std::vector<char> _serialize(FileMode mode, const SerialConfig& config = SerialConfig()) const
430 {
431 std::uint64_t raw_size = 0u;
432 FEAT::Real tolerance = config.get_tolerance();
433
434 CompressionModes compress = config.get_elements_compression() | config.get_indices_compression();
435 Pack::Type compression_type_elements = Pack::deduct_type<DT2_>();
436 Pack::Type compression_type_index = Pack::deduct_type<IT2_>();
437 if((compress & CompressionModes::elements_mask) == CompressionModes::elements_zfp) // If zfp is used for elements
438 {
439 compression_type_elements = compression_type_elements | Pack::Type::Mask_P;
440 }
441 else if((compress & CompressionModes::elements_mask) == CompressionModes::elements_zlib) //If zlib is used and zfp is not used elements
442 {
443 compression_type_elements = compression_type_elements | Pack::Type::Mask_Z;
444 }
445 if((compress & CompressionModes::indices_mask) == CompressionModes::indices_zlib) // If zlib is used for indices
446 compression_type_index = compression_type_index | Pack::Type::Mask_Z;
447
448 Container<DT2_, IT2_> tc;
449 tc.assign(*this);
450
451 std::uint64_t gsize = this->template _serialized_size<DT2_, IT2_>(config);
452
453 std::vector<char> result((size_t(gsize)));
454 char * array(result.data());
455 std::uint64_t * uiarray(reinterpret_cast<std::uint64_t *>(array));
456 DT2_ * dtarray(reinterpret_cast<DT2_ *>(array));
457 IT2_ * itarray(reinterpret_cast<IT2_ *>(array));
458
460 #if defined(FEAT_COMPILER_CLANG)
461 std::uint64_t magic = (std::uint64_t)static_cast<__underlying_type(FileMode)>(mode);
462 #else
463 std::uint64_t magic = (std::uint64_t)static_cast<typename std::underlying_type<FileMode>::type>(mode);
464 #endif
465 uiarray[0] = gsize; //we will overwrite this at the end with the finalized data...
466 uiarray[1] = magic;
467 uiarray[2] = Type::Traits<DT_>::feature_hash();
468 uiarray[3] = Type::Traits<IT_>::feature_hash();
469 uiarray[4] = tc._elements.size();
470 uiarray[5] = tc._indices.size();
471 uiarray[6] = tc._elements_size.size();
472 uiarray[7] = tc._indices_size.size();
473 uiarray[8] = tc._scalar_index.size();
474 uiarray[9] = tc._scalar_dt.size();
475 uiarray[10] = (std::uint64_t)compress;
476 raw_size += 11 * (std::uint64_t)sizeof(std::uint64_t);
477
478 Index global_i(11); // count how many elements of std::uint64_t have been inserted so far
479 Index local_elements(0); // count where elements array bytesize starts
480 Index local_index(0); // same for index...
481
482 for (Index i(0) ; i < tc._elements_size.size() ; ++i)
483 {
484 uiarray[i + global_i] = tc._elements_size.at(i);
485 }
486 global_i += Index(tc._elements_size.size());
487 local_elements = global_i;
488 for (Index i(0) ; i < tc._elements_size.size() ; ++i)
489 {
490 uiarray[i + global_i] = (std::uint64_t)0u; //placeholder, until we calculate the real data...
491 }
492 global_i += Index(tc._elements_size.size());
493 raw_size += 2 * std::uint64_t(tc._elements_size.size()) * std::uint64_t(sizeof(std::uint64_t));
494
495 for (Index i(0) ; i < tc._indices_size.size() ; ++i)
496 {
497 uiarray[i + global_i] = tc._indices_size.at(i);
498 }
499 global_i += Index(tc._indices_size.size());
500 local_index = global_i;
501 for (Index i(0) ; i < tc._indices_size.size() ; ++i)
502 {
503 uiarray[i + global_i] = 0;
504 }
505 global_i += Index(tc._indices_size.size());
506 raw_size += 2 * std::uint64_t(tc._indices_size.size()) * std::uint64_t(sizeof(std::uint64_t));
507
508 for (Index i(0) ; i < tc._scalar_index.size() ; ++i)
509 {
510 uiarray[i + global_i] = tc._scalar_index.at(i);
511 }
512 global_i += Index(tc._scalar_index.size());
513 raw_size += std::uint64_t(tc._scalar_index.size()) * std::uint64_t(sizeof(std::uint64_t));
514
515 global_i = Index((global_i * sizeof(std::uint64_t) + sizeof(DT2_) - 1u) / sizeof(DT2_)); // now counting how many DT2 have been inserted so far
516
517 for (Index i(0) ; i < tc._scalar_dt.size() ; ++i)
518 {
519 dtarray[i + global_i] = tc._scalar_dt.at(i);
520 }
521 global_i += Index(tc._scalar_dt.size());
522 raw_size += std::uint64_t(tc._scalar_dt.size()) * std::uint64_t(sizeof(DT2_));
523
524 if((compress & CompressionModes::elements_mask) != CompressionModes::elements_off)
525 {
526 global_i = Index((global_i * sizeof(DT2_) + sizeof(char) - 1u) / sizeof(char)); // now counting how many bytes/chars have been inserted so far
527 for (Index i(0) ; i < tc._elements.size() ; ++i)
528 {
529 Memory::TypedView<DT2_> v(tc._elements.at(i).view(Memory::Location::main, Memory::Access::read));
530 char* dest = &result[global_i];
531 std::size_t est_buff_size = Pack::estimate_size(tc._elements_size.at(i), compression_type_elements, (double)tolerance);
532 std::size_t real_size = Pack::encode<DT2_>(dest, v.get_r(), est_buff_size, tc._elements_size.at(i), compression_type_elements, false, (double)tolerance);
533 XASSERTM(real_size != 0u, "could not encode elements");
534 uiarray[i + local_elements] = (std::uint64_t)real_size; //save used memory for compressed array
535 global_i += (Index)real_size; //go forward the number of bytes(chars)
536 raw_size += (std::uint64_t)real_size * (std::uint64_t)sizeof(char);
537 }
538 global_i = Index((global_i * sizeof(char) + sizeof(DT2_) - 1u) / sizeof(DT2_)); // go back to DT2_ indexing
539 }
540 else
541 {
542 for (Index i(0) ; i < tc._elements.size() ; ++i)
543 {
544 Memory::View v = tc._elements.at(i).view(Memory::Location::main, Memory::Access::read);
545 Memory::memcopy_main(&dtarray[global_i], v.raw_r(), tc._elements_size.at(i) * sizeof(DT2_));
546 uiarray[i + local_elements] = (std::uint64_t) tc._elements_size.at(i) * sizeof(DT2_);
547 global_i += tc._elements_size.at(i);
548 raw_size += (std::uint64_t) tc._elements_size.at(i) * sizeof(DT2_);
549 }
550 }
551 if((compress & CompressionModes::indices_mask) != CompressionModes::indices_off)
552 {
553 global_i = Index((global_i * sizeof(DT2_) + sizeof(char) - 1u) / sizeof(char)); // now counting how many bytes/chars have been inserted so far
554 for (Index i(0) ; i < tc._indices.size() ; ++i)
555 {
556 Memory::TypedView<IT2_> v(tc._indices.at(i).view(Memory::Location::main, Memory::Access::read));
557 char* dest = &result[global_i];
558 std::size_t est_buff_size = Pack::estimate_size(tc._indices_size.at(i), compression_type_index);
559 std::size_t real_size = Pack::encode<IT2_>(dest, v.get_r(), est_buff_size, tc._indices_size.at(i), compression_type_index, false);
560 XASSERTM(real_size != 0u, "could not encode indices");
561 uiarray[i + local_index] = (std::uint64_t) real_size;
562 global_i += Index(real_size);
563 raw_size += std::uint64_t(real_size) * std::uint64_t(sizeof(char));
564 }
565 }
566 else
567 {
568 global_i = Index((global_i * sizeof(DT2_) + sizeof(IT2_) - 1u) / sizeof(IT2_)); // now counting IT2_ elements
569 for (Index i(0) ; i < tc._indices.size() ; ++i)
570 {
571 Memory::View v = tc._indices.at(i).view(Memory::Location::main, Memory::Access::read);
572 Memory::memcopy_main(&itarray[global_i], v.raw_r(), tc._indices_size.at(i) * sizeof(IT2_));
573 uiarray[i + local_index] = (std::uint64_t) tc._indices_size.at(i) * sizeof(IT2_);
574 global_i += tc._indices_size.at(i);
575 raw_size += (std::uint64_t) tc._indices_size.at(i) * sizeof(IT2_);
576 }
577 }
578 uiarray[0] = raw_size + 16u; //padding
579 //std::cout << "Compressed size is: " << uiarray[0] << "\n";
580
581 result.resize(std::size_t(uiarray[0]));
582 return result;
583 }

614 template <typename DT2_ = DT_, typename IT2_ = IT_>
615 void _deserialize(FileMode mode, std::vector<char> & input)
616 {
617 this->clear();
618 Container<DT2_, IT2_> tc;
619 tc.clear();
620
621 char * array(input.data());
622 std::uint64_t * uiarray(reinterpret_cast<std::uint64_t *>(array));
623 DT2_ * dtarray(reinterpret_cast<DT2_ *>(array));
624 IT2_ * itarray(reinterpret_cast<IT2_ *>(array));
625
627#if defined(FEAT_COMPILER_CLANG)
628 std::uint64_t magic = (std::uint64_t)static_cast<__underlying_type(FileMode)>(mode);
629#else
630 std::uint64_t magic = (std::uint64_t)static_cast<typename std::underlying_type<FileMode>::type>(mode);
631#endif
632 XASSERTM(magic == uiarray[1], "_deserialize: given FileMode incompatible with given array!");
633
634 //ensure that we have the same integral/floating type configuration, that was used when storing the serialized data
635 XASSERT(Type::Traits<DT_>::is_int == Type::Helper::extract_intness(uiarray[2]));
636 XASSERT(Type::Traits<DT_>::is_float == Type::Helper::extract_floatness(uiarray[2]));
637 XASSERT(Type::Traits<DT_>::is_signed == Type::Helper::extract_signedness(uiarray[2]));
638 XASSERT(Type::Traits<IT_>::is_int == Type::Helper::extract_intness(uiarray[3]));
639 XASSERT(Type::Traits<IT_>::is_float == Type::Helper::extract_floatness(uiarray[3]));
640 XASSERT(Type::Traits<IT_>::is_signed == Type::Helper::extract_signedness(uiarray[3]));
641
642 if (sizeof(DT_) > Type::Helper::extract_type_size(uiarray[2]))
643 std::cerr<<"Warning: You are reading a container floating point in higher precision then it was saved before!\n";
644
645 if (sizeof(IT_) > Type::Helper::extract_type_size(uiarray[3]))
646 std::cerr<<"Warning: You are reading a container integral type in higher precision then it was saved before!\n";
647
648 CompressionModes compress = (CompressionModes)uiarray[10];
649 //test whether system has right third_party software loaded:
650#ifndef FEAT_HAVE_ZLIB
651 XASSERTM((compress & CompressionModes::elements_mask) != CompressionModes::elements_zlib,
652 "Data was compressed with ZLIB! To read in data, please configure FEAT with zlib enabled.\n For more information see FEAT documentation.");
653 XASSERTM((compress & CompressionModes::indices_mask) != CompressionModes::indices_zlib,
654 "Data was compressed with ZLIB! To read in data, please configure FEAT with zlib enabled.\n For more information see FEAT documentation.");
655#endif
656#ifndef FEAT_HAVE_ZFP
657 XASSERTM((compress & CompressionModes::elements_mask) != CompressionModes::elements_zfp,
658 "Data was compressed with zfp! To read in data, please configure FEAT with zfp enabled.\n For more information see FEAT documentation.");
659#endif
660 std::vector<std::uint64_t> elements_bytes(uiarray[6]); //temp arrays to keep track of bytessize of compressed arrays
661 std::vector<std::uint64_t> indices_bytes(uiarray[7]);
662 Index global_i(11);
663 for (std::uint64_t i(0) ; i < uiarray[6] ; ++i)
664 {
665 tc._elements_size.push_back(Index(uiarray[i + global_i]));
666 }
667 global_i += Index(uiarray[6]);
668 for (std::uint64_t i(0) ; i < uiarray[6] ; ++i)
669 {
670 elements_bytes[i] = uiarray[i + global_i];
671 }
672 global_i += Index(uiarray[6]);
673
674 for (std::uint64_t i(0) ; i < uiarray[7] ; ++i)
675 {
676 tc._indices_size.push_back(Index(uiarray[i + global_i]));
677 }
678 global_i += Index(uiarray[7]);
679 for (std::uint64_t i(0) ; i < uiarray[7] ; ++i)
680 {
681 indices_bytes[i] = uiarray[i + global_i];
682 }
683 global_i += Index(uiarray[7]);
684
685 for (std::uint64_t i(0) ; i < uiarray[8] ; ++i)
686 {
687 tc._scalar_index.push_back(Index(uiarray[i + global_i]));
688 }
689 global_i += Index(uiarray[8]);
690
691 global_i = Index((global_i * sizeof(std::uint64_t) + sizeof(DT2_) - 1u) / sizeof(DT2_));
692 for (std::uint64_t i(0) ; i < uiarray[9] ; ++i)
693 {
694 tc._scalar_dt.push_back(dtarray[i + global_i]);
695 }
696 global_i += Index(uiarray[9]);
697
698 Pack::Type compression_type_elements = Pack::deduct_type<DT2_>();
699 Pack::Type compression_type_index = Pack::deduct_type<IT2_>();
700 if((compress & CompressionModes::elements_mask) == CompressionModes::elements_zfp)// If zfp is used
701 {
702 compression_type_elements = compression_type_elements | Pack::Type::Mask_P;
703 }
704 else if((compress & CompressionModes::elements_mask) == CompressionModes::elements_zlib) //If zlib is loaded and zfp is not used
705 {
706 compression_type_elements = compression_type_elements | Pack::Type::Mask_Z;
707 }
708 if((compress & CompressionModes::indices_mask) == CompressionModes::indices_zlib)
709 {
710 compression_type_index = compression_type_index | Pack::Type::Mask_Z;
711 }
712
713 if((compress & CompressionModes::elements_mask) != CompressionModes::elements_off)
714 {
715 global_i = Index((global_i * sizeof(DT2_) + sizeof(char) - 1u) / sizeof(char));
716 for(Index i(0); i < Index(uiarray[4]); ++i)
717 {
718 tc._elements.push_back(Memory::Arbiter(sizeof(DT2_) * tc._elements_size.at(i)));
719 Memory::TypedView<DT2_> v(tc._elements.back().view(Memory::Location::main, Memory::Access::write));
720 if(0u == Pack::decode(v.get_w(), &array[global_i], (std::size_t) tc._elements_size.at(i), (std::size_t) elements_bytes[i], compression_type_elements, false))

722 global_i += (Index)elements_bytes[i];
723 }
724 global_i = Index((global_i * sizeof(char) + sizeof(DT2_) - 1u) / sizeof(DT2_));
725 }

731 Memory::View v = tc._elements.back().view(Memory::Location::main, Memory::Access::write);
732 v.copy_raw_from(&dtarray[global_i], Memory::Location::main);
733 global_i += tc._elements_size.at(i);
734 }
735 }
736
737 if((compress & CompressionModes::indices_mask) != CompressionModes::indices_off)
738 {
739 global_i = Index((global_i * sizeof(DT2_) + sizeof(char) - 1u) / sizeof(char));
740 for (Index i(0) ; i < Index(uiarray[5]) ; ++i)
741 {
742 tc._indices.push_back(Memory::Arbiter(sizeof(IT2_) * tc._indices_size.at(i)));
743 Memory::TypedView<IT2_> v(tc._indices.back().view(Memory::Location::main, Memory::Access::write));
744 if(0u == Pack::decode(v.get_w(), &array[global_i], (std::size_t) tc._indices_size.at(i), (std::size_t) indices_bytes[i], compression_type_index, false))
745 XABORTM("Cannot decode compressed indice array");
746 global_i += (Index)indices_bytes[i];
747 }
748 }
749 else
750 {
751 global_i = Index((global_i * sizeof(DT2_) + sizeof(IT2_) - 1u) / sizeof(IT2_));
752 for (Index i(0) ; i < Index(uiarray[5]) ; ++i)
753 {
754 tc._indices.push_back(Memory::Arbiter(sizeof(IT2_) * tc._indices_size.at(i)));
755 Memory::View v = tc._indices.back().view(Memory::Location::main, Memory::Access::write);
756 v.copy_raw_from(&itarray[global_i], Memory::Location::main);
757 global_i += tc._indices_size.at(i);
758 }
759 }
760
761 this->assign(tc);

879 void clone(const Container & source, CloneMode clone_mode = CloneMode::Weak)
880 {
881 if (this == &source)
882 {
883 XABORTM("Trying to self-clone a lafem container!");
884 }
885
886 this->clear();
887
888 this->_scalar_index.assign(source._scalar_index.begin(), source._scalar_index.end());
889 this->_scalar_dt.assign(source._scalar_dt.begin(), source._scalar_dt.end());
890 this->_elements_size.assign(source._elements_size.begin(), source._elements_size.end());
891 this->_indices_size.assign(source._indices_size.begin(), source._indices_size.end());
892
893 this->_indices.resize(source._indices.size());
894 this->_elements.resize(source._elements.size());
895
896 // 0 = allocate only, 1 = shallow copy, 2 = deep copy
897 int what_indices = 0;
898 int what_elements = 0;
899 switch(clone_mode)
900 {
901 case CloneMode::Allocate:
902 what_indices = 0;
903 what_elements = 0;
904 break;
905
906 case CloneMode::Layout:
907 what_indices = 1;
908 what_elements = 0;
909 break;
910
911 case CloneMode::Shallow:
912 what_indices = 1;
913 what_elements = 1;
914 break;
915
916 case CloneMode::Weak:
917 what_indices = 1;
918 what_elements = 2;
919 break;
920
921 case CloneMode::Deep:
922 what_indices = 2;
923 what_elements = 2;
924 break;
925 }
926
927 // shallow copy indices if required, otherwise allocate new indices
928 if(what_indices == 1)
929 {
930 for (Index i(0) ; i < source._indices.size() ; ++i)
931 this->_indices.at(i).attach(source._indices.at(i));
932 }
933 else
934 {
935 for (Index i(0) ; i < source._indices.size() ; ++i)
936 this->_indices.at(i) = Memory::Arbiter(this->_indices_size.at(i) * sizeof(IT_));
937 }
938
939 // deep copy indices if required
940 if(what_indices == 2)
941 {
942 for (Index i(0) ; i < source._indices.size() ; ++i)
943 this->_indices.at(i).copy(source._indices.at(i));
944 }
945
946 // shallow copy elements if required, otherwise allocate new elements
947 if(what_elements == 1)
948 {
949 for (Index i(0) ; i < source._elements.size() ; ++i)
950 this->_elements.at(i).attach(source._elements.at(i));
951 }
952 else
953 {
954 for (Index i(0) ; i < source._elements.size() ; ++i)
955 this->_elements.at(i) = Memory::Arbiter(this->_elements_size.at(i) * sizeof(DT_));
956 }
957
958 // deep copy elements if required
959 if(what_elements == 2)
960 {
961 for (Index i(0) ; i < source._elements.size() ; ++i)
962 this->_elements.at(i).copy(source._elements.at(i));
963 }
964 }
