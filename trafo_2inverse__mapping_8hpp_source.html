256 void init_bounding_boxes(DataType bbox_tol)
257 {
258 // get the mesh
259 const auto& mesh = _trafo.get_mesh();
260 // get the vertex set
261 const auto& vtx_set = mesh.get_vertex_set();
262 // get the vertices-at-cell index set
263 const auto& vert_idx = mesh.template get_index_set<shape_dim, 0>();
264
265 // resize bounding boxes array
266 _bboxes.resize(std::size_t(mesh.get_num_elements()));
267
268 // loop over all cells
269 for(Index cell(0); cell < vert_idx.get_num_entities(); ++cell)
270 {
271 // get this cell's vertex indices
272 const auto& vidx = vert_idx[cell];
273
274 // get this cell's bounding box
275 BBox& bbox = _bboxes.at(std::size_t(cell));
276
277 // build cell bounding box
278 bbox[0] = bbox[1] = vtx_set[vidx[0]];
279 for(int i(1); i < vert_idx.get_num_indices(); ++i)
280 {
281 // get vertex and update bounding box
282 const auto& vtx = vtx_set[vidx[i]];
283 for(int j(0); j < world_dim; ++j)
284 {
285 Math::minimax(DataType(vtx[j]), bbox[0][j], bbox[1][j]);
286 }
287 }
288
289 // finally, incorporate tolerances
290 for(int j(0); j < world_dim; ++j)
291 {
292 // compute bbox dimension
293 DataType bb_extra = bbox_tol * (bbox[1][j] - bbox[0][j]);
294 bbox[0][j] -= bb_extra;
295 bbox[1][j] += bb_extra;
296 }
297 }
298 }

336 InvMapDataType unmap_point(const ImagePointType& img_point, bool ignore_failures = false) const
337 {
338 // store image point
339 InvMapDataType inv_data;
340 inv_data.img_point = img_point;
341
342 // find candidate cells
343 std::vector<Index> cells;
344 if(!this->find_candidate_cells(cells, img_point))
345 return inv_data;
346
347 // loop over all candidate cells
348 for(Index cell : cells)
349 {
350 // try to unmap the point by newton
351 DomainPointType dom_point;
352 if(!this->unmap_point_by_newton(dom_point, img_point, cell))
353 {
354 // point unmapping failed!
355 if(ignore_failures)
356 continue;
357
358 // throw an inverse mapping error
359 throw InverseMappingError(dom_point, cell);
360 }
361
362 // check if the domain point is on the reference element
363 if(!this->test_domain_point(dom_point))
364 {
365 // nope ==> skip point
366 continue;
367 }
368
369 // okay
370 inv_data.cells.push_back(cell);
371 inv_data.dom_points.push_back(dom_point);
372 }
373
374 // found at least one cell?
375 return inv_data;
376 }

482 bool find_candidate_cells(std::vector<Index>& cells, const ImagePointType& img_point) const
483 {
484 XASSERTM(_bboxes.size() == _trafo.get_mesh().get_num_elements(), "Boundingbox are not initilialized correctly");
485 // loop over all cells/bounding boxes
486 for(std::size_t cell(0); cell < _bboxes.size(); ++cell)
487 {
488 // point inside bounding box?
489 if(is_in_bounding_box(cell, img_point))
490 {
491 cells.push_back(Index(cell));
492 }
493 }
494
495 return !cells.empty();

498 bool is_in_bounding_box(Index cell, const ImagePointType& img_point) const
499 {
500 const BBox& bbox = _bboxes.at(cell);
501
502 // check whether the point is in the bounding box
503 for(int j(0); j < shape_dim; ++j)
504 {
505 if((img_point[j] < bbox[0][j]) || (img_point[j] > bbox[1][j]))
506 {
507 return false;
508 }
509 }
510
511 return true;

530 bool unmap_point_by_newton(DomainPointType& dom_point, const ImagePointType& img_point, const Index cell) const
531 {
532 // define an evaluator
533 typedef typename TrafoType::template Evaluator<ShapeType, DataType>::Type TrafoEvaluator;
534 TrafoEvaluator trafo_eval(_trafo);
535
536 // for Newton iteration we need image points and inverse jacobians
537 static constexpr TrafoTags trafo_tags = TrafoTags::img_point|TrafoTags::jac_inv;
538
539 // define the trafo eval data
540 typename TrafoEvaluator::template ConfigTraits<trafo_tags>::EvalDataType trafo_data;
541
542 // initialize output domain point to cell centre
543 for(int i(0); i < shape_dim; ++i)
544 dom_point[i] = Shape::ReferenceCell<ShapeType>::template centre<DataType>(i);
545
546 // compute squared tolerance
547 const DataType tol_sqr = Math::sqr(_newton_tol);
548
549 // no convergence yet
550 bool converged = false;
551
552 // prepare trafo evaluator
553 trafo_eval.prepare(cell);
554
555 // newton iteration
556 for(Index iter(0); iter < _newton_max_iter; ++iter)
557 {
558 // evaluate trafo in current point
559 trafo_eval(trafo_data, dom_point);
560
561 // compute image point distance vector
562 ImagePointType def = trafo_data.img_point - img_point;
563
564 // check defect for tolerance
565 if(def.norm_euclid_sqr() < tol_sqr)
566 {
567 converged = true;
568 break;
569 }
570
571 // update reference point by newton
572 dom_point -= trafo_data.jac_inv * def;
573 }
574
575 // finish trafo evaluator
576 trafo_eval.finish();
577
578 // Newton converged?
579 return converged;
580 }

734 InvMapDataType unmap_point(const ImagePointType& img_point, bool ignore_failures = false) const
735 {
736 const Index num_layers = _mappings.size();
737
738 XASSERT(num_layers > 0);
739 XASSERT(_mappings.size() == _coarse_fine_mappings.size() + 1);
740
741 // Double buffered cells and domain points for transferring between layers
742 InvMapDataType current;
743 InvMapDataType next;
744
745 current.img_point = img_point;
746 next.img_point = img_point;
747
748 // Regular inverse mapping on coarsest mesh
749 current = _mappings.front().get().unmap_point(img_point, ignore_failures);
750
751 if(num_layers == 1)
752 {
753 // There is no hierarchy to check. We are done
754 return current;
755 }
756
757 // First layer is coarse layer. Skip it
758 for(Index level(1); level < num_layers; level++)
759 {
760 const CFMappingType& coarse_to_fine = _coarse_fine_mappings[level - 1];
761 const InverseMappingType& fine_inv_mapping = _mappings[level];
762 for(Index cell : current.cells)
763 {
764 const auto fbegin = coarse_to_fine.image_begin(cell);
765 const auto fend = coarse_to_fine.image_end(cell);
766
767 for(auto it = fbegin; it != fend; ++it)
768 {
769 DomainPointType domain_point{};
770 if(try_unmap(fine_inv_mapping, domain_point, img_point, *it, ignore_failures))
771 {
772 next.cells.push_back(*it);
773 next.dom_points.push_back(domain_point);
774 }
775 }
776 }
777
778 // Switch mappings for next layer
779 std::swap(current, next);
780
781 // Empty new fine mappings
782 next.cells.clear();
783 next.dom_points.clear();
784 }
785
786 // current contains result of last iteration
787 return current;
788 }
789 private:
804 static bool try_unmap(
805 const InverseMappingType& mapping,
806 DomainPointType& domain_point,
807 const ImagePointType& img_point,
808 const Index cell,
809 const bool ignore_failures = false)
810 {
811 if(!mapping.is_in_bounding_box(cell, img_point))
812 {
813 return false;
814 }
815
816 if(!mapping.unmap_point_by_newton(domain_point, img_point, cell))
817 {
818 if(!ignore_failures)
819 {
820 throw InverseMappingError(domain_point, cell);
821 }
822 else
823 {
824 return false;
825 }
826 }
827
828 return mapping.test_domain_point(domain_point);
829 }

835 template<int dim_>
836 struct InverseMappingHelper<Shape::Simplex<dim_>>
837 {
838 // checks whether p is on the reference cell
839 template<typename PT_, typename DT_>
840 static bool is_on_ref(const PT_& p, const DT_ tol)
841 {
842 // reference simplex: all coords are in range [0,1]
843 // and the sum of all coords is <= 1
844 DT_ s = DT_(0);
845 for(int i(0); i < dim_; ++i)
846 {
847 // coord outside range [-tol, 1+tol] ?
848 if((p[i] < -tol) || (p[i] > DT_(1)+tol))
849 return false;
850 s += p[i];
851 }
852 // coord sum <= 1+tol ?
853 return (s <= DT_(1) + tol);
854 }
855 };
856
857 template<int dim_>
858 struct InverseMappingHelper<Shape::Hypercube<dim_>>
859 {
860 // checks whether p is on the reference cell
861 template<typename PT_, typename DT_>
862 static bool is_on_ref(const PT_& p, const DT_ tol)
863 {
864 // reference hypercube: all coordinates are in range [-1,+1]
865 for(int i(0); i < dim_; ++i)
866 {
867 // coord outside range [-1-tol, +1+tol] ?
868 if((p[i] < -DT_(1)-tol) || (p[i] > DT_(1)+tol))
869 return false;
870 }
871 return true;
872 }
