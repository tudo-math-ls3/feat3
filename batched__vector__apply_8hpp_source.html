24 // The local dot-product results are passed as a Memory::Arbiter (rather than a raw host pointer), so
25 // the batched kernels acquire the view they need themselves and the arbiter transparently mediates
26 // the host<->device transfers and the on-device accumulation across meta-vector components. The
27 // arbiter must already hold the (zeroed or partially accumulated) values that are added upon.
28 struct BatchedDotProduct
29 {
30 template<typename DT_, typename DTT_>
31 static void exec_generic(Memory::Arbiter& local_dots, const Memory::Arbiter& target, Index dot_size, Memory::Arbiter const* const* others, Index num_dots)
32 {
33 // the generic kernel operates on main memory, so it acquires main-memory views itself
34 Memory::TypedView<DT_> dots_view(local_dots.view(Memory::Location::main, Memory::Access::read_write));
35 DT_* dots_ptr = dots_view.get_w();
36 const Memory::TypedView<DTT_> target_view(target.view(Memory::Location::main, Memory::Access::read));
37 const DTT_* target_ptr = target_view.get_r();
38
39 for(Index l = 0; l < num_dots; ++l)
40 {
41 const Memory::TypedView<DTT_> other_view(others[l]->view(Memory::Location::main, Memory::Access::read));
42 const DTT_* other_ptr = other_view.get_r();
43 DTT_ loc_v = DTT_(0);
44 FEAT_PRAGMA_OMP(parallel for reduction(+:loc_v))
45 for(Index k = 0; k < dot_size; ++k)
46 {
47 loc_v += DT_(target_ptr[k] * other_ptr[k]);
48 }
49 dots_ptr[l] += loc_v;
50 }
51 }
52
53 template<typename DT_, typename DTT_>
54 static void exec_cuda(Memory::Arbiter& local_dots, const Memory::Arbiter& target, Index dot_size, Memory::Arbiter const * const * others, Index num_dots);
55
56 template<typename DT_, typename DTT_>
57 static void exec(Memory::Arbiter& local_dots, const Memory::Arbiter& target, Index dot_size, Memory::Arbiter const * const * others, Index num_dots)
58 {
59 switch(Backend::get_preferred_backend())
60 {
61 case PreferredBackend::cuda:
62 #ifdef FEAT_HAVE_CUDA
63 exec_cuda<DT_, DTT_>(local_dots, target, dot_size, others, num_dots);
64 #else
65 XABORTM("LAFEM::ARCH::BatchedDotProduct::exec: CUDA backend not available");
66 #endif
67 break;
68 default:
69 exec_generic<DT_, DTT_>(local_dots, target, dot_size, others, num_dots);
70 break;
71 }
72 }
73
74 };

96 FEAT_PRAGMA_OMP(parallel for reduction(+:loc_v))
97 for(Index k = 0; k < dot_size; ++k)
98 {
99 loc_v += DT_(target_ptr[k] * freqs_ptr[k] * other_ptr[k]);
100 }
101 dots_ptr[l] += loc_v;
102 }
103 }

129 struct CumulativeAxpy
130 {
131 template<typename DT_>
132 static void exec_generic(Memory::Arbiter& target, Index vec_size, Memory::Arbiter const * const * others, DT_ const * const local_scales, Index num_axpy, DT_ alpha)
133 {
134 // the generic kernel operates on main memory, so it acquires its main-memory views itself
135 Memory::TypedView<DT_> target_view(target.view(Memory::Location::main, Memory::Access::read_write));
136 DT_* target_ptr = target_view.get_w();
137
138 for(Index l = 0; l < num_axpy; ++l)
139 {
140 const Memory::TypedView<DT_> other_view(others[l]->view(Memory::Location::main, Memory::Access::read));
141 const DT_* other_ptr = other_view.get_r();
142 FEAT_PRAGMA_OMP(parallel for)
143 for(Index k = 0; k < vec_size; ++k)
144 {
145 target_ptr[k] += DT_(alpha * local_scales[l]) * other_ptr[k];
146 }
147 }
148 }
