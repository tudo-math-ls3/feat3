441 std::vector<WeightType> gather_element_weights(const MeshType& mesh) const override
442 {
443 const Index num_elems = mesh.get_num_elements();
444
445 // allocate the weight vector
446 std::vector<WeightType> weights(num_elems, 0.0);
447
448 // get the vertex set and the vertices-at-element index set
449 const auto& vtx = mesh.get_vertex_set();
450 const auto& verts_at_elem = mesh.template get_index_set<dim, 0>();
451
452 // element bounding box coordinates
453 Tiny::Vector<CoordType, dim> elbox_min;
454 Tiny::Vector<CoordType, dim> elbox_max;
455
456 // loop over all elements
457 for(Index ielem(0); ielem < num_elems; ++ielem)
458 {
459 // compute bounding box of element
460 // note: actually, we should always have
461 // elbox_min = vtx[verts_at_elem(ielem, 0)]
462 // elbox_max = vtx[verts_at_elem(ielem, 1 << shape_dim - 1)]
463 // because of the element orientation of the base mesh, but we'll compute the bounding
464 // box for each element manually anyways, just to be sure in case that a derived class
465 // might give us a mesh object that is not directly refined from the base mesh using
466 // the standard 2-level refinement...
467 elbox_min = elbox_max = vtx[verts_at_elem(ielem, 0)];
468 for(int j(1); j < verts_at_elem.num_indices; ++j)
469 {
470 const auto& v = vtx[verts_at_elem(ielem, j)];
471 for(int k(0); k < dim; ++k)
472 {
473 Math::minimax(v[k], elbox_min[k], elbox_max[k]);
474 }
475 }
476
477 // gather element weight based on the bounding box
478 weights[ielem] = this->_voxel_map->sample_box(elbox_min, elbox_max);
479 }
480
481 // returns the element weights
482 return weights;
483 }

548 std::vector<Index> gather_elements(const MeshType& mesh, const Dist::Comm& comm) const override
549 {
550 const Index num_elems = mesh.get_num_elements();
551 const auto num_procs = Index(comm.size());
552 const auto cur_rank = Index(comm.rank());
553 //const Index elems_per_rank = (num_elems / num_procs) + Index((num_elems % num_procs) > 0);
554
555 const Index base_elems = num_elems / num_procs;
556 const Index remainder = num_elems % num_procs;
557 // Assign remaining elements to first remainder ranks
558 const Index num_elements = base_elems + Index(cur_rank < remainder);
559 // Offset is base offset, plus number of remainder elements assigned up to this rank
560 const Index offset = (cur_rank * base_elems) + Math::min(cur_rank, remainder);
561
562 // reserve the vector
563 std::vector<Index> masked_elems;
564 masked_elems.reserve(num_elements);
565
566 // get the vertex set and the vertices-at-element index set
567 const auto& vtx = mesh.get_vertex_set();
568 const auto& verts_at_elem = mesh.template get_index_set<dim, 0>();
569
570 // element bounding box coordinates
571 Tiny::Vector<CoordType, dim> elbox_min;
572 Tiny::Vector<CoordType, dim> elbox_max;
573
574 // loop over all elements
575 for(Index ielem = 0; ielem < num_elements; ielem++)
576 {
577 const Index elem = offset + ielem;
578 // compute bounding box of element
579 // note: actually, we should always have
580 // elbox_min = vtx[verts_at_elem(ielem, 0)]
581 // elbox_max = vtx[verts_at_elem(ielem, 1 << shape_dim - 1)]
582 // because of the element orientation of the base mesh, but we'll compute the bounding
583 // box for each element manually anyways, just to be sure in case that a derived class
584 // might give us a mesh object that is not directly refined from the base mesh using
585 // the standard 2-level refinement...
586 elbox_min = elbox_max = vtx[verts_at_elem(elem, 0)];
587 for(int j(1); j < verts_at_elem.num_indices; ++j)
588 {
589 const auto& v = vtx[verts_at_elem(elem, j)];
590 for(int k(0); k < dim; ++k)
591 {
592 Math::minimax(v[k], elbox_min[k], elbox_max[k]);
593 }
594 }
595
596 // check the bounding box against the voxel map
597 if(this->_voxel_map->check_box(elbox_min, elbox_max))
598 {
599 masked_elems.push_back(elem);
600 }
601 }
602
603 // communicate sizes of buffers
604 std::vector<int> recv_sizes(num_procs);
605 recv_sizes.at(cur_rank) = int(masked_elems.size());
606
607 comm.allgather(recv_sizes.data(), 1, recv_sizes.data(), 1);
608
609 // and now, create big buffer
610 std::vector<Index> gathered_mask(std::size_t(std::accumulate(recv_sizes.begin(), recv_sizes.end(), 0)), 0);
611
612 // and also, we have to caclulate our displacements into our big buffer
613 std::vector<int> displacements;
614 displacements.reserve(recv_sizes.size());
615 std::exclusive_scan(recv_sizes.begin(), recv_sizes.end(), std::back_inserter(displacements), 0);
616
617 // and now gather
618 comm.allgatherv(
619 masked_elems.data(),
620 masked_elems.size(),
621 gathered_mask.data(),
622 recv_sizes.data(),
623 displacements.data());
624
625 // return the list of all elements that we found
626 return gathered_mask;
627 }

762 std::vector<WeightType> gather_element_weights(const MeshType& mesh) const override
763 {
764 if(_cheap_weights)
765 {
766 return {};
767 }
768
769 const Index num_elems = mesh.get_num_elements();
770
771 // allocate the weight vector
772 std::vector<WeightType> weights(num_elems, 0.0);
773
774 Cubature::DynamicFactory cub_fac("refine*4:gauss-legendre:3");
775 Cubature::Rule<typename MeshType::ShapeType> rule;
776 cub_fac.create(rule);
777
778 // guarantee that cgal wrapper is initialized
779 this->_cgal_wrapper->point_not_outside(CoordType(0), CoordType(0), CoordType(0));
780
781 // create trafo
782 using TrafoType = Trafo::Standard::Mapping<MeshType>;
783 using TrafoEvaluator = typename TrafoType::template Evaluator<typename MeshType::ShapeType, CoordType>::Type;
784 static constexpr TrafoTags trafo_config = TrafoTags::dom_point | TrafoTags::img_point | TrafoTags::jac_det;
785
787 using TrafoEvalData = typename TrafoEvaluator::template ConfigTraits<trafo_config>::EvalDataType;
788
789 const TrafoType trafo(const_cast<MeshType&>(mesh)); //NOLINT
790
791 FEAT_PRAGMA_OMP(parallel)
792 {
793 TrafoEvaluator trafo_eval(trafo);
794 TrafoEvalData trafo_data;
795
796 // loop over all elements
797 FEAT_PRAGMA_OMP(for)
798 for(Index ielem = 0; ielem < num_elems; ++ielem)
799 {
800 trafo_eval.prepare(ielem);
801 CoordType loc_sum = CoordType(0);
802 for(int k = 0; k < rule.get_num_points(); ++k)
803 {
804 const auto ref_point = rule.get_point(k);
805 trafo_eval(trafo_data, ref_point);
806 const auto& img_point = trafo_data.img_point;
807 const CoordType weight = trafo_data.jac_det * rule.get_weight(k);
808 loc_sum +=
809 weight * CoordType(this->_cgal_wrapper->point_not_outside(img_point[0], img_point[1], img_point[2]));
810 }
811 loc_sum /= trafo_eval.volume();
812 ASSERTM(loc_sum >= CoordType(0) && loc_sum <= CoordType(1), "Local sum not between zero and one");
813
814 // gather element weight based on the bounding box
815 weights[ielem] = loc_sum;
816 }
817 }
818
819 // returns the element weights
820 return weights;
821 }

823 template<typename VtxSet_, typename IdxMap_>
824 std::vector<Index> _offset_gather_mask_omp(const VtxSet_& vtx, const IdxMap_& verts_at_elem, Index offset, Index num_elems) const
825 {
826 // reserve the vector
827 std::vector<std::uint8_t> masked_elems;
828 masked_elems.resize(num_elems);
829
830 FEAT_PRAGMA_OMP(parallel for)
831 for(std::size_t k = 0; k < num_elems; ++k)
832 {
833 masked_elems[k] = 0u;
834 }
835
836 std::size_t num_mask = 0u;
837 // loop over all elements
838 FEAT_PRAGMA_OMP(parallel for reduction(+ : num_mask))
839 for(Index ielem = 0; ielem < num_elems; ++ielem)
840 {
841 std::array<Tiny::Vector<CoordType, dim>, std::decay_t<decltype(verts_at_elem)>::num_indices> verts;
842 for(std::size_t k = 0; k < verts.size(); ++k)
843 {
844 verts[k] = vtx[verts_at_elem(ielem+offset, int(k))];
845 }
846
847 if(this->_cgal_wrapper->intersects_polygon(verts))
848 {
849 masked_elems.at(ielem) = 1;
850 ++num_mask;
851 }
852 }
853
854 std::vector<Index> masked_elems_gathered;
855 masked_elems_gathered.reserve(num_mask);
856
857 std::size_t ctr_num = 0;
858 for(Index ielem = 0; ielem < num_elems; ++ielem)
859 {
860 if(masked_elems[ielem] == 1)
861 {
862 ++ctr_num;
863 masked_elems_gathered.push_back(ielem+offset);
864 }
865 }
866
867 XASSERTM(ctr_num == num_mask, "Misscount in gather loop");
868
869 // return the list of all elements that we found
870 return masked_elems_gathered;
871

1563 virtual void _deslag_patch_halos(
1564 MeshNodeType& patch_mesh_node,
1565 const MeshNodeType& base_mesh_node,
1566 const Ancestor& ancestor,
1567 bool is_child)
1568 {
1569 // get the map of the base-mesh halos
1570 const std::map<int, std::unique_ptr<MeshPartType>>& base_halo_map = base_mesh_node.get_halo_map();
1571
1572 // if the base mesh has no halos, then we can jump out of here
1573 if(base_halo_map.empty())
1574 {
1575 return;
1576 }
1577
1578 // get number of halos
1579 const std::size_t num_halos = base_halo_map.size();
1580
1581 // create a halo splitter
1582 Geometry::PatchHaloSplitter<MeshType> halo_splitter(*base_mesh_node.get_mesh(), *base_mesh_node.get_patch(-1));
1583
1584 // add each base-mesh halo to our halo splitter and store the resulting split data size
1585 std::vector<int> halo_ranks;
1586 std::vector<std::size_t> halo_send_sizes;
1587 for(auto it = base_halo_map.begin(); it != base_halo_map.end(); ++it)
1588 {
1589 // store halo rank
1590 halo_ranks.push_back(it->first);
1591
1592 // add halo and compute send buffer size
1593 halo_send_sizes.push_back(halo_splitter.add_halo(it->first, *it->second));
1594 }
1595
1596 // This vector will receive the split halo data from all our potential neighbor processes
1597 std::vector<std::size_t> halo_recv_sizes(num_halos);
1598 std::vector<std::vector<Index>> halo_send_data(num_halos);
1599 std::vector<std::vector<Index>> halo_recv_data(num_halos);
1600 Dist::RequestVector halo_recv_reqs(num_halos);
1601 Dist::RequestVector halo_send_reqs(num_halos);
1602
1603 // get the layer index of our comm layer; the ancestor layer unless we got the child ancestry during
1604 // multi-layered partitioning; this may be equal to -1 in either case if this process does not participate
1605 // in that corresponding layer
1606 const int layer_idx = is_child ? ancestor.layer_p : ancestor.layer;
1607
1608 // split and serialize halos and store sizes
1609 for(std::size_t i(0); i < num_halos; ++i)
1610 {
1611 // serialize the split halo into the send data buffer
1612 if(halo_send_sizes.at(i) > std::size_t(0))
1613 {
1614 halo_send_data.at(i) = halo_splitter.serialize_split_halo(halo_ranks[i], -1);
1615 }
1616 }
1617
1618 // exchange halo send data sizes over the corresponding layer communicator
1619 if(layer_idx >= 0)
1620 {
1621 // get the corresponding layer communicator
1622 const Dist::Comm& layer_comm = this->_layers.at(std::size_t(layer_idx))->comm();
1623
1624 // exchange halos sizes
1625 for(std::size_t i(0); i < num_halos; ++i)
1626 {
1627 // post receive requests
1628 halo_recv_reqs[i] = layer_comm.irecv(&halo_recv_sizes[i], std::size_t(1), halo_ranks[i]);
1629
1630 // post send requests
1631 halo_send_reqs[i] = layer_comm.isend(&halo_send_sizes[i], std::size_t(1), halo_ranks[i]);
1632 }
1633
1634 // wait for sends and receives to finish
1635 halo_recv_reqs.wait_all();
1636 halo_send_reqs.wait_all();
1637
1638 // exchange halo data
1639 for(std::size_t i(0); i < num_halos; ++i)
1640 {
1641 // do we receive any data from our neighbor?
1642 if(halo_recv_sizes[i] > Index(0))
1643 {
1644 // resize buffer and post receive
1645 halo_recv_data.at(i).resize(halo_recv_sizes[i]);
1646 halo_recv_reqs[i] = layer_comm.irecv(halo_recv_data.at(i).data(), halo_recv_sizes.at(i), halo_ranks[i]);
1647 }
1648
1649 // do we send any data to our neighbor?
1650 if(halo_send_sizes.at(i) > Index(0))
1651 {
1652 // post send of actual halo buffer
1653 halo_send_reqs[i] = layer_comm.isend(halo_send_data.at(i).data(), halo_send_sizes.at(i), halo_ranks[i]);
1654 }
1655 }
1656
1657 // wait for sends and receives to finish
1658 halo_recv_reqs.wait_all();
1659 halo_send_reqs.wait_all();
1660 }
1661
1662 // broadcast halo receive data over progeny comm in case of multi-layered partitioning
1663 if(is_child)
1664 {
1665 // broadcast sizes
1666 ancestor.progeny_comm.bcast(halo_recv_sizes.data(), num_halos, 0);
1667
1668 // allocate buffers
1669 if(ancestor.progeny_comm.rank() != 0)
1670 {
1671 for(std::size_t i(0); i < num_halos; ++i)
1672 {
1673 halo_recv_data.at(i).resize(halo_recv_sizes[i]);
1674 }
1675 }
1676
1677 // broadcast all halos
1678 for(std::size_t i(0); i < num_halos; ++i)
1679 {
1680 if(halo_recv_sizes[i] > std::size_t(0))
1681 {
1682 ancestor.progeny_comm.bcast(halo_recv_data[i].data(), halo_recv_sizes[i], 0);
1683 }
1684 }
1685 }
1686
1687 /*{
1688 String s;
1689 for(std::size_t i(0); i < num_halos; ++i)
1690 {
1691 s += stringify(halo_ranks[i]);
1692 s += " >";
1693 for(Index j(halo_recv_data[i]); j < halo_recv_data[i+1]; ++j)
1694 (s += " ") += stringify(halo_recv_data[j]);
1695 s += "\n";
1696 }
1697 this->_comm.allprint(s);
1698 }*/
1699
1700 // create a vector of split halo sizes
1701 std::vector<std::array<Index, shape_dim + 1>> halo_send_intsec_sizes(num_halos);
1702 std::vector<std::array<Index, shape_dim + 1>> halo_recv_intsec_sizes(num_halos);
1703
1704 // process all halos
1705 for(std::size_t i(0); i < num_halos; ++i)
1706 {
1707 // did we receive any data from our neighbor?
1708 if(halo_recv_sizes.at(i) == Index(0))
1709 {
1710 continue;
1711 }
1712
1713 // intersect with our other halo
1714 if(!halo_splitter.intersect_split_halo(halo_ranks[i], halo_recv_data.at(i), 0))
1715 {
1716 continue; // no intersection between halos
1717 }
1718
1719 // create the new halo
1720 std::unique_ptr<MeshPartType> split_halo = halo_splitter.make_unique();
1721
1722 // store the entity counts of the halo in our split sizes vector
1723 for(int j(0); j <= shape_dim; ++j)
1724 {
1725 halo_send_intsec_sizes[i][Index(j)] = split_halo->get_num_entities(j);
1726 }
1727
1728 // create new halo mesh-part
1729 patch_mesh_node.add_halo(halo_ranks[i], std::move(split_halo));
1730 }
1731
1732 // exchange halo intersected halo sizes over progeny comm; we do this just to ensure consistency
1733 if((ancestor.layer_p >= 0) || (!is_child && (ancestor.layer >= 0)))
1734 {
1735 // get the corresponding layer communicator
1736 const LayerType& layer = *this->_layers.at(std::size_t(is_child ? ancestor.layer_p : ancestor.layer));
1737
1738 // exchange intersected halo sizes
1739 for(std::size_t i(0); i < num_halos; ++i)
