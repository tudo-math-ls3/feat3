35 static std::pair<CoordVecType, CoordVecType> calculate_global_boundingbox(const DomainControl_& domain, const Index level = Index(0))
36 {
37 XASSERTM(domain.size_virtual() >= level, "Chosen level is larger than virtual size");
38 // calculate local boundingbox for all relevant domain level
39 CoordVecType bb_min(Math::Limits<DataType>::max()), bb_max(Math::Limits<DataType>::lowest());
40 if(level < domain.size_physical())
41 {
42 // calculate local boundingbox
43 const auto& mesh = domain.at(level)->get_mesh();
44 const auto& vtx_set = mesh.get_vertex_set();
45
46 const Index n = vtx_set.get_num_vertices();
47 for(Index i(0); i < n; ++i)
48 {
49 for(int k = 0; k < dim; ++k)
50 {
51 bb_min[k] = Math::min(bb_min[k], vtx_set[i][k]);
52 bb_max[k] = Math::max(bb_max[k], vtx_set[i][k]);
53 }
54 }
55 }
56 // and now do a communication over all processes
57 const Dist::Comm& comm = domain.comm();
58 comm.allreduce(static_cast<DataType*>(&bb_min[0]), static_cast<DataType*>(&bb_min[0]), std::size_t(dim), Dist::op_min);
59 comm.allreduce(static_cast<DataType*>(&bb_max[0]), static_cast<DataType*>(&bb_max[0]), std::size_t(dim), Dist::op_max);
60
61 return std::make_pair(bb_min, bb_max);
62 }
63 };
