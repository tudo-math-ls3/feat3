24 // Architecture Notes. Read class documentation of ArgParser first.
25 // The core of this argument parser implementation consists of four classes
26 // - ArgParser is the actual parser
27 // - Parameter is the user-facing interface of a parameter
28 // - ParameterCore is the inner interface of a parameter used by the ArgParser
29 // - ParameterBuilder is a class implementing the builder pattern to define parameters
30 //
31 // The split into Parameter and ParameterCore is a matter of type-erasure.
32 // We want to both present a neat and well-typed interface to the user and
33 // automatically handle the parameters in the parser.
34 // But the parser can not handle the templated Parameters directly, because
35 // they form a non-homogenous collection. To solve this the Parameter class
36 // is just a thin wrapper around a type-erased ParameterCore.
37 // The parser can handle the ParameterCores, and the Parameter presents a
38 // well-typed interface.
39 //
40 // To achieve this type-erasure the ParameterCore stores its parsed values
41 // in a std::any (essentially a void* and a type tag). Any type dependent
42 // logic required by the ParameterCore (parsing and validation, for now)
43 // is handled by supplying functions that accept std::any to the ParameterCore.
44 // These functions are constructed during the parameter definition
45 // (when type information is still available), by wrapping the properly
46 // typed functions into std::any_casts.
47 //
48 // Thus the ArgParser holds a list of shared pointers to ParameterCores and has
49 // the Parameters as members. The parameters hold a shared pointer to their
50 // respective ParameterCore as well. The ArgParser then uses its list of
51 // parameter cores to match arguments to parameters during parsing.
52 //
53 // Start reading at ArgParser::parameter to see how parameters are created and defined.
54 // Start reading at ArgParser::parse to see how parsing is implemented.

117 template<typename T_>
118 static void parse_deque(std::any& any, const String& s)
119 {
120 std::deque<T_> collection;
121 for(const String& value_str : s.split_by_whitespaces())
122 {
123 T_ tmp{};
124 if(!value_str.parse(tmp))
125 {
126 throw std::invalid_argument("Parsing failed for string '" + s + "'");
127 }
128 collection.push_back(tmp);
129 }
130
131 any = collection;
132 }
133
135 template<typename K_, typename V_>
136 static void parse_map(std::any& any, const String& s)
137 {
138 std::map<K_, V_> map;
139 auto strs = s.split_by_whitespaces();
140
141 if(!(strs.size() % 2 == 0))
142 {
143 throw std::invalid_argument("Parsing failed for string '" + s + "': Odd number of strings in key-value pairs.");
144 }
145
146 for(Index i(0); i < strs.size(); i += 2)
147 {
148 K_ key{};
149 V_ value{};
150
151 if(!strs[i].parse(key))
152 {
153 throw std::invalid_argument("Parsing failed for string '" + s + "': Could not parse key '" + strs[i] + "'");
154 }
155
156 if(!strs[i + 1].parse(value))
157 {
158 throw std::invalid_argument("Parsing failed for string '" + s + "': Could not parse value '" + strs[i] + "'");
159 }
160
161 map.insert({key, value});
162 }
163
164 any = map;
165 }

196 template<typename T_>
197 static void (*make_parser())(std::any&, const String&)
198 {
199 // If T_ has no input operator then instantiating a parser function for T_ will fail,
200 // because String::parse depends on that operator.
201 // But we want to give the user a chance to set a custom parser,
202 // so we guard the instantiation of the default parser behind
203 // can_default_parse_v
204
205 if constexpr(!can_default_parse_v<T_>)
206 {
207 return nullptr;
208 }
209 else
210 {
211 if constexpr(is_std_deque_v<T_>)
212 {
213 return parse_deque<typename T_::value_type>;
214 }
215 else if constexpr(is_std_map_v<T_>)
216 {
217 return parse_map<typename T_::key_type, typename T_::mapped_type>;
218 }
219 else
220 {
221 return parse_single<T_>;
222 }
223 }
224 }
225
236 template<typename T_>
237 static void (*make_printer())(std::ostream&, const std::any&)
238 {
239 if constexpr(is_std_deque_v<T_>)
240 {
241 return [](std::ostream& s, const std::any& v)
242 {
243 s << "[";
244 for(const auto& i : std::any_cast<const T_&>(v))
245 {
246 s << i << ", ";
247 }
248 s << "]";
249 };
250 }
251 else if constexpr(is_std_map_v<T_>)
252 {
253 return [](std::ostream& s, const std::any& v)
254 {
255 s << "{";
256 for(const auto& [key, value] : std::any_cast<const T_&>(v))
257 {
258 s << key << ": " << value << ", ";
259 }
260 s << "}";
261 };
262 }
263 else
264 {
265 return [](std::ostream& s, const std::any& v) { s << std::any_cast<const T_&>(v); };
266 }

907 template<typename U_>
908 ParameterBuilder& derives_from(const Parameter<U_>& param, T_ (*fn)(const U_&))
909 {
910 XASSERTM(param._core != nullptr, "Derivee needs to be initialized");
911
912 // NOTE(mmuegge): It is totally possible to abstract over derives_from_1
913 // and derives_from_2: The easiest way would probably be to convert the
914 // std::deque passed into the deriver into a tuple of the same length as
915 // the number of parameters this parameter is derived from, then cast the
916 // tuple to the proper types via a fold-expression, then use std::apply to
917 // call fn with the arguments from the tuple. All of that code is
918 // possible, but just plain ugly and hard to understand. So I choose the
919 // easy route for now.
920
921 _core->derives_from.clear();
922 _core->derives_from.emplace_back(param._core);
923 _core->deriver = [fn](std::any& out, const std::deque<std::reference_wrapper<std::any>>& args)
924 {
925 XASSERT(args.size() >= 1);
926 out = fn(std::any_cast<const U_&>(args[0]));
927 };
928
929 return *this;
930 }

1427 std::make_shared<Intern::ParameterCore>(std::forward<T_>(default_value));
1428
1429 // Store type-erased core for parsing purposes.
1430 // This allows the parser to treat all parameters the same,
1431 // irrespective of the types of their values.
1432 _parameters.push_back(core);
1433
1434 // Return builder to user for defining the parameter fully
1435 return ParameterBuilder<T_>(std::move(core), _env_prefix, _property_prefix, _flag_prefix);
1436 }

1458 template<typename Parser_>
1459 Parser_ parser(String env_prefix = "", String property_path_prefix = "", String flag_prefix = "")
1460 {
1461 // NOTE: On the implementation of this function. Rather than parsing fully
1462 // recursively, and having to deal with tracking of used arguments and
1463 // error handling across parsers, we just add all parameters of the
1464 // sub-parser (with their prefix) to this parser. The parse method of this
1465 // parser will then parse the parameters the same way it parses its own
1466 // parameters. This might be terribly inconvenient later, but is very easy
1467 // to implement for now.
1468 const auto append = [](const String& a, const String& b, char delimiter) -> String
1469 {
1470 if(a.empty())
1471 {
1472 return b;
1473 }
1474 else if(!b.empty())
1475 {
1476 return a + delimiter + b;
1477 }
1478 else
1479 {
1480 return "";
1481 }
1482 };
1483
1484 // Create sub-parser
1485 Parser_ parser(
1486 append(_env_prefix, env_prefix, '_'),
1487 append(_property_prefix, property_path_prefix, '/'),
1488 append(_flag_prefix, flag_prefix, '-'));
1489
1490 // Add sub-parsers parameters to our own
1491 _parameters.insert(_parameters.end(), parser._parameters.begin(), parser._parameters.end());
1492
1493 // Return parser to store it in current parser
1494 return parser;
1495 }
