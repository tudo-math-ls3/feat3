665 template<typename DT2_, typename IT2_>
666 Index mask_scatter(const DenseVector<DT2_, IT2_>& vector, std::vector<int>& mask,
667 const int value, const Index offset = Index(0)) const
668 {
669 XASSERT(Index(mask.size()) >= vector.size());
670 XASSERT(Index(mask.size()) >= this->size() + offset);
671 const Index n = this->num_indices();
672 const Memory::TypedView<IT_> idx = this->indices_view_r();
673
674 for(Index i(0); i < n; ++i)
675 mask[offset + idx[i]] = value;
676
677 return vector.size();
678 }

724 template<typename DT2_, typename IT2_, int block_size_>
725 Index mask_scatter_raw(const DenseVectorBlocked<DT2_, IT2_, block_size_>& vector, std::vector<int>& mask,
726 const int value, const Index offset = Index(0)) const
727 {
728 XASSERT(Index(mask.size()) >= vector.size());
729 const Index n = this->num_indices();
730 const Memory::TypedView<IT_> idx = this->indices_view_r();
731
732 XASSERT(Index(mask.size()) >= Index(block_size_)*this->size() + offset);
733 for(Index i(0); i < n; ++i)
734 {
735 const Index ibs = idx[i] * Index(block_size_);
736 for(int k(0); k < block_size_; ++k)
737 mask[offset + ibs + Index(k)] = value;
738 }
739 return vector.size();
740 }
