656 template<typename DT2_, typename IT2_>
657 Index mask_scatter(const DenseVector<DT2_, IT2_>& vector, std::vector<int>& mask,
658 const int value, const Index offset = Index(0)) const
659 {
660 XASSERT(Index(mask.size()) >= vector.size());
661 XASSERT(Index(mask.size()) >= this->size() + offset);
662 const Index n = this->num_indices();
663 const Memory::TypedView<IT_> idx = this->indices_view_r();
664
665 for(Index i(0); i < n; ++i)
666 mask[offset + idx[i]] = value;
667
668 return vector.size();
669 }

715 template<typename DT2_, typename IT2_, int block_size_>
716 Index mask_scatter_raw(const DenseVectorBlocked<DT2_, IT2_, block_size_>& vector, std::vector<int>& mask,
717 const int value, const Index offset = Index(0)) const
718 {
719 XASSERT(Index(mask.size()) >= vector.size());
720 const Index n = this->num_indices();
721 const Memory::TypedView<IT_> idx = this->indices_view_r();
722
723 XASSERT(Index(mask.size()) >= Index(block_size_)*this->size() + offset);
724 for(Index i(0); i < n; ++i)
725 {
726 const Index ibs = idx[i] * Index(block_size_);
727 for(int k(0); k < block_size_; ++k)
728 mask[offset + ibs + Index(k)] = value;
729 }
730 return vector.size();
731 }
