20 template <typename DT_, typename IT_>
21 __global__ void cuda_apply_bcsr(DT_ * r, const DT_ alpha, const DT_ * x, const DT_ * val, const IT_ * col_idx,
22 const IT_ * row_ptr, const Index count, const int block_height, const int block_width)
23 {
24 // grid strided for loop
25 for(Index idx = threadIdx.x + blockDim.x * blockIdx.x; idx < count; idx += blockDim.x * gridDim.x)
26 {
27 /// \todo remove hardcoded number
28 DT_ bsum[10];
29 for (int j(0) ; j < block_height ; ++j)
30 {
31 bsum[j] = DT_(0);
32 }
33 const IT_ end(row_ptr[idx + 1]);
34 for (IT_ i(row_ptr[idx]) ; i < end ; ++i)
35 {
36 for (int h(0) ; h < block_height ; ++h)
37 {
38 for (int w(0) ; w < block_width ; ++w)
39 {
40 bsum[h] += val[i * block_height * block_width + h * block_width + w] * x[col_idx[i] * block_width + w];
41 }
42 }
43 }
44 for (int j(0) ; j < block_height ; ++j)
45 {
46 r[idx * block_height + j] += alpha * bsum[j];
47 }
48 }
49 }

51 template <typename DT_, typename IT_>
52 void bcsr_intern_cuda(DT_ * r, const DT_ alpha, const DT_ * const x, const DT_ * const y, const DT_ * const val,
53 const IT_ * const col_idx, const IT_ * const row_ptr, const Index rows, const Index cols, const Index /*nonzeros*/,
54 const int block_height, const int block_width, const bool transposed)
55 {
56 dim3 block(static_cast<unsigned int>(Util::cuda_blocksize_spmv));
57 dim3 grid((static_cast<unsigned int>(rows) + block.x - 1u) / block.x); // round up to next integer
58
59 Index rsize = (transposed ? cols*Index(block_width) : rows*Index(block_height));
60 if(y == nullptr)
61 Memory::memset_cuda(r, 0, rsize * sizeof(DT_));
62 else if (r != y)
63 Memory::memcopy_cuda(r, y, rsize * sizeof(DT_));
64
65 XASSERTM(!transposed, "transposed BCSR matvec for CUDA not implemented");
66 XASSERT(block_height < 10); // see cuda_apply_bcsr implementation
67
68 cuda_apply_bcsr<DT_, IT_><<<grid, block>>>(r, alpha, x, val, col_idx, row_ptr, rows, block_height, block_width);
69
70#ifdef DEBUG
71 cudaDeviceSynchronize();
72 cudaError_t last_error(cudaGetLastError());
73 if (cudaSuccess != last_error)
74 throw InternalError(__func__, __FILE__, __LINE__, "CUDA error occurred in execution!\n" + stringify(cudaGetErrorString(last_error)));
75#endif
76 }

109 template <typename DT_, typename IT_>
110 void bcsr_intern_cusparse(DT_ * r, const DT_ alpha, const DT_ * const x, const DT_ * const y, const DT_ * const val, const IT_ * const col_idx, const IT_ * const row_ptr, const Index rows, const Index cols, const Index nonzeros, const int block_size, const bool transposed)
111 {
112 Index rsize = (transposed ? cols : rows) * Index(block_size);
113 if(y == nullptr)
114 Memory::memset_cuda(r, 0, rsize * sizeof(DT_));
115 else if (r != y)
116 Memory::memcopy_cuda(r, y, rsize * sizeof(DT_));
117
118 cusparseMatDescr_t descr=0;
119 cusparseCreateMatDescr(&descr);
120 cusparseSetMatType(descr, CUSPARSE_MATRIX_TYPE_GENERAL);
121 cusparseSetMatIndexBase(descr, CUSPARSE_INDEX_BASE_ZERO);
122
123 cusparseOperation_t trans = (transposed ? CUSPARSE_OPERATION_TRANSPOSE : CUSPARSE_OPERATION_NON_TRANSPOSE);
124
125 DT_ b = DT_(y == nullptr ? 0 : 1);
126 cusparse_apply_bcsr(CUSPARSE_DIRECTION_ROW, trans, (int)rows, (int)cols, (int)nonzeros, &alpha, descr, val, (int*)row_ptr, (int*)col_idx,
127 block_size, x, &b, r);
128
129 cusparseDestroyMatDescr(descr);
130
131#ifdef DEBUG
132 cudaDeviceSynchronize();
133 cudaError_t last_error(cudaGetLastError());
134 if (cudaSuccess != last_error)
135 throw InternalError(__func__, __FILE__, __LINE__, "CUDA error occurred in execution!\n" + stringify(cudaGetErrorString(last_error)));
136#endif
137 }

140 template <typename DT_, typename IT_>
141 void MatVecMultBCSRDense::exec_cuda_impl(DT_ * r, const DT_ alpha, const DT_ * const x, const DT_ * const y, const DT_ * const val, const IT_ * const col_idx, const IT_ * const row_ptr, const Index rows, const Index cols, const Index nonzeros, const int block_height, const int block_width, const bool transposed)
142 {
143 if((block_height == 1) && (block_width == 1))
144 {
145 MatVecMultCSRDense::exec_cuda_impl<DT_, IT_>(r, alpha, x, y, val, col_idx, row_ptr, rows, cols, nonzeros, transposed);
146 }
147 else if((sizeof(IT_) == 4u) && (block_height == block_width))
148 {
149 bcsr_intern_cusparse<DT_, IT_>(r, alpha, x, y, val, col_idx, row_ptr, rows, cols, nonzeros, block_height, transposed);
150 }
151 else
152 {
153 bcsr_intern_cuda<DT_, IT_>(r, alpha, x, y, val, col_idx, row_ptr, rows, cols, nonzeros, block_height, block_width, transposed);
154 }
155 }
