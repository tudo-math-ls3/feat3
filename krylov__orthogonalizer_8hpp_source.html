144 template<typename Vector_>
145 class ModifiedGramSchmidtOrthogonalizer : public KrylovSubspaceOrthogonalizer<Vector_>
146 {
147 public:
148 typedef KrylovSubspaceOrthogonalizer<Vector_> BaseClass;
149 typedef typename BaseClass::VectorType VectorType;
150 typedef typename BaseClass::DataType DataType;
151
152 public:
153 explicit ModifiedGramSchmidtOrthogonalizer() = default;
154
155 ModifiedGramSchmidtOrthogonalizer(const Vector_& vec_ref, Index krylov_dimension) :
156 BaseClass(vec_ref, krylov_dimension)
157 {
158 }
159
160 ModifiedGramSchmidtOrthogonalizer(const ModifiedGramSchmidtOrthogonalizer&) = delete;
161 ModifiedGramSchmidtOrthogonalizer& operator=(const ModifiedGramSchmidtOrthogonalizer&) = delete;
162 ModifiedGramSchmidtOrthogonalizer(ModifiedGramSchmidtOrthogonalizer&&) = default;
163 ModifiedGramSchmidtOrthogonalizer& operator=(ModifiedGramSchmidtOrthogonalizer&&) = default;
164
165 virtual ~ModifiedGramSchmidtOrthogonalizer()=default;
166
167 DataType orthogonalize(DataType* dot_products, const Index target_index) override
168 {
169 XASSERT(target_index <= this->_krylov_dimension);
170 // Gram-Schmidt process
171 for(Index k(0); k < target_index; ++k)
172 {
173 const auto dot = this->_subspace[target_index].dot(this->_subspace[k]);
174 this->_subspace[target_index].axpy(this->_subspace[k], -dot);
175 dot_products[k] = dot;
176 }
177 return this->_subspace[target_index].norm2();
178 }

196 template<typename Vector_>
197 class ClassicGramSchmidtOrthogonalizer : public KrylovSubspaceOrthogonalizer<Vector_>
198 {
199 public:
200 typedef KrylovSubspaceOrthogonalizer<Vector_> BaseClass;
201 typedef typename BaseClass::VectorType VectorType;
202 typedef typename BaseClass::DataType DataType;
203
204 protected:
206 std::vector<Memory::Arbiter const*> _vec_arbiter;
209 Memory::Arbiter _dots;
210
223 void _compute_dots(const VectorType& target, const Index num_basis, DataType* dot_out, DataType& self_dot)
224 {
225 // zero the accumulation arbiter on all locations, then accumulate the batched dots onto it
226 _dots.format();
227 // the batched dot also picks up the self-dot: target == _subspace[num_basis], so slot num_basis
228 // of the reduction holds <target,target>
229 Intern::BatchedHelper<VectorType>::template batched_dot<DataType>(
230 _dots, target, _vec_arbiter.data(), num_basis + 1, this->_subspace.size());
231
232 // read the results back into the host output (triggers a device->host copy if needed)
233 Memory::TypedView<DataType> dv(_dots.view(Memory::Location::main, Memory::Access::read));
234 const DataType* dp = dv.get_r();
235 for(Index k(0); k < num_basis; ++k)
236 dot_out[k] = dp[k];
237 self_dot = dp[num_basis];
238 }
239
243 DataType _finalize_norm(const DataType nrm_sqr, const VectorType& target) const
244 {
245 // trust the algebraic norm only when it is safely positive; otherwise fall back to an explicit
246 // norm2 (accurate, and guarantees a positive result so the normalization cannot produce a NaN)
247 if(nrm_sqr <= Math::Limits<DataType>::epsilon() * DataType(100))
248 return target.norm2();
249 return Math::sqrt(nrm_sqr);
250 }
251
252 public:
253 explicit ClassicGramSchmidtOrthogonalizer() = default;
254
255 ClassicGramSchmidtOrthogonalizer(const Vector_& vec_ref, Index krylov_dimension) :
256 BaseClass(vec_ref, krylov_dimension)
257 {
258 // one slot per basis vector plus one for the appended self-dot <w,w>; the largest reduction
259 // projects against all krylov_dimension basis vectors, so krylov_dimension+1 slots suffice
260 _dots = Memory::Arbiter((this->_krylov_dimension + 1) * sizeof(DataType));
261 // pass pointers to the persistent krylov basis vectors; gather_arbiter records the addresses
262 // of their (nested) element arbiters, which must remain valid for the solver's lifetime
263 std::vector<const VectorType*> vec_v_ptrs;
264 vec_v_ptrs.reserve(this->_subspace.size());
265 for(const auto& v : this->_subspace)
266 vec_v_ptrs.push_back(&v);
267 Intern::BatchedHelper<VectorType>::gather_arbiter(_vec_arbiter, vec_v_ptrs.data(), this->_subspace.size());
268 }
269
270 ClassicGramSchmidtOrthogonalizer(const ClassicGramSchmidtOrthogonalizer&) = delete;
271 ClassicGramSchmidtOrthogonalizer& operator=(const ClassicGramSchmidtOrthogonalizer&) = delete;
272 ClassicGramSchmidtOrthogonalizer(ClassicGramSchmidtOrthogonalizer&&) = default;
273 ClassicGramSchmidtOrthogonalizer& operator=(ClassicGramSchmidtOrthogonalizer&&) = default;
274
275 virtual ~ClassicGramSchmidtOrthogonalizer()=default;
276
277 DataType orthogonalize(DataType* dot_products, const Index target_index) override
278 {
279 XASSERT(target_index <= this->_krylov_dimension);
280 // orthogonalize against the target_index leading basis vectors v_0,...,v_{target_index-1}
281 const Index num_basis = target_index;
282 VectorType& target = this->_subspace[target_index];
283
284 // a single batched reduction yields the orthogonalization dots h[0..num_basis) and the self-dot
285 DataType self_dot(0);
286 _compute_dots(target, num_basis, dot_products, self_dot);
287
288 DataType sum_sqr(0);
289 for(Index k(0); k < num_basis; ++k)
290 sum_sqr += Math::sqr(dot_products[k]);
291
292 // subtract the projection in a single batched cumulative axpy: target -= sum_k h[k] * v[k]
293 this->correct_vector(target, dot_products, num_basis, DataType(-1));
294
295 // recover the post-orthogonalization norm algebraically (with a norm2 fallback)
296 return _finalize_norm(self_dot - sum_sqr, target);
297 }
298
302 void correct_vector(VectorType& target, const DataType* local_scales, const Index num_axpy, const DataType alpha) const override
303 {
304 XASSERT(num_axpy <= this->_krylov_dimension);
305 Intern::BatchedHelper<VectorType>::template cumulative_axpy<DataType>(
306 target, _vec_arbiter.data(), local_scales, num_axpy, alpha, this->_subspace.size());
307 }
308 };

325 template<typename Vector_>
326 class CGS2Orthogonalizer : public ClassicGramSchmidtOrthogonalizer<Vector_>
327 {
328 public:
329 typedef ClassicGramSchmidtOrthogonalizer<Vector_> BaseClass;
330 typedef typename BaseClass::VectorType VectorType;
331 typedef typename BaseClass::DataType DataType;
332
333 protected:
335 std::vector<DataType> _corr;
338 DataType _reortho_tol = DataType(0.3);
340 bool _force_reortho = false;
341
342 public:
343 explicit CGS2Orthogonalizer() = default;
344
345 CGS2Orthogonalizer(const Vector_& vec_ref, Index krylov_dimension) :
346 BaseClass(vec_ref, krylov_dimension)
347 {
348 _corr.resize(krylov_dimension);
349 }
350
351 CGS2Orthogonalizer(const CGS2Orthogonalizer&) = delete;
352 CGS2Orthogonalizer& operator=(const CGS2Orthogonalizer&) = delete;
353 CGS2Orthogonalizer(CGS2Orthogonalizer&&) = default;
354 CGS2Orthogonalizer& operator=(CGS2Orthogonalizer&&) = default;
355
356 virtual ~CGS2Orthogonalizer()=default;
357
359 void set_reortho_tol(DataType tol)
360 {
361 XASSERT(tol >= DataType(0));
362 _reortho_tol = tol;
363 }
364
366 void set_force_reortho(bool force)
367 {
368 _force_reortho = force;
369 }
370
371 DataType orthogonalize(DataType* dot_products, const Index target_index) override
372 {
373 XASSERT(target_index <= this->_krylov_dimension);
374 const Index num_basis = target_index;
375 VectorType& target = this->_subspace[target_index];
376
377 // first classical Gram-Schmidt sweep (batched dots + self-dot, then batched subtraction)
378 DataType self_dot(0);
379 this->_compute_dots(target, num_basis, dot_products, self_dot);
380 DataType sum_sqr(0);
381 for(Index k(0); k < num_basis; ++k)
382 sum_sqr += Math::sqr(dot_products[k]);
383 this->correct_vector(target, dot_products, num_basis, DataType(-1));
384 DataType nrm_sqr = self_dot - sum_sqr;
385
386 // corrective reorthogonalization (CGS2), forced or when heavy cancellation is detected; the
387 // second batched reduction supplies both the corrections and the updated self-dot, so the
388 // corrected norm is again obtained without an extra reduction
389 if(_force_reortho || (nrm_sqr < Math::sqr(_reortho_tol) * self_dot))
390 {
391 DataType self_dot2(0);
392 this->_compute_dots(target, num_basis, _corr.data(), self_dot2);
393 DataType sum_sqr2(0);
394 for(Index k(0); k < num_basis; ++k)
395 {
396 dot_products[k] += _corr[k];
397 sum_sqr2 += Math::sqr(_corr[k]);
398 }
399 this->correct_vector(target, _corr.data(), num_basis, DataType(-1));
400 nrm_sqr = self_dot2 - sum_sqr2;
401 }
402
403 return this->_finalize_norm(nrm_sqr, target);
404 }
405 };
