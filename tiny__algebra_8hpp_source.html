577 CUDA_HOST_DEVICE Vector& normalize()
578 {
579 #ifndef __CUDACC__
580 const DataType norm2(this->norm_euclid());
581 ASSERTM(norm2 > Math::eps<DataType>(), "Trying to normalize a null vector!");
582 return ((*this) *= (DataType(1)/norm2));
583 #else
584 const DataType norm2_sqr(this->norm_euclid_sqr());
585 ASSERTM(norm2_sqr > CudaMath::cuda_get_eps<DataType>(), "Trying to normalize a null vector!");
586 return ((*this) *= CudaMath::cuda_rsqrt(norm2_sqr));
587 #endif
588 }

3985 T_ det = T_(1);

3989 {
3990 // step 1: find a pivot for the elimination of column k
3991 {

3993 // rows p[j] with j < k have already been eliminated and are
3994 // therefore not candidates for pivoting
3995 #ifdef __CUDACC__

3997 #else
3998 T_ pivot = Math::abs(b(p[k], p[k]));
3999 #endif
4000 int i = k;
4001
4002 // loop over all unprocessed rows
4003 for(int j(k+1); j < n_; ++j)
4004 {
4005 // get our matrix value and check whether it can be a pivot
4006 #ifdef __CUDACC__
4007 T_ abs_val = CudaMath::cuda_abs(b(p[j], p[j]));
4008 #else

4025 }

4035

4037 b(p[k], p[k]) = T_(1);
